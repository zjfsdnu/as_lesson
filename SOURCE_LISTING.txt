; ===========================================================================
; hello_exit.asm —— 第 01 课：一个字节都不输出，只负责"退出"
;
; 用途：退出码（exit status）平时看不见，用 shell 的 $? 才能看到。
;       Linux 里退出码只有最低 8 位有效，所以 0~255。
;
; 运行：
;   ./hello_exit
;   echo $?         # 期望打印 42
; ===========================================================================

default rel

section .text
    global _start

_start:
    mov     rax, 60                 ; rax = 系统调用号；60 == exit
    mov     rdi, 42                 ; 第 1 个参数：退出码 42
    syscall                         ; 进程到此结束，shell 里用 $? 读到的就是 42

; ===========================================================================
; hello.asm —— 第 01 课：最小的 "Hello, World!"（纯系统调用版，不链接 libc）
;
; 构建（本课 Makefile 里就是这么干的）：
;   nasm -f elf64 -g -F dwarf -I ../include -o hello.o hello.asm
;   ld -o hello hello.o
;
; 运行：
;   ./hello
; ===========================================================================

default rel                     ; 内存操作数默认用 RIP 相对寻址（理由见 README「常见坑」）

; ---------------------------------------------------------------------------
; 数据段：程序运行期间要用的数据。这里的数据会原样写进可执行文件。
; ---------------------------------------------------------------------------
section .data
    msg     db  "Hello, World!", 10     ; 13 个可见字符 + 1 个换行符（10 == '\n'）
    msg_len equ $ - msg                 ; 长度 = 当前地址 $ 减去 msg 的起始地址

; ---------------------------------------------------------------------------
; 代码段：CPU 真正执行的指令
; ---------------------------------------------------------------------------
section .text
    global _start                       ; 告诉链接器：程序入口符号叫 _start

_start:
    ; ---- 第 1 步：write(1, msg, msg_len) ----
    mov     rax, 1                      ; rax = 系统调用号；1 == write
    mov     rdi, 1                      ; 第 1 个参数：文件描述符 1 == stdout
    lea     rsi, [rel msg]              ; 第 2 个参数：缓冲区地址（msg 的地址）
    mov     rdx, msg_len                ; 第 3 个参数：要写多少字节（= 14）
    syscall                             ; 进入内核，内核照 rax 里的号码执行 write

    ; ---- 第 2 步：exit(0) ----
    mov     rax, 60                     ; rax = 系统调用号；60 == exit
    xor     edi, edi                    ; 第 1 个参数：退出码 0
                                        ; xor edi,edi 与 mov rdi,0 等价，但机器码更短
    syscall                             ; 退出进程，这条之后不会再有指令执行

; ===========================================================================
; mem.asm —— 第 02 课：程序里的三种数据位置
;
;   1) .data  已初始化数据：有初值，初值会写进可执行文件
;   2) .bss   未初始化数据：可执行文件里只记"要多少字节"，加载时被清零
;   3) 栈     运行期的临时空间：sub rsp, N 开出来，add rsp, N 还回去
;
; 构建/运行：
;   make && ./mem
; ===========================================================================

default rel

; ---------------------------------------------------------------------------
; 已初始化数据段
; ---------------------------------------------------------------------------
section .data
    banner      db  "=== 第 02 课：三种数据位置 .data / .bss / 栈 ===", 10
    banner_len  equ $ - banner

    d_val       dq  0x1122334455667788      ; 8 字节，有初值
    d_msg       db  "d_msg：我在 .data，初值已经写进可执行文件", 10
    d_msg_len   equ $ - d_msg

    t1      db  "[.data] d_val 的初值               = "
    t1_len  equ $ - t1
    t2      db  "[.data] d_msg 的内容               : "
    t2_len  equ $ - t2
    t3      db  "[.bss ] b_num 没赋过值，读出来是   = "
    t3_len  equ $ - t3
    t4      db  "[.bss ] 往 b_num 写 0x2a 再读      = "
    t4_len  equ $ - t4
    t5      db  "[.bss ] b_buf[0] 当数字看          = "
    t5_len  equ $ - t5
    t6      db  "[.bss ] b_buf 当字符串看           = "
    t6_len  equ $ - t6
    t7      db  "[栈   ] sub rsp,16 之后 [rsp]      = "
    t7_len  equ $ - t7
    t8      db  "[栈   ] 同一块空间的 [rsp+8]       = "
    t8_len  equ $ - t8
    t9      db  "[栈   ] add rsp,16 后 (备份rsp-rsp)= "
    t9_len  equ $ - t9
    t9_tail db  "   ← 0 表示栈指针已经复原"
    t9_tail_len equ $ - t9_tail

    ta      db  "[地址 ] d_val(.data) = "
    ta_len  equ $ - ta
    tb      db  "   b_num(.bss) = "
    tb_len  equ $ - tb
    tc      db  "   rsp(栈顶) = "
    tc_len  equ $ - tc
    tnote   db  "（地址每次运行都可能不同，这是正常的）", 10
    tnote_len equ $ - tnote

; ---------------------------------------------------------------------------
; 未初始化数据段：只声明"要多少字节"，不占可执行文件的体积
; ---------------------------------------------------------------------------
section .bss
    b_num   resq 1                  ; 8 字节（1 个 qword）
    b_buf   resb 8                  ; 8 字节

section .text
    global _start
    %include "io.inc"

_start:
    lea     rsi, [rel banner]
    mov     rdx, banner_len
    call    io_print

    ; ======================= .data =======================
    lea     rsi, [rel t1]
    mov     rdx, t1_len
    call    io_print
    mov     rdi, [rel d_val]            ; 从 .data 里读 8 字节
    call    io_print_hex
    call    io_newline

    lea     rsi, [rel t2]
    mov     rdx, t2_len
    call    io_print
    lea     rsi, [rel d_msg]
    mov     rdx, d_msg_len
    call    io_print                    ; d_msg 自带换行，不用再补

    ; ======================= .bss ========================
    lea     rsi, [rel t3]
    mov     rdx, t3_len
    call    io_print
    mov     rdi, [rel b_num]            ; 从来没写过，读出来一定是 0
    call    io_print_hex
    call    io_newline

    mov     qword [rel b_num], 0x2a     ; 往 .bss 里写 8 字节
    lea     rsi, [rel t4]
    mov     rdx, t4_len
    call    io_print
    mov     rdi, [rel b_num]
    call    io_print_hex
    call    io_newline

    mov     byte [rel b_buf], 0x41      ; 'A'
    mov     byte [rel b_buf + 1], 0x42  ; 'B'
    mov     byte [rel b_buf + 2], 0     ; 字符串结尾的 0

    lea     rsi, [rel t5]
    mov     rdx, t5_len
    call    io_print
    movzx   rdi, byte [rel b_buf]       ; 只读 1 个字节
    call    io_print_hex                ; 0x41
    call    io_newline

    lea     rsi, [rel t6]
    mov     rdx, t6_len
    call    io_print
    lea     rsi, [rel b_buf]
    mov     edx, 2                      ; 只打印 "AB" 这 2 个字节
    call    io_print
    call    io_newline

    ; ======================= 栈 ==========================
    mov     r12, rsp                    ; 备份 rsp（r12 不会被 io.inc 破坏）
    sub     rsp, 16                     ; 在栈上开 16 字节的临时空间

    mov     qword [rsp], 0x1111         ; 第 1 个 8 字节槽
    mov     qword [rsp + 8], 0x2222     ; 第 2 个 8 字节槽

    lea     rsi, [rel t7]
    mov     rdx, t7_len
    call    io_print
    mov     rdi, [rsp]                  ; 从栈上读回来
    call    io_print_hex                ; 0x1111
    call    io_newline

    lea     rsi, [rel t8]
    mov     rdx, t8_len
    call    io_print
    mov     rdi, [rsp + 8]              ; 也读回来
    call    io_print_hex                ; 0x2222
    call    io_newline

    add     rsp, 16                     ; 把栈还回去（sub/add 必须成对）

    lea     rsi, [rel t9]
    mov     rdx, t9_len
    call    io_print
    mov     rax, r12                    ; 备份的 rsp
    sub     rax, rsp                    ; 减去当前 rsp
    mov     rdi, rax
    call    io_print_hex                ; 差是 0x0 就说明复原了
    lea     rsi, [rel t9_tail]
    mov     rdx, t9_tail_len
    call    io_print
    call    io_newline

    ; ======================= 地址对比 ====================
    lea     rsi, [rel ta]
    mov     rdx, ta_len
    call    io_print
    lea     rdi, [rel d_val]            ; .data 里 d_val 的地址
    call    io_print_hex

    lea     rsi, [rel tb]
    mov     rdx, tb_len
    call    io_print
    lea     rdi, [rel b_num]            ; .bss 里 b_num 的地址
    call    io_print_hex

    lea     rsi, [rel tc]
    mov     rdx, tc_len
    call    io_print
    mov     rdi, rsp                    ; 当前栈顶
    call    io_print_hex
    call    io_newline

    lea     rsi, [rel tnote]
    mov     rdx, tnote_len
    call    io_print

    xor     edi, edi
    call    io_exit

; ===========================================================================
; regs.asm —— 第 02 课：通用寄存器的宽度（64/32/16/8 位别名）
;
; 这个程序演示三件事：
;   1. rax / eax / ax / al / ah 指的是同一个物理寄存器，只是"窗口"大小不同
;   2. 写 32 位（eax）会自动把高 32 位清零；写 16/8 位（ax/al/ah）不会
;   3. movzx（零扩展）与 movsx（符号扩展）的区别
;
; 构建/运行：
;   make && ./regs
; ===========================================================================

default rel

section .data
    banner      db  "=== 第 02 课：寄存器宽度演示 ===", 10
    banner_len  equ $ - banner

    s1      db  "1) mov rax, 0x1122334455667788          -> rax = "
    s1_len  equ $ - s1
    s2      db  "2) 再 mov ax, 0xaabb（只动低 16 位）    -> rax = "
    s2_len  equ $ - s2
    s3      db  "3) 重来再 mov al, 0xcc（只动低 8 位）   -> rax = "
    s3_len  equ $ - s3
    s4      db  "4) 重来 mov eax, 0xdeadbeef（清高 32）  -> rax = "
    s4_len  equ $ - s4
    s5      db  "5) 接上一步 mov ah, 0x77（第 2 字节）   -> rax = "
    s5_len  equ $ - s5
    s6a     db  "6) rax = -1，当有符号数看               -> "
    s6a_len equ $ - s6a
    s6b     db  "   rax = -1，当无符号数看               -> "
    s6b_len equ $ - s6b
    s7a     db  "7) mov al,0xFF 后 movzx rax,al          -> "
    s7a_len equ $ - s7a
    s7b     db  "   mov al,0xFF 后 movsx rax,al          -> "
    s7b_len equ $ - s7b

section .text
    global _start
    %include "io.inc"

_start:
    ; ---------------- 标题 ----------------
    lea     rsi, [rel banner]
    mov     rdx, banner_len
    call    io_print

    ; ---------------- 1. 64 位写入 ----------------
    lea     rsi, [rel s1]
    mov     rdx, s1_len
    call    io_print                    ; 先输出说明文字
    mov     rax, 0x1122334455667788      ; 64 位立即数（NASM 会用 movabs 编码）
    mov     rdi, rax                    ; io_print_hex 的参数在 rdi
    call    io_print_hex                ; 0x1122334455667788
    call    io_newline

    ; ---------------- 2. 只写低 16 位 ----------------
    lea     rsi, [rel s2]
    mov     rdx, s2_len
    call    io_print

    mov     rax, 0x1122334455667788      ; 先恢复初值
    mov     ax, 0xaabb                  ; 只改最低 16 位，高 48 位不动
    mov     rdi, rax
    call    io_print_hex                ; 0x112233445566aabb
    call    io_newline

    ; ---------------- 3. 只写低 8 位 ----------------
    lea     rsi, [rel s3]
    mov     rdx, s3_len
    call    io_print

    mov     rax, 0x1122334455667788
    mov     al, 0xcc                    ; 只改最低 8 位
    mov     rdi, rax
    call    io_print_hex                ; 0x11223344556677cc
    call    io_newline

    ; ---------------- 4. 写 32 位会清零高 32 位 ----------------
    lea     rsi, [rel s4]
    mov     rdx, s4_len
    call    io_print

    mov     rax, 0x1122334455667788
    mov     eax, 0xdeadbeef             ; 写 eax ⇒ rax 的高 32 位自动变成 0
    mov     rdi, rax
    call    io_print_hex                ; 0xdeadbeef
    call    io_newline

    ; ---------------- 5. ah 是第 2 个字节 ----------------
    lea     rsi, [rel s5]
    mov     rdx, s5_len
    call    io_print

    mov     rax, 0x1122334455667788
    mov     eax, 0xdeadbeef
    mov     ah, 0x77                    ; ah = rax 的 bit8..bit15
    mov     rdi, rax
    call    io_print_hex                ; 0xdead77ef
    call    io_newline

    ; ---------------- 6. 同一个位模式，两种解释 ----------------
    lea     rsi, [rel s6a]
    mov     rdx, s6a_len
    call    io_print

    mov     rax, -1                     ; 64 个二进制 1
    mov     rdi, rax
    call    io_print_int                ; 有符号解释：-1
    call    io_newline

    lea     rsi, [rel s6b]
    mov     rdx, s6b_len
    call    io_print

    mov     rax, -1
    mov     rdi, rax
    call    io_print_uint               ; 无符号解释：18446744073709551615
    call    io_newline

    ; ---------------- 7. movzx 与 movsx ----------------
    lea     rsi, [rel s7a]
    mov     rdx, s7a_len
    call    io_print

    mov     rax, 0x1122334455667788
    mov     al, 0xFF                    ; al = 0xFF，rax 其余位还是原来那堆
    movzx   rax, al                     ; 零扩展：高 56 位全部清零
    mov     rdi, rax
    call    io_print_int                ; 255
    call    io_newline

    lea     rsi, [rel s7b]
    mov     rdx, s7b_len
    call    io_print

    mov     rax, 0x1122334455667788
    mov     al, 0xFF
    movsx   rax, al                     ; 符号扩展：0xFF 当成 -1
    mov     rdi, rax
    call    io_print_int                ; -1
    call    io_newline

    ; ---------------- 结束 ----------------
    xor     edi, edi                    ; 退出码 0
    call    io_exit                     ; io_exit 不会返回

; ===========================================================================
; data.asm —— 第 03 课：数据定义 db / dw / dd / dq / equ / times
;
;   db = Define Byte      （1 字节）
;   dw = Define Word      （2 字节，注意 x86 里 "word" 是 2 字节不是 4 字节）
;   dd = Define Doubleword（4 字节）
;   dq = Define Quadword  （8 字节）
;   equ  = 汇编期常量，不占内存
;   times = 把同一份数据重复 N 次
;
; 构建/运行：
;   make && ./data
; ===========================================================================

default rel

section .data
    banner      db  "=== 第 03 课：数据定义 db/dw/dd/dq/equ/times ===", 10
    banner_len  equ $ - banner

    ; ---- db：一个字节一个字节地摆 ----
    b1      db  0x41, 0x42, 'C', 10     ; 'A' 'B' 'C' 换行（0x41 == 65 == 'A'）
    b1_len  equ $ - b1                  ; = 4（含那个换行）

    ; ---- dw：2 字节 ----
    w1      dw  0x1234

    ; ---- dd：4 字节 ----
    d1      dd  0x89abcdef

    ; ---- dq：8 字节 ----
    q1      dq  0x1122334455667788
q1_end:                                 ; 紧跟在 q1 后面，用来算大小
    Q1_LEN  equ q1_end - q1             ; = 8

    ; ---- 字符串 + 结尾 0，长度用 equ 算出来 ----
    str1        db  "abc", 0
    str1_len    equ $ - str1 - 1        ; = 3（不减 1 会把结尾的 0 也算进去）

    ; ---- times：重复 ----
    stars   times 8 db 0x2a             ; 8 个 '*'（0x2a == 42 == '*'）
    pair    times 3 db "ab"             ; "ab" 重复 3 次 = "ababab"（6 字节）

    ; ---- 下面都是输出用的小标签 ----
    t_db    db  "[db   ] b1 的 4 个字节原样输出   : "
    t_db_len equ $ - t_db
    t_db0   db  "[db   ] b1[0] 当数字看           = "
    t_db0_len equ $ - t_db0
    t_dw    db  "[dw   ] w1（2 字节）             = "
    t_dw_len equ $ - t_dw
    t_dd    db  "[dd   ] d1（4 字节，mov edi 读） = "
    t_dd_len equ $ - t_dd
    t_dd2   db  "[dd   ] d1 当有符号 32 位看      = "
    t_dd2_len equ $ - t_dd2
    t_dq    db  "[dq   ] q1（8 字节）             = "
    t_dq_len equ $ - t_dq
    t_equ   db  "[equ  ] Q1_LEN = q1_end - q1     = "
    t_equ_len equ $ - t_equ
    t_str   db  "[str  ] str1 的内容              = "
    t_str_len equ $ - t_str
    t_str2  db  "  长度 = "
    t_str2_len equ $ - t_str2
    t_tm1   db  "[times] stars = 8 个 0x2a        = "
    t_tm1_len equ $ - t_tm1
    t_tm2   db  '[times] pair   = 3 组 "ab"       = '  ; NASM 没有 "" 转义，改用单引号
    t_tm2_len equ $ - t_tm2

section .text
    global _start
    %include "io.inc"

_start:
    lea     rsi, [rel banner]
    mov     rdx, banner_len
    call    io_print

    ; ---------------- db：原样输出 4 个字节 ----------------
    lea     rsi, [rel t_db]
    mov     rdx, t_db_len
    call    io_print
    lea     rsi, [rel b1]
    mov     rdx, b1_len
    call    io_print                    ; 输出 "ABC" 加上 b1 里自带的换行

    ; db 的字节也能当数字用
    lea     rsi, [rel t_db0]
    mov     rdx, t_db0_len
    call    io_print
    movzx   rdi, byte [rel b1]          ; 只取 1 个字节，零扩展成 64 位
    call    io_print_int                ; 65
    call    io_newline

    ; ---------------- dw ----------------
    lea     rsi, [rel t_dw]
    mov     rdx, t_dw_len
    call    io_print
    movzx   rdi, word [rel w1]          ; 取 2 个字节
    call    io_print_hex                ; 0x1234
    call    io_newline

    ; ---------------- dd ----------------
    lea     rsi, [rel t_dd]
    mov     rdx, t_dd_len
    call    io_print
    mov     edi, [rel d1]               ; 读 4 字节到 edi，高 32 位自动补 0
    call    io_print_hex                ; 0x89abcdef
    call    io_newline

    ; 同样 4 字节，按"有符号"扩展到 64 位就是另一个数
    lea     rsi, [rel t_dd2]
    mov     rdx, t_dd2_len
    call    io_print
    movsxd  rdi, dword [rel d1]         ; 符号扩展
    call    io_print_int                ; -1985229329
    call    io_newline

    ; ---------------- dq ----------------
    lea     rsi, [rel t_dq]
    mov     rdx, t_dq_len
    call    io_print
    mov     rdi, [rel q1]               ; 读 8 字节
    call    io_print_hex                ; 0x1122334455667788
    call    io_newline

    ; ---------------- equ：汇编期常量，不占内存 ----------------
    lea     rsi, [rel t_equ]
    mov     rdx, t_equ_len
    call    io_print
    mov     rdi, Q1_LEN                 ; 汇编时就被替换成 8
    call    io_print_int                ; 8
    call    io_newline

    ; ---------------- 字符串与它的长度 ----------------
    lea     rsi, [rel t_str]
    mov     rdx, t_str_len
    call    io_print
    lea     rsi, [rel str1]
    mov     rdx, str1_len
    call    io_print                    ; abc
    lea     rsi, [rel t_str2]
    mov     rdx, t_str2_len
    call    io_print
    mov     rdi, str1_len
    call    io_print_int                ; 3
    call    io_newline

    ; ---------------- times ----------------
    lea     rsi, [rel t_tm1]
    mov     rdx, t_tm1_len
    call    io_print
    lea     rsi, [rel stars]
    mov     edx, 8
    call    io_print                    ; ********
    call    io_newline

    lea     rsi, [rel t_tm2]
    mov     rdx, t_tm2_len
    call    io_print
    lea     rsi, [rel pair]
    mov     edx, 6
    call    io_print                    ; ababab
    call    io_newline

    xor     edi, edi
    call    io_exit

; ===========================================================================
; strlen.asm —— 第 03 课：自己写循环数字符串长度（完全不碰 libc）
;
; C 里一句 strlen(s) 背后就是一个循环：从字符串开头一个字节一个字节往后看，
; 遇到结尾的 0 就停下，走过的字节数就是长度。这里把它写成汇编。
;
; 输出格式：字符串 [原文] 长度 = N
;
; 构建/运行：
;   make && ./strlen
; ===========================================================================

default rel

section .data
    banner      db  "=== 第 03 课：自己数长度 my_strlen ===", 10
    banner_len  equ $ - banner

    p_pre       db  "字符串 ["
    p_pre_len   equ $ - p_pre
    p_mid       db  "] 长度 = "
    p_mid_len   equ $ - p_mid

    ; 四个测试样本：普通串、空串、带连字符的串、含中文的串
    s1      db  "Hello", 0
    s2      db  0                       ; 空字符串 == 只有一个结尾 0
    s3      db  "x86-64 assembly", 0
    s4      db  "长度是按字节数的，不是按字符", 0

section .text
    global _start
    %include "io.inc"

_start:
    lea     rsi, [rel banner]
    mov     rdx, banner_len
    call    io_print

    lea     rdi, [rel s1]               ; 参数：字符串地址
    call    show_len

    lea     rdi, [rel s2]
    call    show_len

    lea     rdi, [rel s3]
    call    show_len

    lea     rdi, [rel s4]
    call    show_len

    xor     edi, edi
    call    io_exit

; ---------------------------------------------------------------------------
; show_len —— 打印一行「字符串 [原文] 长度 = N」
; 入参：rdi = 以 0 结尾的字符串地址
; ---------------------------------------------------------------------------
show_len:
    push    r12                         ; io.inc 保证 rbx/rbp/r12-r15 不被破坏，
    push    r13                         ; 所以用它俩把数据带过若干次 io_* 调用
    mov     r13, rdi                    ; r13 = 字符串首地址

    call    my_strlen                   ; 返回长度在 rax
    mov     r12, rax                    ; r12 = 长度

    lea     rsi, [rel p_pre]
    mov     rdx, p_pre_len
    call    io_print                    ; 「字符串 [」

    mov     rsi, r13                    ; 原文首地址
    mov     rdx, r12                    ; 原文长度（正是刚刚数出来的）
    call    io_print                    ; 输出原文

    lea     rsi, [rel p_mid]
    mov     rdx, p_mid_len
    call    io_print                    ; 「] 长度 = 」

    mov     rdi, r12
    call    io_print_int                ; 输出长度

    call    io_newline

    pop     r13
    pop     r12
    ret

; ---------------------------------------------------------------------------
; my_strlen —— 数到第一个 0 为止，返回"不含结尾 0"的长度
; 入参：rdi = 字符串地址        返回：rax = 长度
; 只使用 rax / rdi，不碰其它寄存器
; ---------------------------------------------------------------------------
my_strlen:
    xor     eax, eax                    ; rax = 0，长度计数器（xor 同时清掉高 32 位）
.loop:
    cmp     byte [rdi], 0               ; 看 rdi 指向的那个字节是不是 0
    je      .done                       ; 是 0 → 数完了
    inc     rax                         ; 长度 + 1
    inc     rdi                         ; 指针往后挪一个字节
    jmp     .loop                       ; 回到循环开头继续看
.done:
    ret                                 ; 长度在 rax 里返回

; ===========================================================================
; arith.asm —— 算术指令：add / sub / imul / div / idiv / inc / dec / neg
;
; 纯系统调用程序：入口 _start，nasm + ld 直接链接，不依赖 libc。
; 运行：  make && ./arith
;
; 重点：div / idiv 的被除数是 rdx:rax（128 位）。做 64 位除法之前，
;       无符号要 xor rdx,rdx，有符号要 cqo —— 忘了就是本文件最后一段的下场。
; ===========================================================================
default rel

section .data
    val_a   dq 15                   ; 用内存里的两个数做算术，顺便复习 [rel 标号]
    val_b   dq 27

    ; 提示文字；长度一律用 equ $ - 标号 算出来，不手数
    t_add    db "add:     15 + 27               = "
    l_add    equ $ - t_add
    t_sub    db "sub:     15 - 27               = "
    l_sub    equ $ - t_sub
    t_mul    db "imul:    15 * 27               = "
    l_mul    equ $ - t_mul
    t_bigmul db "imul:    1000000 * 1000000     = "
    l_bigmul equ $ - t_bigmul
    t_mul128 db "imul:    (2^32)*(2^32) rdx:rax = "
    l_mul128 equ $ - t_mul128
    t_colon  db ":"
    l_colon  equ $ - t_colon
    t_div    db "div:     100 / 7               = "
    l_div    equ $ - t_div
    t_rem    db "  rem "
    l_rem    equ $ - t_rem
    t_idiv   db "idiv:    -100 / 7              = "
    l_idiv   equ $ - t_idiv
    t_inc    db "inc:     41 + 1                = "
    l_inc    equ $ - t_inc
    t_dec    db "dec:     42 - 1                = "
    l_dec    equ $ - t_dec
    t_neg    db "neg:     -(5)                  = "
    l_neg    equ $ - t_neg
    t_hi     db "div:     2^64 / 10             = "
    l_hi     equ $ - t_hi
    t_bug    db "div:     100 / 7 (rdx=1)       = "
    l_bug    equ $ - t_bug

section .text
    global _start
%include "io.inc"                   ; 只能在 .text 里、_start 之前 include 一次

_start:
; ---- 1) add：内存操作数直接参与运算，宽度由 dq 决定 ----
    lea  rsi, [rel t_add]
    mov  rdx, l_add
    call io_print
    mov  rax, [rel val_a]
    add  rax, [rel val_b]           ; 15 + 27 = 42
    mov  rdi, rax
    call io_print_int
    call io_newline

; ---- 2) sub：15 - 27 得到负数的补码 0xfffffffffffffff4 ----
    lea  rsi, [rel t_sub]
    mov  rdx, l_sub
    call io_print
    mov  rax, [rel val_a]
    sub  rax, [rel val_b]           ; = -12
    mov  rdi, rax
    call io_print_int
    call io_newline

; ---- 3) imul：两个操作数的写法只保留低 64 位乘积，rdx 不变 ----
    lea  rsi, [rel t_mul]
    mov  rdx, l_mul
    call io_print
    mov  rax, [rel val_a]
    mov  rbx, [rel val_b]
    imul rax, rbx                   ; 15 * 27 = 405
    mov  rdi, rax
    call io_print_int
    call io_newline

    lea  rsi, [rel t_bigmul]
    mov  rdx, l_bigmul
    call io_print
    mov  rax, 1000000
    mov  rbx, 1000000
    imul rax, rbx                   ; 10^12，64 位放得下
    mov  rdi, rax
    call io_print_int
    call io_newline

; 一个操作数的 imul：rdx:rax = rax * 操作数（完整 128 位乘积）
    lea  rsi, [rel t_mul128]
    mov  rdx, l_mul128
    call io_print
    mov  rax, 0x100000000           ; 2^32
    mov  rbx, 0x100000000
    imul rbx                        ; 2^32 * 2^32 = 2^64 → rdx=1, rax=0
    mov  r12, rax                   ; 低位先搬走：后面 io 例程会破坏 rdx
    mov  r13, rdx                   ; 高位（r12/r13 不会被 io 例程破坏）
    mov  rdi, r13
    call io_print_uint
    lea  rsi, [rel t_colon]
    mov  rdx, l_colon
    call io_print
    mov  rdi, r12
    call io_print_uint
    call io_newline

; ---- 4) div：无符号除法。商在 rax、余数在 rdx，做 64 位除法前必须清 rdx ----
    lea  rsi, [rel t_div]
    mov  rdx, l_div
    call io_print
    mov  rax, 100
    xor  rdx, rdx                   ; ★ 被除数高 64 位清 0，否则商完全不对
    mov  rbx, 7
    div  rbx                        ; rax = 14（商），rdx = 2（余数）
    mov  r12, rdx                   ; 余数先搬走：io_print_int 会破坏 rdx
    mov  rdi, rax
    call io_print_int
    lea  rsi, [rel t_rem]
    mov  rdx, l_rem
    call io_print
    mov  rdi, r12
    call io_print_int
    call io_newline

; ---- 5) idiv：有符号除法，除法前用 cqo 把 rax 的符号位扩展到 rdx ----
    lea  rsi, [rel t_idiv]
    mov  rdx, l_idiv
    call io_print
    mov  rax, -100
    cqo                             ; ★ rdx:rax = 符号扩展后的 -100
    mov  rbx, 7
    idiv rbx                        ; rax = -14，rdx = -2（余数符号跟被除数）
    mov  r12, rdx
    mov  rdi, rax
    call io_print_int
    lea  rsi, [rel t_rem]
    mov  rdx, l_rem
    call io_print
    mov  rdi, r12
    call io_print_int
    call io_newline

; ---- 6) inc / dec：加一 / 减一，会改 ZF/SF/OF，但不碰 CF ----
    lea  rsi, [rel t_inc]
    mov  rdx, l_inc
    call io_print
    mov  rax, 41
    inc  rax                        ; 42
    mov  rdi, rax
    call io_print_int
    call io_newline

    lea  rsi, [rel t_dec]
    mov  rdx, l_dec
    call io_print
    mov  rax, 42
    dec  rax                        ; 41
    mov  rdi, rax
    call io_print_int
    call io_newline

; ---- 7) neg：取相反数，等价于 0 - x，也等价于 not x 再 +1 ----
    lea  rsi, [rel t_neg]
    mov  rdx, l_neg
    call io_print
    mov  rax, 5
    neg  rax                        ; -5
    mov  rdi, rax
    call io_print_int
    call io_newline

; ---- 8) 证明 rdx 就是「被除数的高 64 位」：rdx=1 时算的是 2^64 / 10 ----
    lea  rsi, [rel t_hi]
    mov  rdx, l_hi
    call io_print
    xor  eax, eax                   ; 低 64 位 = 0
    mov  rdx, 1                     ; 高 64 位 = 1 → 被除数 = 2^64
    mov  rbx, 10
    div  rbx
    mov  r12, rdx
    mov  rdi, rax
    call io_print_uint              ; 商 1844674407370955161
    lea  rsi, [rel t_rem]
    mov  rdx, l_rem
    call io_print
    mov  rdi, r12
    call io_print_uint              ; 余 6
    call io_newline

; ---- 9) 反面教材：忘了清 rdx（这里故意留成 1，保证输出可复现）。
;         真实 bug 里 rdx 是上次除法的余数或别的残留值，症状是「商大得离谱」 ----
    lea  rsi, [rel t_bug]
    mov  rdx, l_bug
    call io_print
    mov  rax, 100
    mov  rdx, 1                     ; ← 忘了清零！被除数变成 2^64 + 100
    mov  rbx, 7
    div  rbx                        ; 商 2635249153387078816，余 4
    mov  r12, rdx
    mov  rdi, rax
    call io_print_uint
    lea  rsi, [rel t_rem]
    mov  rdx, l_rem
    call io_print
    mov  rdi, r12
    call io_print_uint
    call io_newline

; ---- 结束进程：io_exit 内部是 mov rax,60 / syscall，不返回 ----
    xor  edi, edi
    call io_exit

; ===========================================================================
; bitops.asm —— 位运算：and / or / xor / not / shl / shr / sar
;
; 纯系统调用程序：入口 _start。运行：  make && ./bitops
;
; 移位次数只有两种写法：shl rax, 4（立即数）或 shl rax, cl（寄存器）。
; 寄存器形式**必须**用 cl，别的寄存器不行；64 位下只取 cl 的低 6 位，
; 所以 cl=65 实际只移 1 位。
; ===========================================================================
default rel

section .data
    t_and  db "and:  0x1234abcd & 0x0000ffff = "
    l_and  equ $ - t_and
    t_or   db "or:   0x1000 | 0x0001         = "
    l_or   equ $ - t_or
    t_xor  db "xor:  0b1100 ^ 0b1010         = "
    l_xor  equ $ - t_xor
    t_not  db "not:  ~0x0f                   = "
    l_not  equ $ - t_not
    t_shl4 db "shl:  1 << 4                  = "
    l_shl4 equ $ - t_shl4
    t_shl63 db "shl:  1 << 63                 = "
    l_shl63 equ $ - t_shl63
    t_shr  db "shr:  0x8000000000000000 >> 3 = "
    l_shr  equ $ - t_shr
    t_sar  db "sar:  -16 >> 2 (arith)        = "
    l_sar  equ $ - t_sar
    t_shrn db "shr:  -16 >> 2 (logical)      = "
    l_shrn equ $ - t_shrn
    t_cl8  db "shl:  1 << cl (cl = 8)        = "
    l_cl8  equ $ - t_cl8
    t_cl65 db "shl:  1 << cl (cl = 65)       = "
    l_cl65 equ $ - t_cl65
    t_zero db "xor:  eax ^= eax (zero idiom) = "
    l_zero equ $ - t_zero

section .text
    global _start
%include "io.inc"

_start:
; ---- 1) and：按位与，用来「取某些位」（掩码 mask），test 指令内部就是它 ----
    lea  rsi, [rel t_and]
    mov  rdx, l_and
    call io_print
    mov  rax, 0x1234abcd
    and  rax, 0x0000ffff            ; 只保留低 16 位 → 0xabcd
    mov  rdi, rax
    call io_print_hex
    call io_newline

; ---- 2) or：按位或，用来「置某些位」 ----
    lea  rsi, [rel t_or]
    mov  rdx, l_or
    call io_print
    mov  rax, 0x1000
    or   rax, 0x0001                ; 最低位置 1 → 0x1001
    mov  rdi, rax
    call io_print_hex
    call io_newline

; ---- 3) xor：相同为 0、不同为 1；0b1100 ^ 0b1010 = 0b0110 ----
    lea  rsi, [rel t_xor]
    mov  rdx, l_xor
    call io_print
    mov  rax, 0b1100
    xor  rax, 0b1010                ; → 0b110（去掉了前导 0）
    mov  rdi, rax
    call io_print_bin
    call io_newline

; ---- 4) not：按位取反，64 位寄存器就是全部 64 位取反 ----
    lea  rsi, [rel t_not]
    mov  rdx, l_not
    call io_print
    mov  rax, 0x0f
    not  rax                        ; 0xfffffffffffffff0
    mov  rdi, rax
    call io_print_hex
    call io_newline

; ---- 5) shl：逻辑左移，低位补 0，移出去的最高位进 CF ----
    lea  rsi, [rel t_shl4]
    mov  rdx, l_shl4
    call io_print
    mov  rax, 1
    shl  rax, 4                     ; 1 << 4 = 16
    mov  rdi, rax
    call io_print_uint
    call io_newline

    lea  rsi, [rel t_shl63]
    mov  rdx, l_shl63
    call io_print
    mov  rax, 1
    shl  rax, 63                    ; 1 << 63 = 0x8000000000000000
    mov  rdi, rax
    call io_print_hex
    call io_newline

; ---- 6) shr：逻辑右移，高位一律补 0（把操作数当无符号数看） ----
    lea  rsi, [rel t_shr]
    mov  rdx, l_shr
    call io_print
    mov  rax, 0x8000000000000000
    shr  rax, 3                     ; → 0x1000000000000000
    mov  rdi, rax
    call io_print_hex
    call io_newline

; ---- 7) sar：算术右移，高位补符号位。
;         对比：同一个 -16 用 shr 会变成一个很大的正数 ----
    lea  rsi, [rel t_sar]
    mov  rdx, l_sar
    call io_print
    mov  rax, -16                   ; 0xfffffffffffffff0
    sar  rax, 2                     ; -16 / 4 = -4
    mov  rdi, rax
    call io_print_int
    call io_newline

    lea  rsi, [rel t_shrn]
    mov  rdx, l_shrn
    call io_print
    mov  rax, -16
    shr  rax, 2                     ; 高位补 0 → 0x3ffffffffffffffc
    mov  rdi, rax
    call io_print_uint              ; 4611686018427387900
    call io_newline

; ---- 8) 移位次数放在 cl 里：cl=65 只取低 6 位（65 & 63 = 1），等于左移 1 位 ----
    lea  rsi, [rel t_cl8]
    mov  rdx, l_cl8
    call io_print
    mov  rax, 1
    mov  cl, 8                      ; 只用到 cl，rcx 高位是什么都不影响
    shl  rax, cl                    ; 1 << 8 = 256
    mov  rdi, rax
    call io_print_uint
    call io_newline

    lea  rsi, [rel t_cl65]
    mov  rdx, l_cl65
    call io_print
    mov  rax, 1
    mov  cl, 65                     ; 65 & 63 = 1
    shl  rax, cl                    ; 实际只左移 1 位 → 2
    mov  rdi, rax
    call io_print_uint
    call io_newline

; ---- 9) xor 自己 = 清零：x86 上把寄存器置 0 的推荐写法（比 mov 短，
;         而且不依赖寄存器的旧值；副作用是 ZF=1、CF=OF=0） ----
    lea  rsi, [rel t_zero]
    mov  rdx, l_zero
    call io_print
    xor  eax, eax                   ; rax = 0（写 32 位寄存器会顺带清高 32 位）
    mov  rdi, rax
    call io_print_uint
    call io_newline

; ---- 结束进程 ----
    xor  edi, edi
    call io_exit

; ===========================================================================
; flags.asm —— 观察 rflags 里的 ZF / CF / SF / OF
;
;   * 算术、逻辑、比较指令都会按结果重设标志位（flag）
;   * pushfq 把整个 rflags 压栈，pop 到通用寄存器后就能逐位检查
;   * setz / setc / sets / seto 把单个标志位搬进寄存器的低 8 位
;
; 纯系统调用程序：入口 _start。运行：  make && ./flags
; ===========================================================================
default rel

%define FLAG_MASK 0x8c1             ; CF=bit0、ZF=bit6、SF=bit7、OF=bit11

section .data
    ; 第一组：用 setcc 逐个打印标志位的名字
    c1 db "sub 5 - 5            -> "
    lc1 equ $ - c1
    c2 db "sub 1 - 2            -> "
    lc2 equ $ - c2
    c3 db "add 0x7fff..ff + 1   -> "
    lc3 equ $ - c3

    s_zf db "ZF="
    l_zf equ $ - s_zf
    s_cf db "  CF="
    l_cf equ $ - s_cf
    s_sf db "  SF="
    l_sf equ $ - s_sf
    s_of db "  OF="
    l_of equ $ - s_of

    ; 第二组：用 pushfq 直接看 rflags 的二进制
    c4 db "add -1 + 1           -> rflags & 0x8c1 = "
    lc4 equ $ - c4
    c5 db "test 0x0f & 0xf0     -> rflags & 0x8c1 = "
    lc5 equ $ - c5
    c6 db "shl 0x8000..00 by 1  -> rflags & 0x8c1 = "
    lc6 equ $ - c6
    legend db "   [bit0=CF bit6=ZF bit7=SF bit11=OF]", 10
    l_legend equ $ - legend

section .text
    global _start
%include "io.inc"

_start:
; ===========================================================================
; 第一组：setcc 把四个标志位分别写进 r12b / r13b / r14b / r15b
;   setz（同义 sete）：ZF=1 写 1；setc（同义 setb）：CF=1 写 1；
;   sets：SF=1 写 1；seto：OF=1 写 1。
; 打印时用 movzx edi, r12b 零扩展再打印：setcc 只写低 8 位，高 56 位是垃圾值。
; ===========================================================================

; ---- 例 1：5 - 5 = 0 → ZF=1；无进位/借位，SF、OF 都是 0 ----
    lea  rsi, [rel c1]
    mov  rdx, lc1
    call io_print
    mov  rax, 5
    sub  rax, 5                     ; ← 被观察的指令
    setz r12b
    setc r13b
    sets r14b
    seto r15b
    lea  rsi, [rel s_zf]
    mov  rdx, l_zf
    call io_print
    movzx edi, r12b
    call io_print_uint
    lea  rsi, [rel s_cf]
    mov  rdx, l_cf
    call io_print
    movzx edi, r13b
    call io_print_uint
    lea  rsi, [rel s_sf]
    mov  rdx, l_sf
    call io_print
    movzx edi, r14b
    call io_print_uint
    lea  rsi, [rel s_of]
    mov  rdx, l_of
    call io_print
    movzx edi, r15b
    call io_print_uint
    call io_newline

; ---- 例 2：1 - 2 = -1 → 借位 CF=1；结果为负 SF=1；两正数相减不溢出 OF=0 ----
    lea  rsi, [rel c2]
    mov  rdx, lc2
    call io_print
    mov  rax, 1
    sub  rax, 2                     ; ← 被观察的指令
    setz r12b
    setc r13b
    sets r14b
    seto r15b
    lea  rsi, [rel s_zf]
    mov  rdx, l_zf
    call io_print
    movzx edi, r12b
    call io_print_uint
    lea  rsi, [rel s_cf]
    mov  rdx, l_cf
    call io_print
    movzx edi, r13b
    call io_print_uint
    lea  rsi, [rel s_sf]
    mov  rdx, l_sf
    call io_print
    movzx edi, r14b
    call io_print_uint
    lea  rsi, [rel s_of]
    mov  rdx, l_of
    call io_print
    movzx edi, r15b
    call io_print_uint
    call io_newline

; ---- 例 3：0x7fffffffffffffff + 1 = 0x8000000000000000
;          无符号无进位 CF=0；有符号「正 + 正 = 负」溢出 OF=1，SF 也随之为 1 ----
    lea  rsi, [rel c3]
    mov  rdx, lc3
    call io_print
    mov  rax, 0x7fffffffffffffff
    add  rax, 1                     ; ← 被观察的指令
    setz r12b
    setc r13b
    sets r14b
    seto r15b
    lea  rsi, [rel s_zf]
    mov  rdx, l_zf
    call io_print
    movzx edi, r12b
    call io_print_uint
    lea  rsi, [rel s_cf]
    mov  rdx, l_cf
    call io_print
    movzx edi, r13b
    call io_print_uint
    lea  rsi, [rel s_sf]
    mov  rdx, l_sf
    call io_print
    movzx edi, r14b
    call io_print_uint
    lea  rsi, [rel s_of]
    mov  rdx, l_of
    call io_print
    movzx edi, r15b
    call io_print_uint
    call io_newline

; ===========================================================================
; 第二组：pushfq / popfq 直接拿完整 rflags
;   pushfq 把 64 位 rflags 压栈（它自己不改标志位），pop rbx 就得到全部标志位；
;   再用 and 只留关心的 4 位，交给 io_print_bin 打成 0bxxxx 的形式。
; ===========================================================================

; ---- 例 4：-1 + 1 = 0，加法产生进位 CF=1、结果为 0 所以 ZF=1
;          → 掩码留下 bit0、bit6 = 0b1000001 ----
    lea  rsi, [rel c4]
    mov  rdx, lc4
    call io_print
    mov  rax, -1                    ; 0xffffffffffffffff
    add  rax, 1                     ; ← 被观察的指令
    pushfq
    pop  rbx                        ; rbx = 完整 rflags
    mov  rdi, rbx
    and  rdi, FLAG_MASK             ; 只留 CF/ZF/SF/OF，其余位清 0
    call io_print_bin
    lea  rsi, [rel legend]
    mov  rdx, l_legend
    call io_print                   ; legend 末尾自带 '\n'

; ---- 例 5：test 只做按位与、不写回，且规定 CF=OF=0
;          0x0f & 0xf0 = 0 → ZF=1 → 只剩 bit6 = 0b1000000 ----
    lea  rsi, [rel c5]
    mov  rdx, lc5
    call io_print
    mov  rax, 0x0f
    test rax, 0xf0                  ; ← 被观察的指令
    pushfq
    pop  rbx
    mov  rdi, rbx
    and  rdi, FLAG_MASK
    call io_print_bin
    lea  rsi, [rel legend]
    mov  rdx, l_legend
    call io_print

; ---- 例 6：0x8000000000000000 << 1 = 0，移出的 1 进 CF=1，结果全 0 → ZF=1，
;          OF = 结果最高位 XOR CF = 0 XOR 1 = 1
;          → bit0+bit6+bit11 = 0b100001000001 ----
    lea  rsi, [rel c6]
    mov  rdx, lc6
    call io_print
    mov  rax, 0x8000000000000000
    shl  rax, 1                     ; ← 被观察的指令
    pushfq
    pop  rbx
    mov  rdi, rbx
    and  rdi, FLAG_MASK
    call io_print_bin
    lea  rsi, [rel legend]
    mov  rdx, l_legend
    call io_print

; ---- 结束进程 ----
    xor  edi, edi
    call io_exit

; ===========================================================================
; cmpjmp.asm —— cmp 与条件转移（conditional jump）
;
; 关键点：同一对操作数，用「有符号」还是「无符号」的跳转指令，结果可能相反。
;
;   cmp a, b   实际做 a - b，但不保存结果，只设置标志位：
;       ZF=1  → a == b
;       CF=1  → 无符号角度 a < b（减法发生了借位）
;       有符号的大小关系由 SF 和 OF 共同决定（相等时还要看 ZF）
;
;   无符号用：jb(below) / jbe / ja(above) / jae
;   有符号用：jl(less)  / jle / jg(greater) / jge
;   相等用  ：je / jne（有符号无符号都一样）
;
; 纯系统调用程序：入口 _start。运行：  make && ./cmpjmp
; ===========================================================================
default rel

section .data
    t1 db "cmp   1, 2     : jl    -> "
    l1 equ $ - t1
    t2 db "cmp  -1, 1     : jl    -> "
    l2 equ $ - t2
    t3 db "cmp  -1, 1     : jg    -> "
    l3 equ $ - t3
    t4 db "cmp  -1, 1     : ja    -> "
    l4 equ $ - t4
    t5 db "cmp  -1, 1     : jb    -> "
    l5 equ $ - t5
    t6 db "cmp   5, 5     : je    -> "
    l6 equ $ - t6
    t7 db "cmp   3, 5     : jge   -> "
    l7 equ $ - t7
    t8 db "test  0, 0     : jz    -> "
    l8 equ $ - t8

    ; 判定结论；末尾直接带换行，省掉一次 io_newline
    s_yes db "taken", 10
    l_yes equ $ - s_yes
    s_no  db "not taken", 10
    l_no  equ $ - s_no

section .text
    global _start
%include "io.inc"

_start:
; ---- 测试 1：cmp 1, 2 → 1-2 借位 CF=1、SF=1，有符号 jl「小于」成立 ----
    lea  rsi, [rel t1]
    mov  rdx, l1
    call io_print
    mov  rax, 1
    mov  rbx, 2
    cmp  rax, rbx
    jl   .yes1
    lea  rsi, [rel s_no]
    mov  rdx, l_no
    jmp  .pr1
.yes1:
    lea  rsi, [rel s_yes]
    mov  rdx, l_yes
.pr1:
    call io_print

; ---- 测试 2：cmp -1, 1 → 有符号看：-1 < 1 成立（SF=1、OF=0 → jl 成立） ----
    lea  rsi, [rel t2]
    mov  rdx, l2
    call io_print
    mov  rax, -1
    mov  rbx, 1
    cmp  rax, rbx
    jl   .yes2
    lea  rsi, [rel s_no]
    mov  rdx, l_no
    jmp  .pr2
.yes2:
    lea  rsi, [rel s_yes]
    mov  rdx, l_yes
.pr2:
    call io_print

; ---- 测试 3：同样的 cmp -1, 1 换成 jg（有符号「大于」）→ 不成立 ----
    lea  rsi, [rel t3]
    mov  rdx, l3
    call io_print
    mov  rax, -1
    mov  rbx, 1
    cmp  rax, rbx
    jg   .yes3
    lea  rsi, [rel s_no]
    mov  rdx, l_no
    jmp  .pr3
.yes3:
    lea  rsi, [rel s_yes]
    mov  rdx, l_yes
.pr3:
    call io_print

; ---- 测试 4：★重点★ 同样的 cmp -1, 1 换成 ja（无符号「大于」）→ 成立！
;           无符号把 -1 看成 0xffffffffffffffff = 18446744073709551615，
;           它当然比 1「大」。ja 的条件是 CF=0 且 ZF=0 ----
    lea  rsi, [rel t4]
    mov  rdx, l4
    call io_print
    mov  rax, -1
    mov  rbx, 1
    cmp  rax, rbx
    ja   .yes4
    lea  rsi, [rel s_no]
    mov  rdx, l_no
    jmp  .pr4
.yes4:
    lea  rsi, [rel s_yes]
    mov  rdx, l_yes
.pr4:
    call io_print

; ---- 测试 5：同样的 cmp -1, 1 换成 jb（无符号「小于」）→ 不成立。
;           这就是为什么处理负数数组、负数下标时必须用 jl/jge ----
    lea  rsi, [rel t5]
    mov  rdx, l5
    call io_print
    mov  rax, -1
    mov  rbx, 1
    cmp  rax, rbx
    jb   .yes5
    lea  rsi, [rel s_no]
    mov  rdx, l_no
    jmp  .pr5
.yes5:
    lea  rsi, [rel s_yes]
    mov  rdx, l_yes
.pr5:
    call io_print

; ---- 测试 6：cmp 5, 5 → ZF=1，je 成立 ----
    lea  rsi, [rel t6]
    mov  rdx, l6
    call io_print
    mov  rax, 5
    mov  rbx, 5
    cmp  rax, rbx
    je   .yes6
    lea  rsi, [rel s_no]
    mov  rdx, l_no
    jmp  .pr6
.yes6:
    lea  rsi, [rel s_yes]
    mov  rdx, l_yes
.pr6:
    call io_print

; ---- 测试 7：cmp 3, 5 → jge（有符号「大于等于」）不成立 ----
    lea  rsi, [rel t7]
    mov  rdx, l7
    call io_print
    mov  rax, 3
    mov  rbx, 5
    cmp  rax, rbx
    jge  .yes7
    lea  rsi, [rel s_no]
    mov  rdx, l_no
    jmp  .pr7
.yes7:
    lea  rsi, [rel s_yes]
    mov  rdx, l_yes
.pr7:
    call io_print

;           还会强制 CF=0、OF=0，所以之后只能用 jz/jnz/jl/js 之类看 ZF/SF ----
    lea  rsi, [rel t8]
    mov  rdx, l8
    call io_print
    xor  eax, eax                   ; rax = 0
    test rax, rax
    jz   .yes8
    lea  rsi, [rel s_no]
    mov  rdx, l_no
    jmp  .pr8
.yes8:
    lea  rsi, [rel s_yes]
    mov  rdx, l_yes
.pr8:
    call io_print

; ---- 结束进程 ----
    xor  edi, edi
    call io_exit

; ===========================================================================
; minmax.asm —— 用条件跳转求最大值/最小值
;
;   第一部分：两个数的最大值。同样的两个数分别用「有符号」和「无符号」比较，
;             结果完全不同 —— 这就是 jg 和 ja 的区别。
;   第二部分：遍历 .data 里的 qword 数组，一边打印一边找最大/最小。
;
; 循环的写法下一课（第 06 课）细讲，这里先看懂「比较 → 条件跳转 → 更新」
; 这个三步套路即可。
;
; 纯系统调用程序：入口 _start。运行： make && ./minmax
; ===========================================================================
default rel

section .data
    arr     dq 17, -5, 42, -100, 0, 33, -7
    arr_len equ ($ - arr) / 8       ; 元素个数 = 总字节数 / 每个 qword 8 字节

    t_smax  db "max_signed(-7, 3)   = "
    l_smax  equ $ - t_smax
    t_umax  db "max_unsigned(-7, 3) = "
    l_umax  equ $ - t_umax
    t_udec  db "  (unsigned: "
    l_udec  equ $ - t_udec
    t_udone db ")"
    l_udone equ $ - t_udone

    t_arr   db "array               = ["
    l_arr   equ $ - t_arr
    t_arr_e db "]", 10                 ; ']' 加换行（.data 里还没 include io.inc，
    l_arr_e equ $ - t_arr_e           ;  所以换行写数字 10，不能写 LF 符号）

    t_max   db "max                 = "
    l_max   equ $ - t_max
    t_min   db "min                 = "
    l_min   equ $ - t_min

    s_comma db ", "
    l_comma equ $ - s_comma

section .text
    global _start
%include "io.inc"

_start:
; ---------------------------------------------------------------------------
; 1) 有符号比较：求 max(-7, 3)
;    套路：先假设第一个数是最大值，再拿第二个数跟它比，更大就替换
;    jle = 有符号「小于等于」：3 <= max 就不用替换
; ---------------------------------------------------------------------------
    lea  rsi, [rel t_smax]
    mov  rdx, l_smax
    call io_print

    mov  rax, -7
    mov  rbx, 3
    mov  r12, rax                   ; 假设 max = -7
    cmp  rbx, r12                   ; 3 和 -7 比
    jle  .s_keep                    ; ★ 有符号比较：3 <= -7 不成立，所以不跳
    mov  r12, rbx                   ; max = 3
.s_keep:
    mov  rdi, r12
    call io_print_int
    call io_newline

; ---------------------------------------------------------------------------
; 2) 无符号比较：同样两个数，-7 被当成 0xfffffffffffffff9
;    jbe = 无符号「小于等于」：3 <= 0xff..f9 成立 → 跳过去，不替换
;    结果 max 是那个「巨大的负数」，打印出来是 0xfffffffffffffff9
; ---------------------------------------------------------------------------
    lea  rsi, [rel t_umax]
    mov  rdx, l_umax
    call io_print

    mov  rax, -7
    mov  rbx, 3
    mov  r12, rax                   ; 假设 max = 0xfffffffffffffff9
    cmp  rbx, r12
    jbe  .u_keep                    ; ★ 无符号比较：3 <= 0xff..f9 成立
    mov  r12, rbx
.u_keep:
    mov  rdi, r12
    call io_print_hex               ; 0xfffffffffffffff9
    lea  rsi, [rel t_udec]
    mov  rdx, l_udec
    call io_print
    mov  rdi, r12
    call io_print_uint              ; 18446744073709551609
    lea  rsi, [rel t_udone]
    mov  rdx, l_udone
    call io_print
    call io_newline

; ---------------------------------------------------------------------------
; 3) 数组：一边打印元素，一边找最大值和最小值
;    寄存器分工（都不能用 rcx/rdx/rax，因为 io 例程会破坏它们）：
;      rbx = 数组首地址（lea rbx, [rel arr]）
;      rbp = 下标 i
;      r12 = 当前最大值    r13 = 当前最小值
;      r14 = 当前元素 —— 元素先放进 r14，再调用 io_print_int，
;            否则打印本身就把 rax 里的元素值冲掉了
; ---------------------------------------------------------------------------
    lea  rsi, [rel t_arr]
    mov  rdx, l_arr
    call io_print

    lea  rbx, [rel arr]             ; rbx → arr[0]
    mov  r12, [rbx]                 ; max 先取 arr[0]
    mov  r13, [rbx]                 ; min 先取 arr[0]
    xor  rbp, rbp                   ; i = 0

.loop:
    cmp  rbp, arr_len
    jge  .loop_done                 ; i >= arr_len 就结束

    mov  r14, [rbx + rbp*8]         ; r14 = arr[i]（下标寻址：基址 + 下标*8）

    cmp  r14, r12
    jle  .no_max                    ; 有符号比较：arr[i] <= max 就不更新
    mov  r12, r14
.no_max:
    cmp  r14, r13
    jge  .no_min                    ; 有符号比较：arr[i] >= min 就不更新
    mov  r13, r14
.no_min:

    mov  rdi, r14                   ; 打印这个元素
    call io_print_int
    inc  rbp                        ; i++
    cmp  rbp, arr_len
    jge  .loop_done                 ; 最后一个元素后面不加逗号
    lea  rsi, [rel s_comma]
    mov  rdx, l_comma
    call io_print
    jmp  .loop

.loop_done:
    lea  rsi, [rel t_arr_e]
    mov  rdx, l_arr_e
    call io_print                   ; 打印 "]" 和换行

; ---------------------------------------------------------------------------
; 4) 打印最大值和最小值
; ---------------------------------------------------------------------------
    lea  rsi, [rel t_max]
    mov  rdx, l_max
    call io_print
    mov  rdi, r12
    call io_print_int
    call io_newline

    lea  rsi, [rel t_min]
    mov  rdx, l_min
    call io_print
    mov  rdi, r13
    call io_print_int
    call io_newline

; ---------------------------------------------------------------------------
; 结束进程
; ---------------------------------------------------------------------------
    xor  edi, edi
    call io_exit

; ===========================================================================
; count.asm —— 计数循环：用 rcx 数数，打印 1..10、10..1，以及 loop 指令
;
; ★ 本课最重要的一条规则 ★
;   io.inc 里的每个例程内部都会执行 syscall，而 syscall 指令会把 rcx
;   和 r11 当作「返回地址/标志位暂存」来用，返回后 rcx 里就不是原来的值了。
;   所以：拿 rcx 当循环计数器时，每次调用 io 例程之前要把 rcx 压栈，
;         调用完再弹回来。不想这么麻烦就换成 rbx/rbp/r12-r15。
;
; 纯系统调用程序：入口 _start。运行： make && ./count
; ===========================================================================
default rel

section .data
    t_a db "A) rcx counter, 1..10   : "
    l_a equ $ - t_a
    t_b db "B) rcx counter, 10..1   : "
    l_b equ $ - t_b
    t_c db "C) loop: 1+2+...+100    = "
    l_c equ $ - t_c
    s_sp db " "
    l_sp equ $ - s_sp

section .text
    global _start
%include "io.inc"

_start:
; ---------------------------------------------------------------------------
; A) 用 rcx 当计数器打印 1 2 3 ... 10（数字之间一个空格，末尾没有空格）
;    每次调用 io 例程前后都 push/pop rcx —— 这就是「保护计数器」
; ---------------------------------------------------------------------------
    lea  rsi, [rel t_a]
    mov  rdx, l_a
    call io_print

    mov  rcx, 1                     ; 计数器：1 .. 10
.a_loop:
    cmp  rcx, 1
    je   .a_first                   ; 第一个数字前面不加空格
    push rcx                        ; ★ 保护计数器（syscall 会破坏 rcx）
    lea  rsi, [rel s_sp]
    mov  rdx, l_sp
    call io_print
    pop  rcx                        ; ★ 取回计数器
.a_first:
    push rcx
    mov  rdi, rcx
    call io_print_int               ; 打印计数器的值
    pop  rcx
    inc  rcx                        ; 计数器 +1
    cmp  rcx, 10
    jle  .a_loop                    ; 还没数到 10 就继续
    call io_newline

; ---------------------------------------------------------------------------
; B) 倒着数 10 9 8 ... 1：用 dec + jnz
;    dec 会设置 ZF：减到 0 时 ZF=1，jnz 就不再跳回去
;    注意 dec 不影响 CF，所以别指望用 jb/jae 判断「减到 0 了」
; ---------------------------------------------------------------------------
    lea  rsi, [rel t_b]
    mov  rdx, l_b
    call io_print

    mov  rcx, 10
.b_loop:
    cmp  rcx, 10
    je   .b_first                   ; 第一个数字（10）前面不加空格
    push rcx
    lea  rsi, [rel s_sp]
    mov  rdx, l_sp
    call io_print
    pop  rcx
.b_first:
    push rcx
    mov  rdi, rcx
    call io_print_int
    pop  rcx
    dec  rcx                        ; 计数器 -1
    jnz  .b_loop                    ; 还不是 0 就继续
    call io_newline

; ---------------------------------------------------------------------------
; C) loop 指令：rcx 减 1，如果 rcx != 0 就跳回标号
;    loop 只认 rcx，而且比「dec rcx + jnz」慢，实际代码里前者更常用。
;    这里循环体里没有 io 调用，所以 rcx 可以放心用。
; ---------------------------------------------------------------------------
    lea  rsi, [rel t_c]
    mov  rdx, l_c
    call io_print

    xor  rax, rax                   ; rax = 累加和 0
    mov  rcx, 100                   ; 从 100 加到 1
.c_sum:
    add  rax, rcx                   ; sum += i
    loop .c_sum                     ; rcx--；rcx != 0 就跳回 .c_sum
    mov  rdi, rax
    call io_print_int               ; 5050
    call io_newline

; ---------------------------------------------------------------------------
; 结束进程
; ---------------------------------------------------------------------------
    xor  edi, edi
    call io_exit

; ===========================================================================
; sum_array.asm —— 遍历 .data 里的 qword 数组：求和 / 数负数 / 求平均值
;
; 数组里故意放了负数，用来练：
;   * [基址 + 下标*8] 这种寻址方式
;   * 有符号比较（test/jns、jl、jge），别误用无符号跳转
;   * 有符号除法前的 cqo
;   * 循环计数器不能放 rcx —— syscall 会破坏它
;
; 纯系统调用程序：入口 _start。运行： make && ./sum_array
; 期望：sum = 80，count = 8，negative = 4，average = 10
; ===========================================================================
default rel

section .data
    arr     dq 10, -3, 25, -8, -1, 100, -50, 7
    arr_len equ ($ - arr) / 8       ; 8 个元素，每个 8 字节

    t_arr   db "array    : ["
    l_arr   equ $ - t_arr
    t_arr_e db "]", 10                 ; .data 里换行只能写 10（此时还没 include io.inc）
    l_arr_e equ $ - t_arr_e
    t_sum   db "sum      = "
    l_sum   equ $ - t_sum
    t_cnt   db "count    = "
    l_cnt   equ $ - t_cnt
    t_neg   db "negative = "
    l_neg   equ $ - t_neg
    t_avg   db "average  = "
    l_avg   equ $ - t_avg
    s_comma db ", "
    l_comma equ $ - s_comma

section .text
    global _start
%include "io.inc"

_start:
; ---------------------------------------------------------------------------
; 1) 打印数组内容，格式：[10, -3, 25, -8, -1, 100, -50, 7]
;    寄存器分工：
;      rbx = 数组首地址   rbp = 下标 i（这两个 io 例程都会原样保留）
;      r14 = 当前元素（先用 r14 存住，因为 io_print_int 会破坏 rax）
; ---------------------------------------------------------------------------
    lea  rsi, [rel t_arr]
    mov  rdx, l_arr
    call io_print

    lea  rbx, [rel arr]             ; rbx → arr[0]
    xor  rbp, rbp                   ; i = 0
.prt_loop:
    cmp  rbp, arr_len
    jge  .prt_done

    mov  r14, [rbx + rbp*8]         ; r14 = arr[i]
    mov  rdi, r14
    call io_print_int

    inc  rbp
    cmp  rbp, arr_len
    jge  .prt_done                  ; 最后一个元素后面不加逗号
    lea  rsi, [rel s_comma]
    mov  rdx, l_comma
    call io_print
    jmp  .prt_loop

.prt_done:
    lea  rsi, [rel t_arr_e]
    mov  rdx, l_arr_e
    call io_print                   ; 打印 "]" 和换行

; ---------------------------------------------------------------------------
; 2) 遍历一遍，同时算出累加和与负数个数
;      r12 = sum    r13 = 负数个数    r15 = 元素总数（一会儿求平均值要用）
;    test r14, r14 后看 SF：SF=1 说明最高位是 1，也就是负数 → jns 是「非负就跳」
; ---------------------------------------------------------------------------
    xor  r12, r12                   ; sum = 0
    xor  r13, r13                   ; 负数个数 = 0
    mov  r15, arr_len               ; 元素总数
    lea  rbx, [rel arr]
    xor  rbp, rbp                   ; i = 0

.sum_loop:
    cmp  rbp, r15
    jge  .sum_done

    mov  r14, [rbx + rbp*8]         ; r14 = arr[i]
    add  r12, r14                   ; sum += arr[i]

    test r14, r14
    jns  .not_neg                   ; SF=0 → 不是负数，跳过计数
    inc  r13
.not_neg:
    inc  rbp
    jmp  .sum_loop

.sum_done:
    ; ---- 打印 sum ----
    lea  rsi, [rel t_sum]
    mov  rdx, l_sum
    call io_print
    mov  rdi, r12
    call io_print_int
    call io_newline

    ; ---- 打印 count ----
    lea  rsi, [rel t_cnt]
    mov  rdx, l_cnt
    call io_print
    mov  rdi, r15
    call io_print_int
    call io_newline

    ; ---- 打印 negative ----
    lea  rsi, [rel t_neg]
    mov  rdx, l_neg
    call io_print
    mov  rdi, r13
    call io_print_int
    call io_newline

; ---------------------------------------------------------------------------
; 3) 平均值 = sum / count
;    80 / 8 = 10；被除数和除数都在寄存器里，除法前必须 cqo（有符号）
;    如果这里错写成 xor rdx,rdx（无符号那套），负数数组就会得到荒唐的结果
; ---------------------------------------------------------------------------
    lea  rsi, [rel t_avg]
    mov  rdx, l_avg
    call io_print

    mov  rax, r12                   ; 被除数 = sum
    cqo                             ; ★ 有符号除法：把符号位扩展到 rdx
    idiv r15                        ; rax = sum / count
    mov  rdi, rax
    call io_print_int
    call io_newline

; ---------------------------------------------------------------------------
; 结束进程
; ---------------------------------------------------------------------------
    xor  edi, edi
    call io_exit

; ===========================================================================
; add_func.asm —— 第 07 课示例 1：第一个函数（call / ret / 返回值在 rax）
;
; 编译（在 lesson07 目录下）：
;   make
; 运行：
;   ./add_func
;
; 知识点：
;   * call 标签 —— 把「返回地址」压栈，然后跳到标签处执行
;   * ret       —— 从栈顶弹出返回地址，跳回去
;   * 参数按 System V 约定放在 rdi, rsi（第 1、2 个参数）
;   * 返回值放在 rax
;   * 被调用者保存（rbx/rbp/r12-r15）与调用者保存的含义
; ===========================================================================

section .data
    ; ---- 提示文字 -------------------------------------------------------
    title      db "=== add_two(a, b)：参数用 rdi/rsi，返回值用 rax ==="
    title_len  equ $ - title

    l_test     db "add_two(a_value, b_value)   = "
    l_test_len equ $ - l_test

    l_inline   db "add_two(1000, 2345)         = "
    l_inline_len equ $ - l_inline

    l_neg      db "add_two(7, -20)             = "
    l_neg_len  equ $ - l_neg

    l_wrap     db "add_two(1, -1)              = "
    l_wrap_len equ $ - l_wrap

    l_four     db "add_four(r12, rbx, 100, -5) = "
    l_four_len equ $ - l_four

    l_after    db "函数返回后 rbx（还是 22）   = "
    l_after_len equ $ - l_after

    ; ---- 演示用的操作数 -------------------------------------------------
    a_value    dq 12
    b_value    dq 30

section .text
    global _start
default rel                             ; 数据寻址默认 RIP 相对，所有 [标签] 都自动变成 [rel 标签]
bits 64

%include "io.inc"

; ---------------------------------------------------------------------------
; _start —— 程序入口（相当于 C 的 main）
; ---------------------------------------------------------------------------
_start:
    ; 打印标题
    lea rsi, [rel title]
    mov rdx, title_len
    call io_println

    ; ---- 第 1 个调用：参数来自内存里的变量 ------------------------------
    lea rsi, [rel l_test]
    mov rdx, l_test_len
    call io_print

    mov rdi, [rel a_value]      ; 第 1 个参数 = 12
    mov rsi, [rel b_value]      ; 第 2 个参数 = 30
    call add_two                ; 返回值在 rax
    mov rdi, rax
    call io_print_int
    call io_newline

    ; ---- 第 2 个调用：参数是立即数 --------------------------------------
    lea rsi, [rel l_inline]
    mov rdx, l_inline_len
    call io_print

    mov edi, 1000
    mov esi, 2345
    call add_two
    mov rdi, rax
    call io_print_int
    call io_newline

    ; ---- 第 3 个调用：带负数（补码加法，和 C 的 int64_t 一样） ----------
    lea rsi, [rel l_neg]
    mov rdx, l_neg_len
    call io_print

    mov rdi, 7
    mov rsi, -20                ; 有符号立即数可以直接写 -20
    call add_two
    mov rdi, rax
    call io_print_int
    call io_newline

    ; ---- 第 4 个调用：1 + (-1) = 0，顺便看 -1 的 64 位补码 --------------
    lea rsi, [rel l_wrap]
    mov rdx, l_wrap_len
    call io_print

    mov rdi, 1
    mov rsi, -1                 ; -1 的 64 位补码就是 0xFFFFFFFFFFFFFFFF
    call add_two
    mov rdi, rax
    call io_print_int
    call io_newline

    ; ---- 第 5 个调用：嵌套调用 + 被调用者保存寄存器 ---------------------
    mov r12, 20                 ; r12 是被调用者保存寄存器
    mov rbx, 22                 ; rbx 也是

    lea rsi, [rel l_four]
    mov rdx, l_four_len
    call io_print

    ; 按 rdi, rsi, rdx, rcx 的顺序放第 1..4 个参数
    mov rdi, r12
    mov rsi, rbx
    mov rdx, 100                ; 写 rdx 而不是 edx，风格统一
    mov rcx, -5                 ; ⚠ 负数千万别写 mov ecx, -5（见 README「常见坑」）
    call add_four               ; 内部会调用 add_two_rbx（故意破坏 rbx）
    mov rdi, rax
    call io_print_int
    call io_newline

    ; 如果 add_two_rbx 老老实实保存、恢复了 rbx，这里 rbx 还应该是 22
    lea rsi, [rel l_after]
    mov rdx, l_after_len
    call io_print
    mov rdi, rbx
    call io_print_int
    call io_newline

    ; ---- 正常退出：退出码 0 --------------------------------------------
    xor rdi, rdi
    call io_exit

; ---------------------------------------------------------------------------
; add_two —— 返回 rdi + rsi
;
;   入参：rdi = 第一个数，rsi = 第二个数
;   返回：rax = 两数之和
;   只用到调用者保存寄存器 rax，所以不需要 push/pop
; ---------------------------------------------------------------------------
add_two:
    mov rax, rdi
    add rax, rsi
    ret

; ---------------------------------------------------------------------------
; add_two_rbx —— 和 add_two 一样，但故意把 rbx 写坏
;
;   目的：演示「被调用者保存」的规矩 —— 谁破坏 rbx，谁负责恢复。
; ---------------------------------------------------------------------------
add_two_rbx:
    push rbx                    ; 先保存调用者的 rbx
    mov rbx, rdi
    add rbx, rsi
    mov rax, rbx
    pop rbx                     ; 返回前恢复，调用者感觉不到 rbx 被用过
    ret

; ---------------------------------------------------------------------------
; add_four —— 返回 a+b+c+d，内部调用 add_two_rbx
;
;   入参：rdi=a, rsi=b, rdx=c, rcx=d
;   返回：rax = a+b+c+d
;
;   注意栈的对齐：这里 push/pop 严格成对（2 个 push、2 个 pop），
;   add_two_rbx 内部也是 1 个 push、1 个 pop，
;   所以进每个函数时 rsp 的对齐状态一路保持不变（详见 README「栈对齐」）。
; ---------------------------------------------------------------------------
add_four:
    push rbp                    ; 标准函数序言
    mov rbp, rsp                ; rbp 指向「栈上保存的旧 rbp」
    push rbx                    ; 本函数要用 rbx，先保存
    push r12                    ; 也要用 r12，先保存（push 两个，对齐不变）

    mov r12, rdx                ; 先把 c 存进 r12（rdx 马上就要被用掉了）
    mov rbx, rcx                ; 再把 d 存进 rbx

    call add_two_rbx            ; 算 a + b（参数已经在 rdi/rsi 里）
    add rax, r12                ; 加 c
    add rax, rbx                ; 加 d

    pop r12
    pop rbx
    leave                       ; = mov rsp, rbp + pop rbp
    ret

; ===========================================================================
; multi_args.asm —— 第 07 课示例 3：6 个参数怎么传（寄存器 + 栈）
;
; 编译（在 lesson07 目录下）：
;   make
; 运行：
;   ./multi_args
;
; System V AMD64 调用约定规定，整数/指针参数按顺序放进：
;   第 1 个 rdi、第 2 个 rsi、第 3 个 rdx、第 4 个 rcx、第 5 个 r8、第 6 个 r9
; 超出的第 7、8、… 个参数才压栈（从右往左压），返回值永远在 rax。
;
; 注意返回值不够用时（比如要返回两个值），rax 只能放一个，
; 另一个就得靠「调用者传进来的内存地址」写回去（见 README 里的说明）。
; ===========================================================================

    l_head_len equ $ - l_head

;
;   入参：rdi=a, rsi=b, rdx=c, rcx=d, r8=e, r9=f
;   返回：rax = 六个数的和
;
;   函数体里只碰了 rax、rdi、rsi、rdx、rcx、r8、r9，
;   全是可以随便改的「调用者保存」寄存器，所以不用 push/pop。
;   累加时特意用 add rax, rXX 一个个加，方便对照它的参数寄存器。
; ---------------------------------------------------------------------------
sum6:
    mov rax, rdi                ; rax = a
    add rax, rsi                ; + b
    add rax, rdx                ; + c
    add rax, rcx                ; + d
    add rax, r8                 ; + e
    add rax, r9                 ; + f
    ret

; ---------------------------------------------------------------------------
; sum7 —— 返回 7 个数的和，第 7 个数在栈上
;
;   入参：rdi..r9 = 前 6 个，栈上 [rsp+8] = 第 7 个（[rsp] 是返回地址）
;   返回：rax
;
;   为什么是 [rsp+8] 而不是 [rsp]？
;   因为 call sum7 先把返回地址压在了栈顶，第 7 个参数在它上面一格。
;   用 rbp 建立帧之后就是 [rbp+16]，两种写法等价（见 README）。
; ---------------------------------------------------------------------------
sum7:
    mov rax, rdi
    add rax, rsi
    add rax, rdx
    add rax, rcx
    add rax, r8
    add rax, r9
    add rax, [rsp + 8]          ; 第 7 个参数在栈上
    ret

; ===========================================================================
; stack_demo.asm —— 第 07 课示例 2：栈是向下增长的
;
; 编译（在 lesson07 目录下）：
;   make
; 运行：
;   ./stack_demo
;
; 运行时会打印一串十六进制地址。这些地址和内核给的初始栈位置有关，
; 每台机器、每次运行都不一样，但有两点永远成立：
;   1. push 之后 rsp 变小，pop / add rsp 之后 rsp 变大 —— 栈往低地址方向长；
;   2. 下面打印的「相差 N 字节」在同一次运行里完全固定。
;
; 本程序把基准 rsp 存在 rbp / r15 里，靠它们算差值。
;   为什么不用 rax 存？因为 io.inc 的打印函数一律只保证
;   rax/rcx/r11 之外的寄存器不变（rbx/rbp/r12-r15 是被调用者保存），
;   拿 rax 存跨调用的值，调用一次 io_print 就被冲掉了。
; ===========================================================================

section .data
    title       db "=== 栈（stack）是向下增长的 ==="
    title_len   equ $ - title

    ; ---- 第一部分：push / pop -------------------------------------------
    s_push      db "--- push 一次，rsp 就减 8（一个 8 字节栈槽）---"
    s_push_len  equ $ - s_push

    s_rsp0      db "起始 rsp（对齐到 16）        = "
    s_rsp0_len  equ $ - s_rsp0

    s_push1     db "第 1 次 push 后的 rsp（下了 "
    s_push1_len equ $ - s_push1

    s_push2     db "第 2 次 push 后的 rsp（下了 "
    s_push2_len equ $ - s_push2

    s_end       db "全部弹干净后的 rsp             = "
    s_end_len   equ $ - s_end

    s_bytes     db " 字节） = "
    s_bytes_len equ $ - s_bytes

    s_unit      db " 字节"
    s_unit_len  equ $ - s_unit

    ; ---- 第二部分：函数帧（frame）---------------------------------------
    s_frame     db "--- 函数帧：push rbp; mov rbp, rsp; sub rsp, 16 ---"
    s_frame_len equ $ - s_frame

    s_rbp       db "函数里 rbp - rsp（局部空间） = "
    s_rbp_len   equ $ - s_rbp

    s_frame_rsp db "函数里 rsp                   = "
    s_frame_rsp_len equ $ - s_frame_rsp

    s_retaddr   db "call 压进去的返回地址槽位    = "
    s_retaddr_len equ $ - s_retaddr

    s_frame_ok  db "返回地址槽里存的确实是下一条指令的地址，"
    s_frame_ok_len equ $ - s_frame_ok
    s_frame_ok2 db "所以 ret 才能准确跳回来。"
    s_frame_ok2_len equ $ - s_frame_ok2

    ; 两个演示用的数（作为立即数 push 到栈上）
    v_one       equ 0x1111
    v_two       equ 0x2222

section .text
    global _start
default rel                             ; 数据寻址默认 RIP 相对，所有 [标签] 都自动变成 [rel 标签]
bits 64

%include "io.inc"

_start:
    ; 把 rsp 对齐到 16，这样打印出来的地址末尾都是 0，好对照
    and rsp, -16
    mov rbp, rsp                ; rbp = rsp0：全程不动的基准
    mov r15, rsp                ; r15 = rsp0（r15 也是被调用者保存）

    lea rsi, [rel title]
    mov rdx, title_len
    call io_println

    lea rsi, [rel s_push]
    mov rdx, s_push_len
    call io_println

    ; ---- 打印起始 rsp ---------------------------------------------------
    lea rsi, [rel s_rsp0]
    mov rdx, s_rsp0_len
    call io_print
    mov rdi, rbp
    call io_print_hex
    call io_newline

    ; ---- push 两个数 ----------------------------------------------------
    push v_one                  ; rsp = rsp0 - 8
    push v_two                  ; rsp = rsp0 - 16

    ; 第 1 次 push 后的 rsp 等于 rbp - 8
    lea rsi, [rel s_push1]
    mov rdx, s_push1_len
    call io_print
    mov rdi, 8                  ; 下了 8 字节
    call print_u64
    lea rsi, [rel s_bytes]
    mov rdx, s_bytes_len
    call io_print
    lea rdi, [rbp - 8]
    call io_print_hex
    call io_newline

    ; 现在（第 2 次 push 之后）的 rsp 就是 rsp0 - 16
    lea rsi, [rel s_push2]
    mov rdx, s_push2_len
    call io_print
    mov rdi, r15
    sub rdi, rsp                ; 下了 16 字节
    call print_u64
    lea rsi, [rel s_bytes]
    mov rdx, s_bytes_len
    call io_print
    mov rdi, rsp
    call io_print_hex
    call io_newline

    ; ---- 弹干净 ---------------------------------------------------------
    ; 效果上等于 pop 两次；这里直接用 add rsp, 16 更直观
    add rsp, 16

    lea rsi, [rel s_end]
    mov rdx, s_end_len
    call io_print
    mov rdi, rsp
    call io_print_hex
    call io_newline

    ; ---- 第二部分：函数帧 ------------------------------------------------
    lea rsi, [rel s_frame]
    mov rdx, s_frame_len
    call io_println
    call frame_probe            ; 内部把三个观测值写进 .bss

    lea rsi, [rel s_rbp]
    mov rdx, s_rbp_len
    call io_print
    mov rdi, [rel frame_size]
    call print_u64
    lea rsi, [rel s_unit]
    mov rdx, s_unit_len
    call io_print
    call io_newline

    lea rsi, [rel s_frame_rsp]
    mov rdx, s_frame_rsp_len
    call io_print
    mov rdi, [rel frame_rsp]
    call io_print_hex
    call io_newline

    lea rsi, [rel s_retaddr]
    mov rdx, s_retaddr_len
    call io_print
    mov rdi, [rel ret_slot]
    call io_print_hex
    call io_newline

    lea rsi, [rel s_frame_ok]
    mov rdx, s_frame_ok_len
    call io_println
    lea rsi, [rel s_frame_ok2]
    mov rdx, s_frame_ok2_len
    call io_println

    xor rdi, rdi
    call io_exit

; ---------------------------------------------------------------------------
; frame_probe —— 演示一个标准函数帧长什么样
;
; 刚进函数时栈上的布局（地址从高到低）：
;       [rsp + 8]   = _start 的 rbp（上一帧的帧基址）
;       [rsp]       = 返回地址（call frame_probe 压进去的）
;
; 序言执行完之后：
;       [rbp + 8]   = 返回地址
;       [rbp]       = 旧的 rbp
;       [rbp - 8]   = 局部空间的高 8 字节
;       [rbp - 16]  = 局部空间的低 8 字节  ← 函数体执行时 rsp 在这里
;
; 本函数把三个观测值记到 .bss 里：帧大小、当时的 rsp、返回地址槽的地址。
; ---------------------------------------------------------------------------
frame_probe:
    push rbp
    mov rbp, rsp
    sub rsp, 16                 ; 开 16 字节局部空间（保持 16 字节对齐）

    mov rax, rbp
    sub rax, rsp                ; 帧大小 = rbp - rsp = 16
    mov [rel frame_size], rax

    mov [rel frame_rsp], rsp

    lea rax, [rbp + 8]          ; 返回地址就在 rbp + 8 处
    mov [rel ret_slot], rax

    leave                       ; = mov rsp, rbp + pop rbp
    ret

; ---------------------------------------------------------------------------
; print_u64 —— 把 rdi 里的无符号数按十进制打印出来
;
;   本课只拿它打印「小整数」（偏移量），所以没有处理 0 之外的特殊情况，
;   但 0 也要能正确打印成 "0"（do-while 结构天然满足）。
; ---------------------------------------------------------------------------
print_u64:
    push rbx                    ; 本函数要用 rbx（被调用者保存），先存起来
    lea rbx, [rel numbuf + 24]  ; rbx 指向缓冲区末尾，从后往前填
    mov rax, rdi
    mov r8, 10                  ; 除数 10

.loop:
    xor rdx, rdx                ; div 前必须清零 rdx（被除数高 64 位）
    div r8                      ; rax = rax / 10，rdx = rax % 10
    add dl, '0'
    dec rbx
    mov [rbx], dl
    test rax, rax
    jnz .loop

    mov rsi, rbx
    lea rdx, [rel numbuf + 24]
    sub rdx, rbx
    call io_print

    pop rbx
    ret

section .bss
    frame_size  resq 1          ; 函数帧大小
    frame_rsp   resq 1          ; 函数里的 rsp
    ret_slot    resq 1          ; 返回地址槽的地址
    numbuf      resb 24         ; 十进制转换缓冲区

; ===========================================================================
; num2str.asm —— 第 08 课示例 1：整数 → 十进制字符串（自己动手写）
;
; 编译（在 lesson08 目录下）：
;   make
; 运行：
;   ./num2str
;
; 本程序故意【不调用 io_print_int / io_print_uint】，
; 而是自己用「除 10 取余」把数字一位一位算出来，
; 看看 printf("%d") 背后到底发生了什么。
;
; 核心难点：除 10 先得到的是【最低位】，而我们想从最高位开始打印。
; 两种常见解法：
;   * 缓冲区倒在栈上：sub rsp 借一块栈空间，指针从末尾往前进
;             —— 见 print_uint_v1
;   * 缓冲区放在 .bss：同上，只是固定一块全局内存，还能把字符串返回给调用者
;             —— 见 u64_to_dec
; ===========================================================================

    title_len   equ $ - title

    l_v1        db "--- v1：栈上开缓冲区，从末尾往前填 ---"
    l_v1_len    equ $ - l_v1

    l_v1_a      db "print_uint_v1(0)                    = "
    l_v1_a_len  equ $ - l_v1_a

    l_v1_b      db "print_uint_v1(7)                    = "
    l_v1_b_len  equ $ - l_v1_b

    l_v1_c      db "print_uint_v1(12345)                = "
    l_v1_c_len  equ $ - l_v1_c

    l_v1_d      db "print_uint_v1(18446744073709551615) = "
    l_v1_d_len  equ $ - l_v1_d

    l_rev       db "print_uint_reversed(12345)          = "
    l_rev_len   equ $ - l_rev

    l_rev2      db "  <- 反面教材：低位先算出来，直接打印就是倒的"
    l_rev2_len  equ $ - l_rev2

    l_trace     db "--- 12345 的除法过程（顺序是反的：先出最低位）---"
    l_trace_len equ $ - l_trace

    l_step      db "  余数(当前最低位) = "
    l_step_len  equ $ - l_step

    l_quot      db "   商 = "
    l_quot_len  equ $ - l_quot

    l_v2        db "--- v2：往缓冲区末尾倒着填，再整体打印 ---"
    l_v2_len    equ $ - l_v2

    l_v2_a      db "u64_to_dec(0)        -> 【"
    l_v2_a_len  equ $ - l_v2_a

    l_v2_b      db "u64_to_dec(12345)    -> 【"
    l_v2_b_len  equ $ - l_v2_b

    l_v2_c      db "u64_to_dec(1000000)  -> 【"
    l_v2_c_len  equ $ - l_v2_c

    l_v2_d      db "u64_to_dec(18446744073709551615) -> 【"
    l_v2_d_len  equ $ - l_v2_d

    l_v2_e_len  equ $ - l_v2_e

    l_v2_f      db "]，只有最后一个字节被改写 —— 字符串是从末尾往前长的"
    l_v2_f_len  equ $ - l_v2_f

    call io_print
    call io_newline

    ; 先往「第 8~19 字节」写上 '#'，转换后哪几个字节被改写一目了然。

    ;   这里 dumpbuf 开了 32 字节，我们只看最后 12 字节那一小段。
    lea rdi, [rel dumpbuf + 8]
    mov rcx, 12
    mov al, '#'
.fill:

    inc rdi
    dec rcx

    mov rdi, 1

    lea rsi, [rel l_v2_e]
    mov rdx, l_v2_e_len

    mov rdx, 12
    call io_print
    lea rsi, [rel l_v2_f]
    mov rdx, l_v2_f_len
    call io_println

    xor rdi, rdi
    call io_exit

; ---------------------------------------------------------------------------
; print_uint_v1 —— 自己写十进制输出（栈上开缓冲区，从末尾往前填）
;
;   入参：rdi = 无符号 64 位整数
;   返回：无；直接写到 stdout
;   破坏：rax/rcx/rdx/rsi/rdi/r8-r11；rbx/rbp/r12-r15 已保存
;
;   思路：
;     1. 反复 div 10：商留给下一轮，余数是「当前最低位」；
;     2. 每算出一位就放到缓冲区的【末尾】，然后指针前移一格；
;     3. 算完时指针正好停在最高位上，缓冲区从那里到末尾就是正序字符串。
;
;   缓冲区直接开在被减小的 rsp 上（sub rsp, 32），不额外用 .bss，
;   这是汇编里很常见的写法：rsp 以下的空间没人用，借来当草稿纸最方便。
; ---------------------------------------------------------------------------
print_uint_v1:
    push rbp
    mov rbp, rsp
    push rbx                    ; 被调用者保存：用 rbx 当除数
    push r12                    ; 用 r12 当「从后往前写」的指针
    sub rsp, 32                 ; ⚠ 缓冲区必须在 push 之后再开，否则会
                                ;   和刚保存的 rbx/r12 叠在一起！

    mov rax, rdi                ; rax = 被除数
    mov rbx, 10                 ; 除数（div 的除数不能是立即数）
    lea r12, [rsp + 32]         ; r12 = 缓冲区末尾（rsp+32 是缓冲区之外）

.loop:
    xor rdx, rdx                ; ⚠ div 之前必须把 rdx（被除数高 64 位）清零
    div rbx                     ; rax = rax / 10，rdx = rax % 10
    add dl, '0'                 ; 0..9 -> '0'..'9'
    dec r12                     ; 指针先退一格
    mov [r12], dl               ; 写入这一位
    test rax, rax
    jnz .loop                   ; 商不为 0 就继续

    mov rsi, r12                ; 起点 = 最高位
    lea rdx, [rsp + 32]         ; 终点 = 缓冲区末尾
    sub rdx, r12                ; 长度 = 终点 - 起点
    call io_print

    add rsp, 32
    pop r12
    pop rbx
    leave
    ret

; ---------------------------------------------------------------------------
; print_uint_reversed —— 反面教材：先算出来的低位先打印，结果是倒的
;
;   入参：rdi = 无符号 64 位整数
;   print_uint_reversed(12345) 会打印成 54321。
;   因为除 10 第一次拿到的就是最低位，直接输出自然是从低位到高位。
; ---------------------------------------------------------------------------
print_uint_reversed:
    push rbp
    mov rbp, rsp
    push rbx

    mov rax, rdi
    mov rbx, 10

.loop:
    xor rdx, rdx
    div rbx
    add dl, '0'
    push rax                    ; 打印会破坏 rax，先把它存好
    mov rdi, rdx                ; rdx 的低 8 位就是这一位的 ASCII
    call io_putchar
    pop rax
    test rax, rax
    jnz .loop

    pop rbx
    leave
    ret

; ---------------------------------------------------------------------------
; u64_to_dec —— 把 rdi 转成十进制字符串，写进 rsi 指向的缓冲区（倒着填）
;
;   入参：rdi = 无符号 64 位整数
;         rsi = 缓冲区首地址（必须至少有 20 字节！）
;   返回：rax = 字符串长度（不含结尾的 0；本函数不写 0）
;         rsi = 字符串真正的起点（可能在缓冲区中间甚至末尾）
;
;   为什么需要 20 字节？u64 最大 18446744073709551615，正好 20 位。
;   为什么从后往前填？因为除 10 先得到最低位，而字符串要求最高位在前。
; ---------------------------------------------------------------------------
u64_to_dec:
    push rbx                    ; 被调用者保存：本函数要用 rbx/r12
    push r12

    mov rax, rdi                ; 被除数
    mov rbx, 10                 ; 除数
    lea r12, [rsi + 20]         ; r12 = 缓冲区末尾（指向最后一个字节之后）

.digits:
    xor rdx, rdx                ; ⚠ 每次 div 前都要清零
    div rbx
    add dl, '0'
    dec r12                     ; 先退一格
    mov [r12], dl               ; 再写入这一位
    test rax, rax
    jnz .digits

    ; 长度 = 缓冲区末尾(rsi+20) - 起点(r12)
    lea rax, [rsi + 20]
    sub rax, r12
    mov rsi, r12                ; rsi = 真正的起点

    pop r12
    pop rbx
    ret

; ---------------------------------------------------------------------------
; print_dec_string —— 把 rdi 转成十进制字符串并打印（调用 u64_to_dec）
;
;   入参：rdi = 无符号 64 位整数
; ---------------------------------------------------------------------------
print_dec_string:
    lea rsi, [rel decbuf]       ; 传缓冲区首地址
    call u64_to_dec
    mov rdx, rax                ; rdx = 长度（rsi 已经是 u64_to_dec 返回的起点）
    call io_print
    ret

; ---------------------------------------------------------------------------
; trace_digits —— 把「除 10 取余」的每一步打出来（演示用）
;
;   入参：rdi = 数值
; ---------------------------------------------------------------------------
trace_digits:
    push rbp
    mov rbp, rsp
    push rbx
    mov rax, rdi
    mov rbx, 10

.loop:
    xor rdx, rdx
    div rbx                     ; rdx = 余数（当前最低位），rax = 商
    push rax                    ; 打印会破坏 rax/rdx，先存到栈上
    push rdx

    lea rsi, [rel l_step]
    mov rdx, l_step_len
    call io_print
    mov rdi, [rsp]              ; 余数
    call io_print_int

    lea rsi, [rel l_quot]
    mov rdx, l_quot_len
    call io_print
    mov rdi, [rsp + 8]          ; 商
    call io_print_int
    call io_newline

    pop rdx                     ; 取回余数（后面不用了，只为平衡栈）
    pop rax                     ; 取回商
    test rax, rax
    jnz .loop

    pop rbx
    leave
    ret

; ===========================================================================
; str2num.asm —— 第 08 课示例 2：十进制字符串 → 整数（atoi 的原理）
;
; 编译（在 lesson08 目录下）：
;   make
; 运行：
;   ./str2num
;   printf '  12345\n' | ./str2num      # 从标准输入读一行再转换
;
; 本程序要处理 atoi 的四个老大难问题：
;   1. 正负号：'+' / '-' 只能出现在开头，'-' 还要照顾好 INT64_MIN；
;   2. 非法字符：遇到不是 '0'..'9' 的字节，要分得清它是「字符串结束符」
;      还是「垃圾字符」；
;   3. 溢出：乘 10 或加本位之后超出 [INT64_MIN, INT64_MAX]；
;   4. 空串 / 只有符号：不能当成 0 蒙混过关。
;
; 解析核心：value = value * 10 + (c - '0')
; 溢出判断用了一个很实用的技巧（见 parse_str 里的注释）：
;   如果 value*10 溢出，那么 (value*10 / 10) 就不再等于原来的 value。
; ===========================================================================

section .data
    ; ---- 提示文字 -------------------------------------------------------
    title      db "=== 字符串 -> 整数（atoi 手写实现）==="
    title_len  equ $ - title

    l_head     db "输入                    值(十进制)                值(十六进制)          位数  状态"
    l_head_len equ $ - l_head

    ; ---- 内建测试用例（每个都以 NUL 结尾）-------------------------------
    t1         db "12345", 0
    t2         db "-987", 0
    t3         db "+42", 0
    t4         db "0000", 0
    t5         db "9223372036854775807", 0        ; INT64_MAX
    t6         db "-9223372036854775808", 0       ; INT64_MIN
    t7         db "9223372036854775808", 0        ; 比 INT64_MAX 大 1：溢出
    t8         db "99999999999999999999", 0       ; 远远溢出
    t9         db 0                                ; 真正的空串
    t10        db "-", 0                          ; 只有符号
    t11        db "12a3", 0                       ; 中间夹了非法字符

    ; 指针表：dq 存的是各字符串的地址，配合循环遍历
    table      dq t1, t2, t3, t4, t5, t6, t7, t8, t9, t10, t11
    table_cnt  equ ($ - table) / 8

    ; 状态码对应的文字
    txt_ok        db "OK"
    txt_ok_len    equ $ - txt_ok
    txt_bad        db "非法字符"
    txt_bad_len    equ $ - txt_bad
    txt_ovf        db "溢出"
    txt_ovf_len    equ $ - txt_ovf
    txt_nul        db "空指针"
    txt_nul_len    equ $ - txt_nul
    txt_emp        db "空串/只有符号"
    txt_emp_len    equ $ - txt_emp

    ; ---- 从标准输入读的那个演示 -----------------------------------------
    l_stdin    db "--- 从标准输入读一行（可用 printf '...' | ./str2num 喂进去）---"
    l_stdin_len equ $ - l_stdin

    l_prompt   db "读出（已去掉换行）= "
    l_prompt_len equ $ - l_prompt

    l_eof      db "读到 0 字节（EOF），没有输入可解析，直接退出。"
    l_eof_len  equ $ - l_eof

    empty_mark db "(空串)"
    empty_mark_len equ $ - empty_mark

section .bss
    inbuf      resb 64          ; 读标准输入用的缓冲区

section .text
    global _start
default rel                             ; 数据寻址默认 RIP 相对，所有 [标签] 都自动变成 [rel 标签]
bits 64

%include "io.inc"

_start:
    lea rsi, [rel title]
    mov rdx, title_len
    call io_println

    lea rsi, [rel l_head]
    mov rdx, l_head_len
    call io_println

    ; ---- 遍历内建用例表 --------------------------------------------------
    lea rbx, [rel table]        ; rbx 指向指针表（rbx 被调用者保存）
    mov rbp, table_cnt          ; rbp 当循环计数器（同样被调用者保存）

.next:
    mov rdi, [rbx]              ; rdi = 当前测试字符串
    call show_one
    add rbx, 8
    dec rbp
    jnz .next

    call io_newline

    ; ---- 从标准输入读一行 ------------------------------------------------
    lea rsi, [rel l_stdin]
    mov rdx, l_stdin_len
    call io_println

    ; read(0, inbuf, 64)：从标准输入最多读 64 字节
    xor edi, edi                ; fd = STDIN
    lea rsi, [rel inbuf]
    mov edx, 64
    mov rax, SYS_read
    syscall                     ; 返回值在 rax（负数表示出错）

    test rax, rax
    jle .eof                    ; 0 = EOF，负数 = 错误，都直接收工

    ; ---- 手工去掉结尾的 '\n' 和 '\r' ------------------------------------
    mov rcx, rax                ; rcx = 读到的字节数
    lea rdi, [rel inbuf]
.strip:
    test rcx, rcx
    jz .stripped
    mov al, [rdi + rcx - 1]     ; 看最后一个字节
    cmp al, LF
    je .cut
    cmp al, 13                  ; '\r'（Windows 换行）
    je .cut
    jmp .stripped
.cut:
    dec rcx
    jmp .strip
.stripped:
    mov byte [rdi + rcx], 0     ; 打上字符串结束符

    lea rsi, [rel l_prompt]
    mov rdx, l_prompt_len
    call io_print
    lea rsi, [rel inbuf]
    mov rdx, rcx                ; 长度 = 去掉换行后的字节数
    call io_println

    lea rdi, [rel inbuf]
    call show_one

    xor rdi, rdi
    call io_exit

.eof:
    lea rsi, [rel l_eof]
    mov rdx, l_eof_len
    call io_println
    xor rdi, rdi
    call io_exit

; ---------------------------------------------------------------------------
; show_one —— 解析并打印一个用例：输入串、值、十六进制、位数、状态
;
;   入参：rdi = 指向 NUL 结尾字符串的指针
;   如果字符串是空串（连符号都没有），终端里什么都看不见，
;   所以显示的时候换成 "(空串)" 字样，解析时用的还是那个空串本身。
; ---------------------------------------------------------------------------
show_one:
    push rbx                    ; 要用 rbx/r12-r15 存跨调用的值，先保存
    push r12
    push r13
    push r14
    push r15

    mov rbx, rdi                ; rbx = 要解析的字符串

    cmp byte [rbx], 0           ; 真·空串？
    jne .parse
    lea rdi, [rel empty_mark + empty_mark_len]  ; 解析用末尾的 NUL
    mov rbx, rdi
    sub rbx, empty_mark_len     ; 但显示的时候换成 "(空串)" 字样
.parse:
    call parse_str
    mov r12, rax                ; r12 = 值
    mov r13, rdx                ; r13 = 状态码
    mov r14, rcx                ; r14 = 位数（要留到第四列）

    ; ---- 第一列：输入字符串，不足 20 列补空格 ---------------------------
    mov rsi, rbx
    call cstr_len
    mov r15, rax                ; r15 = 这一列的长度（r14 要留着装位数）
    mov rsi, rbx
    mov rdx, r15
    call io_print
    mov rdi, 21                 ; 第一列固定 21 列
    sub rdi, r15                ; 还要补几个空格
    call print_spaces

    ; ---- 第二列：值（十进制，末尾补两格）-------------------------------
    mov rdi, r12
    call print_int64
    mov edi, ' '
    call io_putchar
    call io_putchar
    call io_putchar             ; 多补几格，让 19 位数也留出间距

    ; ---- 第三列：十六进制 ------------------------------------------------
    mov rdi, r12
    call io_print_hex
    mov edi, ' '
    call io_putchar
    call io_putchar
    call io_putchar
    call io_putchar

    ; ---- 第四列：位数 ----------------------------------------------------
    mov rdi, r14
    call print_u64
    mov edi, ' '
    call io_putchar
    mov edi, ' '
    call io_putchar

    ; ---- 第五列：状态 ----------------------------------------------------
    mov rdx, r13
    call print_status

    call io_newline

    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

; ---------------------------------------------------------------------------
; parse_str —— 把 rdi 指向的十进制字符串转成 64 位有符号整数
;
;   入参：rdi = 指向 NUL 结尾字符串的指针
;   返回：rax = 解析出的数值
;         rdx = 状态码：0 成功 / 1 非法字符 / 2 溢出 / 3 空指针 / 4 空串或只有符号
;         rcx = 有效数字位数（不含 '+'/'-'；状态不为 0 时无意义）
;
;   寄存器分工（故意全用「被调用者保存」寄存器，因为中途要调 io_putchar）：
;         rbx = 扫描指针
;         r12 = 累加器 value
;         r13 = 符号标志（0 = 正，1 = 负）
;         r14 = 常量 '0'
;         r15 = 第一个数字的位置（最后用它算位数）
;
;   溢出检测两处：
;     * value*10 溢出 —— 乘完再除以 10，回不到原来的 value；
;     * + digit 溢出  —— 64 位无符号加法进位（jc）。
;   最后再按符号和 INT64_MAX / INT64_MIN 比较。
; ---------------------------------------------------------------------------
parse_str:
    push rbx
    push r12
    push r13
    push r14
    push r15                    ; 5 个 push 是奇数：进函数时 rsp ≡ 8 (mod 16)，
                                ; 5*8 = 40 之后 rsp ≡ 0 (mod 16)，
                                ; 函数体里 call 就满足「rsp 是 16 的倍数」

    ; ---- 0. 空指针检查（防止调用者忘了判空） ----------------------------
    test rdi, rdi
    jnz .have_ptr
    xor eax, eax
    mov rdx, 3
    xor ecx, ecx
    jmp .ret

.have_ptr:
    mov rbx, rdi                ; 扫描指针
    xor r12, r12                ; value = 0
    xor r13, r13                ; 符号 = 正

    ; ---- 1. 跳过开头的空白字符（空格 / TAB / 回车 / 换行）--------------
    ; 「  -12345」这种输入很常见（从终端读进来时尤其如此），
    ; 真正的 atoi 也是这么处理的
.skip_ws:
    mov al, [rbx]
    cmp al, ' '
    je .ws
    cmp al, TAB
    je .ws
    cmp al, 13
    je .ws
    cmp al, LF
    jne .sign
.ws:
    inc rbx
    jmp .skip_ws

    ; ---- 2. 可选的正负号（只认第一个非空白字符）------------------------
.sign:
    mov al, [rbx]
    cmp al, '-'
    jne .chk_plus
    mov r13, 1
    inc rbx
    jmp .digits
.chk_plus:
    cmp al, '+'
    jne .digits
    inc rbx

.digits:
    mov r15, rbx                ; 记下第一个数字的位置
    mov r14b, '0'               ; 常量 '0' 暂放 r14 低 8 位（本函数没别人用 r14）

.loop:
    movzx eax, byte [rbx]       ; 读一个字节（movzx：高位清零）
    test al, al
    jz .end                     ; NUL：字符串正常结束
    sub al, r14b                ; al = c - '0'
    cmp al, 9
    ja .bad_char                ; 不是 '0'..'9'

    ; --- value = value * 10 + digit，带两处溢出检查 ---------------------
    ; ⚠ 不要把 digit 暂存在 ah 里！div 会写 rax，把 ah 一起冲掉
    ;   （这个坑在 README 里也会讲）。这里用 r8 这个「调用者保存」寄存器。
    movzx r8d, al               ; r8 = digit（下面一直要活着）

    mov rcx, r12                ; rcx = 乘之前的 value
    imul r12, r12, 10           ; r12 *= 10
    mov rax, r12
    xor edx, edx
    mov r9, 10
    div r9                      ; rax = (value*10) / 10
    cmp rax, rcx
    jne .overflow               ; 除不回来 -> 乘 10 时就溢出了

    add r12, r8                 ; value += digit
    jc .overflow                ; 无符号加法进位 -> 溢出

    inc rbx
    jmp .loop

.bad_char:
    mov rdx, 1
    xor eax, eax
    xor ecx, ecx
    jmp .ret

.end:
    mov rcx, rbx
    sub rcx, r15                ; 位数 = 扫描位置 - 第一个数字的位置
    test rcx, rcx
    jz .empty                   ; 一个数字都没读到："", "-", "+"

    test r13, r13
    jnz .neg

    mov rax, 0x7fffffffffffffff ; INT64_MAX
    cmp r12, rax
    ja .overflow
    mov rax, r12
    xor rdx, rdx
    jmp .ret

.neg:
    ; 负数最小可以到 -2^63，即无符号写法 0x8000000000000000
    mov rax, 0x8000000000000000
    cmp r12, rax
    ja .overflow
    mov rax, r12
    neg rax                     ; 取负；0x8000000000000000 取负还是它自己
    xor rdx, rdx
    jmp .ret

.empty:
    xor eax, eax
    mov rdx, 4
    xor ecx, ecx
    jmp .ret

.overflow:
    xor eax, eax
    mov rdx, 2
    xor ecx, ecx
.ret:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

; ---------------------------------------------------------------------------
; 下面是给示例配套的小工具函数
; ---------------------------------------------------------------------------

; str_eq —— 比较 rdi 和 rsi 两个 NUL 结尾字符串
;   返回：rax = 1 相等 / 0 不等
str_eq:
    xor eax, eax
.loop:
    mov cl, [rdi]
    mov dl, [rsi]
    cmp cl, dl
    jne .no
    test cl, cl
    jz .yes                     ; 同时到 NUL：相等
    inc rdi
    inc rsi
    jmp .loop
.yes:
    mov eax, 1
.no:
    ret

; cstr_len —— 求 rsi 指向的 NUL 结尾字符串长度
;   返回：rax = 长度（不含 NUL）
cstr_len:
    xor eax, eax
.loop:
    cmp byte [rsi + rax], 0
    je .done
    inc rax
    jmp .loop
.done:
    ret

; print_spaces —— 打印 rdi 个空格
;   入参：rdi = 个数（<= 0 时什么都不打）
print_spaces:
    push rbx
    mov rbx, rdi
    test rbx, rbx
    jle .done
.loop:
    mov rdi, ' '
    call io_putchar
    dec rbx
    jnz .loop
.done:
    pop rbx
    ret

; print_int64 —— 自己写的有符号十进制输出（负数先打 '-' 再取绝对值）
;   入参：rdi = 有符号整数
print_int64:
    test rdi, rdi
    js .neg
    jmp print_u64
.neg:
    push rdi
    mov rdi, '-'
    call io_putchar
    pop rdi
    mov rax, rdi
    neg rax
    mov rdi, rax
    jmp print_u64

; print_u64 —— 无符号十进制输出（栈上开缓冲区，从末尾往前填）
;   入参：rdi = 无符号整数
print_u64:
    push rbp
    mov rbp, rsp
    push rbx
    push r12
    sub rsp, 32                 ; ⚠ 缓冲区必须在 push 之后开

    mov rax, rdi
    mov rbx, 10
    lea r12, [rsp + 32]         ; 缓冲区末尾
.loop:
    xor rdx, rdx                ; ⚠ div 前清零
    div rbx
    add dl, '0'
    dec r12
    mov [r12], dl
    test rax, rax
    jnz .loop

    mov rsi, r12
    lea rdx, [rsp + 32]
    sub rdx, r12                ; 长度
    call io_print

    add rsp, 32
    pop r12
    pop rbx
    leave
    ret

; print_status —— 打印状态码对应的文字
;   入参：rdx = 状态码
print_status:
    cmp rdx, 0
    je .s0
    cmp rdx, 1
    je .s1
    cmp rdx, 2
    je .s2
    cmp rdx, 3
    je .s3
    lea rsi, [rel txt_emp]
    mov rdx, txt_emp_len
    jmp io_print
.s0:
    lea rsi, [rel txt_ok]
    mov rdx, txt_ok_len
    jmp io_print
.s1:
    lea rsi, [rel txt_bad]
    mov rdx, txt_bad_len
    jmp io_print
.s2:
    lea rsi, [rel txt_ovf]
    mov rdx, txt_ovf_len
    jmp io_print
.s3:
    lea rsi, [rel txt_nul]
    mov rdx, txt_nul_len
    jmp io_print

; ===========================================================================
; copy.asm —— 第 09 课示例 1：命令行参数的拷贝工具
;
; 编译（在 lesson09 目录下）：
;   make
; 用法：
;   ./copy 源文件 目标文件
; 例子：
;   printf 'hello, file\n' > /tmp/src.txt
;   ./copy /tmp/src.txt /tmp/dst.txt
;   cat /tmp/dst.txt
;
; 用到的系统调用：
;   openat(257) 打开/创建文件
;   read(0)     读数据
;   write(1)    写数据
;   close(3)    关闭文件
;
; 本课最关键的三个新知识：
;   1. 程序启动时栈上是 [rsp]=argc, [rsp+8]=argv[0], [rsp+16]=argv[1]…
;      取参数之前不能先 push 别的东西（否则偏移全乱）；
;   2. 系统调用第 4 个参数是 r10，不是 rcx；
;   3. syscall 的返回值是负数表示出错（-errno），文件描述符也要判负。
;
; 退出码约定（test.sh 靠它判断错误类型）：
;   0 成功 / 1 参数个数不对 / 2 打不开源文件 / 3 读取失败
;   4 写入失败 / 5 关闭失败 / 6 创建不了目标文件
; ===========================================================================

; AT_FDCWD：openat 的 dirfd 参数，意思是「相对当前工作目录」
; （syscalls.inc 里没有这个常量，本节自己定义）
%define AT_FDCWD -100

section .data
    u1      db "用法: ./copy 源文件 目标文件"
    u1_len  equ $ - u1

    e_open_s db "copy: 打不开源文件: "
    e_open_s_len equ $ - e_open_s

    e_open_d db "copy: 创建不了目标文件: "
    e_open_d_len equ $ - e_open_d

    e_read  db "copy: 读取源文件出错"
    e_read_len equ $ - e_read

    e_write db "copy: 写入目标文件出错"
    e_write_len equ $ - e_write

    e_close db "copy: 关闭文件出错"
    e_close_len equ $ - e_close

    msg_ok  db "复制完成，共 "
    msg_ok_len equ $ - msg_ok

    msg_bytes db " 字节"
    msg_bytes_len equ $ - msg_bytes

    msg_to  db "  -> "
    msg_to_len equ $ - msg_to

    buf_size equ 65536          ; 每次读写 64 KiB（是 8 的倍数，保持栈对齐）

section .bss
    buf     resb buf_size       ; 数据缓冲区（.bss 不占可执行文件体积）

section .text
    global _start
default rel                             ; 数据寻址默认 RIP 相对，所有 [标签] 都自动变成 [rel 标签]
bits 64

%include "io.inc"

; ---------------------------------------------------------------------------
; _start —— 入口。第一件事就是把 argc/argv 从栈上取下来
; ---------------------------------------------------------------------------
_start:
    ; ---- 0. 取命令行参数（必须在任何 push 之前做）----------------------
    mov r12, [rsp]              ; r12 = argc
    lea r13, [rsp + 8]          ; r13 = &argv[0]

    ; ---- 1. 检查参数个数：argc >= 3（程序名 + 源 + 目标）----------------
    cmp r12, 3
    jge .args_ok
    lea rsi, [rel u1]
    mov rdx, u1_len
    call io_println
    mov rdi, 1
    call io_exit

.args_ok:
    mov r12, [r13 + 8]          ; r12 = argv[1]（源文件名）
    mov r13, [r13 + 16]         ; r13 = argv[2]（目标文件名）

    ; 把两个路径存进 .bss，后面打印/关闭时还要用
    mov [rel path_src], r12
    mov [rel path_dst], r13

    call copy_file              ; 真正的拷贝逻辑

    ; ---- 2. 打印结果：复制完成，共 N 字节  ->  目标 ---------------------
    mov r12, rax                ; r12 = 字节数（copy_file 的返回值）
    lea rsi, [rel msg_ok]
    mov rdx, msg_ok_len
    call io_print
    mov rdi, r12
    call print_u64
    lea rsi, [rel msg_bytes]
    mov rdx, msg_bytes_len
    call io_print

    lea rsi, [rel msg_to]
    mov rdx, msg_to_len
    call io_print
    mov rsi, [rel path_dst]
    call cstr_len
    mov rdx, rax
    call io_println

    xor rdi, rdi
    call io_exit

; ---------------------------------------------------------------------------
; copy_file —— 把 r12 指向的文件拷贝到 r13 指向的文件
;
;   入参：r12 = 源文件路径，r13 = 目标文件路径（都是 NUL 结尾）
;   返回：rax = 成功时拷贝的字节数；失败时打印信息并直接 exit
;
;   寄存器约定：本函数保存并恢复 rbx/rbp/r12-r15，调用者放心。
;   循环里用的 rcx/r8/r9/r10/r11 都是「调用者保存」寄存器；
;   循环内部不调用任何函数（只有 syscall），所以可以放心占用。
;
;   ⚠ 一个真实的坑：syscall 会破坏 rcx 和 r11（内核用它们存 rip/rflags），
;     所以它们只能「在两次 syscall 之间」用，不能跨 syscall 保存东西。
;     真正要跨 syscall 活得久的值（比如累计字节数），这里放在 r15 里。
; ---------------------------------------------------------------------------
copy_file:
    push rbx                    ; 6 个 push = 48 字节，成对 pop 保持栈平衡
    push rbp
    push r12
    push r13
    push r14
    push r15

    lea r14, [rel buf]          ; r14 = 数据缓冲区地址

    ; ---- 1. openat(AT_FDCWD, 源文件, O_RDONLY) --------------------------
    ; 参数顺序：rdi=dirfd, rsi=path, rdx=flags, r10=mode
    mov rax, SYS_openat
    mov rdi, AT_FDCWD           ; -100：相对当前工作目录
    mov rsi, r12                ; 路径
    mov rdx, O_RDONLY           ; 只读
    xor r10d, r10d              ; mode（没有 O_CREAT 时忽略）
    syscall

    test rax, rax
    jns .src_ok                 ; 非负 = 文件描述符
    ; 出错：打印文件名，然后退出
    lea rsi, [rel e_open_s]
    mov rdx, e_open_s_len
    call io_print
    mov rsi, r12
    call cstr_len
    mov rdx, rax
    call io_println
    mov rdi, 2
    call io_exit

.src_ok:
    mov rbx, rax                ; rbx = 源文件描述符

    ; ---- 2. openat(AT_FDCWD, 目标, O_WRONLY|O_CREAT|O_TRUNC, 0644) ------
    mov rax, SYS_openat
    mov rdi, AT_FDCWD
    mov rsi, r13
    mov rdx, O_WRONLY | O_CREAT | O_TRUNC
    mov r10, 0o644              ; 权限（最终权限还要看 umask）
    syscall

    test rax, rax
    jns .dst_ok
    lea rsi, [rel e_open_d]
    mov rdx, e_open_d_len
    call io_print
    mov rsi, r13
    call cstr_len
    mov rdx, rax
    call io_println
    mov rdi, 6
    call io_exit

.dst_ok:
    mov rbp, rax                ; rbp = 目标文件描述符
    xor r15, r15                ; r15 = 累计拷贝的字节数

    ; ---- 3. 读-写循环 ---------------------------------------------------
.read_loop:
    ; read(源fd, buf, buf_size)
    mov rax, SYS_read
    mov rdi, rbx                ; 源 fd
    mov rsi, r14                ; 缓冲区
    mov rdx, buf_size
    syscall                     ; ⚠ 这之后 rcx、r11 都不可信了

    test rax, rax
    js .read_err                ; 负数 = 出错
    jz .done                    ; 0 = EOF，收工

    ; rcx = 本轮读到的字节数。它只能活到下一次 syscall 之前，
    ; 而下面这个写循环全是 syscall……所以先把它转存到 r9，
    ; 用 r9（跨 syscall 安全）当「本轮总数」。
    mov r9, rax                 ; r9 = 本轮读到的字节数

    ; ---- 4. 把 r9 个字节全都写出去（处理「短写」）----------------------
    ; 分工如下（这两个都要跨 syscall 保持，所以不能用 rcx/r11）：
    ;   r9  = 本轮总共要写的字节数
    ;   r10 = 本轮已经写出去的字节数
    xor r10, r10
.write_loop:
    mov rax, SYS_write
    mov rdi, rbp                ; 目标 fd
    mov rsi, r14
    add rsi, r10                ; 从还没写完的位置开始
    mov rcx, r9
    sub rcx, r10                ; 还要写多少 = 总数 - 已写
    mov rdx, rcx
    syscall

    test rax, rax
    jle .write_err              ; 负数出错；0 表示写不动了
    add r10, rax                ; 累计本轮已写
    add r15, rax                ; 累计总字节
    cmp r10, r9
    jb .write_loop              ; 短写：继续把剩下的写完
    jmp .read_loop

.done:
    ; ---- 5. 关闭两个文件 ------------------------------------------------
    mov rax, SYS_close
    mov rdi, rbx
    syscall
    test rax, rax
    js .close_err

    mov rax, SYS_close
    mov rdi, rbp
    syscall
    test rax, rax
    js .close_err

    mov rax, r15                ; 返回拷贝的字节数
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbp
    pop rbx
    ret

.read_err:
    lea rsi, [rel e_read]
    mov rdx, e_read_len
    call io_println
    mov rdi, 3
    call io_exit

.write_err:
    lea rsi, [rel e_write]
    mov rdx, e_write_len
    call io_println
    mov rdi, 4
    call io_exit

.close_err:
    lea rsi, [rel e_close]
    mov rdx, e_close_len
    call io_println
    mov rdi, 5
    call io_exit

; ---------------------------------------------------------------------------
; cstr_len —— 求 rsi 指向的 NUL 结尾字符串长度
;   返回：rax = 长度
; ---------------------------------------------------------------------------
cstr_len:
    xor eax, eax
.loop:
    cmp byte [rsi + rax], 0
    je .done
    inc rax
    jmp .loop
.done:
    ret

; ---------------------------------------------------------------------------
; print_u64 —— 把 rdi 里的无符号数按十进制打印出来
; ---------------------------------------------------------------------------
print_u64:
    push rbp
    mov rbp, rsp
    push rbx
    push r12
    sub rsp, 32                 ; 栈上缓冲区（必须在 push 之后再开）

    mov rax, rdi
    mov rbx, 10
    lea r12, [rsp + 32]
.loop:
    xor rdx, rdx                ; div 前清零 rdx
    div rbx
    add dl, '0'
    dec r12
    mov [r12], dl
    test rax, rax
    jnz .loop

    mov rsi, r12
    lea rdx, [rsp + 32]
    sub rdx, r12
    call io_print

    add rsp, 32
    pop r12
    pop rbx
    leave
    ret

section .bss
    path_src resq 1             ; 保存源文件名指针（打印用）
    path_dst resq 1             ; 保存目标文件名指针（打印用）

; ===========================================================================
; wc.asm —— 第 09 课示例 2：统计文件的字节数 / 行数 / 单词数
;
; 编译（在 lesson09 目录下）：
;   make
; 用法：
;   ./wc 文件名
;   ./wc            # 不写文件名就从标准输入读
; 例子：
;   printf 'hello world\nfoo bar\n' > /tmp/t.txt
;   ./wc /tmp/t.txt
;   printf 'a b\n' | ./wc
;
; 输出三列（和系统的 wc 默认顺序一致）：
;   行数  单词数  字节数  文件名
;
; 和 copy.asm 相比，这里多学三点：
;   1. 统计要「跨缓冲区边界」仍然正确 —— 一个单词可能正好被切成两半，
;      所以「当前是否在单词里」这个状态必须保留到下一轮（存在 rbp 里）；
;   2. 不带参数时从标准输入（fd 0）读，让程序更实用；
;   3. 右对齐输出：先算数字有几位，再补空格（print_padded）。
; ===========================================================================

%define AT_FDCWD -100

section .data
    dash        db "-"
    dash_len    equ $ - dash
    dash_nul    db 0            ; ⚠ cstr_len 需要 NUL 结尾，所以单独补一个 0
                                ;   （这样 dash_len 还是 1，不会被算进去）

    e_open      db "wc: 打不开文件: "
    e_open_len  equ $ - e_open

    e_read      db "wc: 读取出错"
    e_read_len  equ $ - e_read

    buf_size    equ 65536       ; 64 KiB（8 的倍数，保持栈对齐）

section .bss
    argv_base   resq 1          ; 保存 &argv[0]（程序一开始从栈上取下来的）
    buf         resb buf_size   ; 读取缓冲区
    nbytes      resq 1          ; 字节数
    nlines      resq 1          ; 行数
    nwords      resq 1          ; 单词数

section .text
    global _start
default rel                             ; 数据寻址默认 RIP 相对，所有 [标签] 都自动变成 [rel 标签]
bits 64

%include "io.inc"

_start:
    ; ---- 0. 取 argc / argv（必须在任何 push 之前）-----------------------
    mov r12, [rsp]              ; r12 = argc
    lea rax, [rsp + 8]
    mov [rel argv_base], rax    ; 把 &argv[0] 存到 .bss
                                ; （不用寄存器存它：r12-r15 下面都有正经用途）

    mov rbx, STDIN              ; rbx = 要读的文件描述符，默认标准输入
    lea r14, [rel dash]         ; r14 = 显示用的文件名，默认 "-"（NUL 结尾）
                                ;       （很多工具的惯例：stdin 显示成 -）

    cmp r12, 2                  ; 有给文件名吗？
    jl .count                   ; argc < 2：从标准输入读

    mov rax, [rel argv_base]
    mov r14, [rax + 8]          ; r14 = argv[1]，也就是文件名

    ; openat(AT_FDCWD, 文件名, O_RDONLY)
    mov rax, SYS_openat
    mov rdi, AT_FDCWD
    mov rsi, r14                ; 路径
    mov rdx, O_RDONLY
    xor r10d, r10d
    syscall

    test rax, rax
    jns .opened
    ; 打开失败：打印错误并退出（退出码 2）
    lea rsi, [rel e_open]
    mov rdx, e_open_len
    call io_print
    mov rsi, r14
    call cstr_len
    mov rdx, rax
    call io_println
    mov rdi, 2
    call io_exit

.opened:
    mov rbx, rax                ; rbx = 真正的文件描述符

.count:
    ; ---- 1. 初始化计数器 ------------------------------------------------
    mov qword [rel nbytes], 0
    mov qword [rel nlines], 0
    mov qword [rel nwords], 0
    xor ebp, ebp                ; rbp = 「当前是否在单词里」：0 不在 / 1 在

    ; ---- 2. 读-统计循环 -------------------------------------------------
.read_loop:
    mov rax, SYS_read
    mov rdi, rbx                ; 读哪个 fd
    lea rsi, [rel buf]          ; 缓冲区
    mov rdx, buf_size
    syscall                     ; ⚠ 这之后 rcx、r11 不可信

    test rax, rax
    js .read_err                ; 负数 = 出错
    jz .report                  ; 0 = EOF，统计完了

    add [rel nbytes], rax       ; 字节数直接累加

    mov r15, rax                ; r15 = 本轮读到的字节数（跨下面的调用保持）
    xor ecx, ecx                ; rcx = 本轮已处理的下标
                                ;     本轮里没有 syscall，所以用它没问题

.byte_loop:
    lea rsi, [rel buf]
    movzx eax, byte [rsi + rcx] ; 取第 rcx 个字节

    ; ---- 2.1 行数：数 '\n' 的个数 ---------------------------------------
    cmp al, LF
    jne .word_part
    inc qword [rel nlines]

    ; ---- 2.2 单词边界：分隔符 -> 非分隔符 才算一个新单词 ----------------
.word_part:
    xor r8d, r8d                ; r8 = 0：先假设不是分隔符
    cmp al, ' '
    je .is_sep
    cmp al, TAB
    je .is_sep
    cmp al, LF
    je .is_sep
    cmp al, 13                  ; '\r'
    je .is_sep
    cmp al, 12                  ; '\f'
    je .is_sep
    cmp al, 11                  ; '\v'
    je .is_sep
    jmp .sep_done
.is_sep:
    mov r8d, 1
.sep_done:

    test r8d, r8d
    jz .not_sep
    xor ebp, ebp                ; 是分隔符：离开单词状态
    jmp .next_byte

.not_sep:
    test rbp, rbp
    jnz .next_byte              ; 本来就在单词里，不重复计数
    inc qword [rel nwords]      ; 从「不在单词里」变成「在单词里」
    mov ebp, 1

.next_byte:
    inc rcx
    cmp rcx, r15
    jb .byte_loop               ; 本轮还没处理完
    jmp .read_loop              ; 继续读下一块

.report:
    ; ---- 3. 打开过文件就先关掉 ------------------------------------------
    cmp rbx, STDIN
    je .print
    mov rax, SYS_close
    mov rdi, rbx
    syscall                     ; 关闭失败也没关系，统计结果已经拿到了

.print:
    ; ---- 4. 三列输出，每列右对齐 7 位，最后跟文件名 ---------------------
    ; 列的顺序和系统 wc 一样：行、词、字节
    ;
    ; ⚠ 调用约定小提醒：rdi 放数值、rsi 放列宽，两个都要设对。
    ;   如果中间插了 lea rsi, [...] 之类的指令，就会把列宽冲掉。
    mov rdi, [rel nlines]
    mov rsi, 7
    call print_padded

    mov rdi, [rel nwords]
    mov rsi, 7
    call print_padded

    mov rdi, [rel nbytes]
    mov rsi, 7
    call print_padded

    mov edi, ' '
    call io_putchar
    mov rsi, r14                ; 文件名（或者 "-"）
    call cstr_len
    mov rdx, rax
    call io_println

    xor rdi, rdi
    call io_exit

.read_err:
    lea rsi, [rel e_read]
    mov rdx, e_read_len
    call io_println
    mov rdi, 3
    call io_exit

; ---------------------------------------------------------------------------
; print_padded —— 把 rdi 里的无符号数以「右对齐 rsi 列」的方式打印
;
;   入参：rdi = 数值，rsi = 列宽
;   例子：print_padded(42, 7) 会输出 "     42"（5 个空格 + 两位数字）
;   做法：先算数字有几位，再补「列宽 - 位数」个空格。
; ---------------------------------------------------------------------------
print_padded:
    push rbx                    ; 本函数要用 rbx/r12/r13，先保存
    push r12
    push r13

    mov r12, rdi                ; r12 = 要打印的数值
    mov r13, rsi                ; r13 = 列宽

    ; ---- 1. 数一数这个数有几位（0 也算 1 位）----------------------------
    ; 这段没有 syscall，用 rcx 当除数完全安全
    mov rbx, 1                  ; rbx = 位数
    mov rax, r12
.digit_count:
    cmp rax, 10
    jb .counted
    xor rdx, rdx
    mov rcx, 10
    div rcx                     ; rax /= 10
    inc rbx
    jmp .digit_count

.counted:
    mov rax, r13
    sub rax, rbx                ; rax = 需要补的空格数 = 列宽 - 位数
    mov rbx, rax                ; ⚠ 计数器放 rbx（被调用者保存）
                                ;   io_putchar 内部有 syscall，而内核会破坏 rcx；
                                ;   拿 rcx 计数会变成天文数字，最后把栈都冲掉
    test rbx, rbx
    jle .print_num              ; 不需要补空格就直接打印
.pad:
    mov edi, ' '
    call io_putchar
    dec rbx
    jnz .pad

.print_num:
    mov rdi, r12
    call print_u64

    pop r13
    pop r12
    pop rbx
    ret

; ---------------------------------------------------------------------------
; cstr_len —— 求 rsi 指向的 NUL 结尾字符串长度
;   返回：rax = 长度
; ---------------------------------------------------------------------------
cstr_len:
    xor eax, eax
.loop:
    cmp byte [rsi + rax], 0
    je .done
    inc rax
    jmp .loop
.done:
    ret

; ---------------------------------------------------------------------------
; print_u64 —— 无符号十进制输出（栈上开缓冲区，从末尾往前填）
;   入参：rdi = 数值
; ---------------------------------------------------------------------------
print_u64:
    push rbp
    mov rbp, rsp
    push rbx
    push r12
    sub rsp, 32                 ; 缓冲区必须在 push 之后再开

    mov rax, rdi
    mov rbx, 10
    lea r12, [rsp + 32]
.loop:
    xor rdx, rdx                ; div 前清零 rdx
    div rbx
    add dl, '0'
    dec r12
    mov [r12], dl
    test rax, rax
    jnz .loop

    mov rsi, r12
    lea rdx, [rsp + 32]
    sub rdx, r12
    call io_print

    add rsp, 32
    pop r12
    pop rbx
    leave
    ret

; ===========================================================================
; 第 10 课示例 2：lc_float_div.asm
;   整数除法 vs 浮点除法：同一个算式为什么结果差这么多？
;
; 编译运行：
;   nasm -f elf64 -g -F dwarf -I ../include -o lc_float_div.o lc_float_div.asm
;   gcc -no-pie -Wall -O0 -o lc_float_div lc_float_div.o
;   ./lc_float_div
;
; 要点：
;   1. idiv 是整数除法，商向 0 截断，余数在 rdx；7 / 2 = 3 余 1
;   2. divsd 是浮点除法，7.0 / 2.0 = 3.5
;   3. 每个 printf 只打印一个 double（一个 %f 对应一个 xmm），
;      所以每个调用前都是 mov eax, 1；纯整数输出的调用前是 xor eax, eax。
; ===========================================================================

default rel
bits 64

section .data
    seven       dq 7.0
    two         dq 2.0
    three       dq 3.0                        ; 用来做 7.0 / 3.0
    pct_num     dq 17.5                       ; 17.5 / 25.0 = 0.7 = 70%
    pct_den     dq 25.0
    showsign    dq 1234.5678                  ; 演示 %+.2f

    fmt_head    db "=== 整数除法 vs 浮点除法 ===", 10, 0
    fmt_int     db "整数: 7 / 2 = %d 余 %d", 10, 0
    fmt_flt     db "浮点: 7.0 / 2.0 = %f", 10, 0
    fmt_seven3  db "7.0 / 3.0 = %.6f", 10, 0
    fmt_pct     db "17.5 / 25.0 = %.1f%%   (printf 里 %% 打印一个百分号)", 10, 0
    fmt_neg     db "7.0 / (-2.0) = %f", 10, 0
    fmt_sign    db "%%+.2f 会强制显示正负号 : %+.2f", 10, 0
    fmt_bits    db "1234.5678 的 IEEE-754 位模式: 0x%016lx", 10, 0
    fmt_tail    db "结论: 需要小数就必须用浮点指令", 10, 0

section .text
    extern printf
    global main

main:
    push rbp
    mov  rbp, rsp

    lea  rdi, [rel fmt_head]
    xor  eax, eax
    call printf

    ; -----------------------------------------------------------------------
    ; 1) 整数版：7 / 2
    ;    idiv 的规则：被除数在 rdx:rax，商回 rax，余数回 rdx。
    ;    这里被除数只有 64 位，所以先把 rdx 清零。
    ;    注意先算 idiv、再设置 printf 的参数，因为 idiv 会破坏 rdx。
    ; -----------------------------------------------------------------------
    mov  rax, 7
    xor  edx, edx                       ; 64 位被除数的高 64 位 = 0
    mov  rcx, 2
    idiv rcx                            ; rax = 3（商）, rdx = 1（余数）
    mov  rsi, rax                       ; printf 第 2 个参数：商 -> rsi
    lea  rdi, [rel fmt_int]             ; 第 1 个参数：格式串 -> rdi
    xor  eax, eax                       ; 没有浮点参数
    call printf

    ; -----------------------------------------------------------------------
    ; 2) 浮点版：7.0 / 2.0
    ;    divsd 是「标量双精度除法」，只算低 64 位。
    ;    %f 取浮点参数 xmm0，所以 eax = 1。
    ; -----------------------------------------------------------------------
    movsd xmm0, [rel seven]
    divsd xmm0, [rel two]               ; xmm0 = 3.5
    lea  rdi, [rel fmt_flt]
    mov  eax, 1
    call printf

    ; -----------------------------------------------------------------------
    ; 3) 除不尽的时候：7.0 / 3.0 = 2.333333...
    ;    浮点数只能存有限位，%.6f 会四舍五入到 6 位小数。
    ; -----------------------------------------------------------------------
    movsd xmm0, [rel seven]
    divsd xmm0, [rel three]
    lea  rdi, [rel fmt_seven3]
    mov  eax, 1
    call printf

    ; -----------------------------------------------------------------------
    ; 4) 打印百分号：printf 里要写 %% 才输出一个 %
    ;    这里用纯浮点算法得到 70.0：先 17.5/25.0 = 0.7，再乘 100.0。
    ; -----------------------------------------------------------------------
    movsd xmm0, [rel pct_num]
    divsd xmm0, [rel pct_den]           ; xmm0 = 0.7
    mov  ecx, 100
    cvtsi2sd xmm1, ecx                  ; xmm1 = 100.0
    mulsd xmm0, xmm1                    ; xmm0 = 70.0
    lea  rdi, [rel fmt_pct]
    mov  eax, 1                         ; 格式串只读 xmm0（xmm1 只是算中间结果）
    call printf

    ; -----------------------------------------------------------------------
    ; 5) 负数浮点除法：xorpd 把 xmm1 清零，再减去 7.0 得到 -7.0
    ;    除法结果 xmm0 = -3.5，%f 会带出负号
    ; -----------------------------------------------------------------------
    movsd xmm0, [rel seven]
    xorpd xmm1, xmm1                    ; xmm1 = 0.0（按位异或自己清零，比 movsd 0 更常用）
    subsd xmm1, xmm0                    ; xmm1 = 0.0 - 7.0 = -7.0
    movsd xmm0, xmm1                    ; 搬到 xmm0（第 1 个 %f 用 xmm0）
    divsd xmm0, [rel two]               ; xmm0 = -3.5
    lea  rdi, [rel fmt_neg]
    mov  eax, 1
    call printf

    ; -----------------------------------------------------------------------
    ; 6) %+.2f：强制显示正负号；顺便看看 double 在内存里的原始位模式
    ; -----------------------------------------------------------------------
    movsd xmm0, [rel showsign]
    lea  rdi, [rel fmt_sign]
    mov  eax, 1
    call printf

    ; 把 8 字节的 double 当成 64 位整数读出来（mov rsi, [rel ...] 就是原样搬 8 字节）
    ; %016lx 会打印 16 位十六进制（不够前面补 0，对应 IEEE-754 的 1+11+52 位）
    mov  rsi, [rel showsign]
    lea  rdi, [rel fmt_bits]
    xor  eax, eax
    call printf

    lea  rdi, [rel fmt_tail]
    xor  eax, eax
    call printf

    xor  eax, eax
    leave
    ret

; ===========================================================================
; 第 10 课示例 1：lc_float.asm
;   SSE2 标量浮点运算 + libc printf 输出
;
; 编译运行：
;   nasm -f elf64 -g -F dwarf -I ../include -o lc_float.o lc_float.asm
;   gcc -no-pie -Wall -O0 -o lc_float lc_float.o
;   ./lc_float
;
; 要点：
;   1. 文件名以 lc_ 开头、入口是 global main（因为要链 libc，crt1.o 里已有 _start）
;   2. 函数开头 push rbp / mov rbp, rsp，结尾 leave / ret，保证 call 时 rsp 是 16 的倍数
;   3. 调用变参函数 printf 之前，eax 必须等于「本次调用真正用到的向量寄存器个数」。
;      本程序每次只打印一个 double，所以每个 printf 前面都是 mov eax, 1。
;   4. 所有浮点常量都放在 section .data 里，用 [rel 标签] 取
; ===========================================================================

default rel
bits 64

; ---------------------------------------------------------------------------
; 数据段：浮点常量和 printf 格式串
;   dq 定义 8 字节（64 位双精度）
;   NASM 的 dq 直接写十进制小数即可，汇编器会把它转成 IEEE-754 位模式
;
;   注意：格式串里的 % 在 NASM 里照常写。
;   只有「行首以 % 开头」的词才会被当成预处理指令（%define/%macro/%if…），
;   写在 " 双引号字符串里绝对不会被预处理。
; ---------------------------------------------------------------------------
section .data
    a           dq 2.5                  ; 演示用操作数 a = 2.5
    b           dq 7.0                  ; 演示用操作数 b = 7.0
    pi          dq 3.14159265358979     ; 圆周率
    e           dq 2.718281828459045    ; 自然常数

    ; 每个格式串只带一个 %f —— 一个 printf 只取一个 xmm 寄存器
    fmt_a       db "a = %.6f   (位模式 0x%016lx)", 10, 0
    fmt_b       db "b = %.6f   (位模式 0x%016lx)", 10, 0
    fmt_sum     db "a + b = %.6f", 10, 0
    fmt_diff    db "a - b = %.6f", 10, 0
    fmt_prod    db "a * b = %.6f", 10, 0
    fmt_quot    db "a / b = %.6f", 10, 0
    fmt_pi      db "pi    = %.10f", 10, 0
    fmt_e       db "e     = %.10f", 10, 0
    fmt_sqrt    db "sqrt(2) = %.10f", 10, 0
    fmt_cvt     db "%d 转成 double 后是 %.1f", 10, 0
    fmt_done    db "浮点运算演示结束", 10, 0

section .text
    extern printf                       ; 声明用到的 libc 函数
    global main

main:
    ; ---- 标准函数序言：这两条让 rsp 在 call 时是 16 的倍数 ----------------
    push rbp
    mov  rbp, rsp

    ; ---- 打印 a 和 b 本身，顺便看看 double 在内存里的位模式 ----------------
    ; 第 1 个整数参数是格式串 -> rdi；%lx 取第 1 个整数参数 -> rsi
    ; 浮点参数 a -> xmm0，所以 eax = 1
    movsd xmm0, [rel a]
    mov  rsi, [rel a]                   ; 把 8 字节原样当整数读出来（0x4004000000000000）
    lea  rdi, [rel fmt_a]
    mov  eax, 1                         ; 只用了 xmm0 这 1 个向量寄存器
    call printf

    movsd xmm0, [rel b]
    mov  rsi, [rel b]
    lea  rdi, [rel fmt_b]
    mov  eax, 1
    call printf

    ; ---- 四种标量双精度运算 ------------------------------------------------
    ; movsd = move scalar double：只搬低 64 位，高 64 位保持不动。
    ; addsd / subsd / mulsd / divsd 是「标量」运算，只算低 64 位的那个 double；
    ; 别写成 addps / addpd，那是对打包的多组浮点同时运算（第 11 课讲）。
    movsd xmm0, [rel a]
    addsd xmm0, [rel b]                 ; xmm0 = 2.5 + 7.0 = 9.5
    lea  rdi, [rel fmt_sum]
    mov  eax, 1
    call printf

    movsd xmm0, [rel a]
    subsd xmm0, [rel b]                 ; xmm0 = 2.5 - 7.0 = -4.5
    lea  rdi, [rel fmt_diff]
    mov  eax, 1
    call printf

    movsd xmm0, [rel a]
    mulsd xmm0, [rel b]                 ; xmm0 = 2.5 * 7.0 = 17.5
    lea  rdi, [rel fmt_prod]
    mov  eax, 1
    call printf

    movsd xmm0, [rel a]
    divsd xmm0, [rel b]                 ; xmm0 = 2.5 / 7.0 = 0.357142857...
    lea  rdi, [rel fmt_quot]
    mov  eax, 1
    call printf

    ; ---- 真实常量的精度：%f 默认 6 位小数，%.10f 给 10 位 -------------------
    movsd xmm0, [rel pi]
    lea  rdi, [rel fmt_pi]
    mov  eax, 1
    call printf

    movsd xmm0, [rel e]
    lea  rdi, [rel fmt_e]
    mov  eax, 1
    call printf

    ; ---- sqrtsd：SSE2 的标量平方根 ----------------------------------------
    ; 2 是整数常量，可以用 cvtsi2sd 直接从 32 位整数转成 double，
    ; 这样就不必为了一个 2.0 再在 .data 里多定义一个 8 字节常量。
    ; 注意 cvtsi2sd 会清零 xmm0 的高 64 位。
    mov  eax, 2
    cvtsi2sd xmm0, eax                  ; xmm0 = (double)2 = 2.0
    sqrtsd xmm0, xmm0                   ; xmm0 = sqrt(2.0) = 1.414213562...
    lea  rdi, [rel fmt_sqrt]
    mov  eax, 1
    call printf

    ; ---- 整数 -> 浮点：一次调用里同时用整数参数和浮点参数 -------------------
    ; %d 取整数参数 rsi，%f 取浮点参数 xmm0；eax 仍然是 1，因为只用了 1 个 xmm。
    mov  rsi, 42
    mov  ecx, 42
    cvtsi2sd xmm0, ecx
    lea  rdi, [rel fmt_cvt]
    mov  eax, 1
    call printf

    lea  rdi, [rel fmt_done]
    xor  eax, eax                       ; 一个浮点参数都没用 -> eax = 0
    call printf

    ; 退出码 0；对应序言的 leave 是 mov rsp, rbp + pop rbp
    xor  eax, eax
    leave
    ret

; 第 11 课示例 1：lc_simd_intro.asm
;   SIMD 入门：一条 paddd 指令同时做 8 组 32 位整数加法

;   movdqu xmm, [mem] / movdqu [mem], xmm   一次搬 16 字节（不要求对齐），SSE2

;   paddd  xmm, xmm      8 组 32 位整数逐组相加（packed add dword），SSE2
;   pmulld xmm, xmm      8 组 32 位整数逐组相乘，**SSE4.1**
;   movq   r64, xmm      把 xmm 低 64 位搬到通用寄存器，SSE2

; 所以每个 printf 之前都是 xor eax, eax。
; ===========================================================================

; ---------------------------------------------------------------------------
; 怎么把 xmm 里的 8 个 int32 拿出来打印？
;
; 最直接的想法是用 pextrd r32, xmm, n 逐个取。它属于 SSE4.1，而且
; 在部分 CPU/虚拟化环境下，n ≥ 4 的那些 lane 结果不对：
;     pextrd eax, xmm0, 4   可能又返回 lane 0 的值
; 这是实测踩到的坑（见 README「常见坑」第 6 条）。所以这里换成最稳的做法：
;
;   先把 SIMD 结果用 movdqa / movdqu 整块写回内存（vec_sum 等），
;   再用普通的 mov 从内存里按 dword 偏移逐个取值交给 printf。
;
; 这样做的好处：
;   * 完全不依赖 pextrd 之类「取 lane」的指令，SSE2 机器也能跑；
;   * 用 32 位子寄存器（esi/edx/ecx/r8d）接收，%d 看到的就正好是那个 int32，
;     不用担心 64 位寄存器高位有脏数据；
;   * 顺便说明一个道理：SIMD 的「一条指令处理多组数据」是真的，
;     但要把结果交给 printf，还是得一个个搬到通用寄存器/内存。
; ---------------------------------------------------------------------------
; ---------------------------------------------------------------------------
; PRINT_VEC — 把内存里连续 8 个 int32 分两行打印
;   %1 = 数据标签（8 个 int32）
;   %2 = 前半行格式串（4 个 %d，不带换行）
;   %3 = 后半行格式串（4 个 %d，带换行）
;
; 为什么分两次 printf：printf 的整数参数只有 6 个（rdi/rsi/rdx/rcx/r8/r9），
; 格式串占掉 rdi，剩下 5 个位置。一次打印 8 个元素就必须压栈并自己保证
; 16 字节栈对齐；拆成 4 + 4 就完全不碰栈，初学者少踩一个坑。
;
; printf 会破坏 rsi/rdx/rcx/r8 这些**调用者保存**寄存器，所以后半行
; 必须重新从内存装一遍参数，不能沿用前半行的寄存器。
; ---------------------------------------------------------------------------
%macro PRINT_VEC 3
    ; ---- 前半行：lane 0..3（用 32 位子寄存器，%d 只需要低 32 位）----
    mov  esi, [rel %1 + 0]
    mov  edx, [rel %1 + 4]
    mov  ecx, [rel %1 + 8]
    mov  r8d, [rel %1 + 12]
    lea  rdi, [rel %2]
    xor  eax, eax
    call printf

    ; ---- 后半行：lane 4..7（printf 会破坏 rsi/rdx/rcx/r8，所以重新装）----
    mov  esi, [rel %1 + 16]
    mov  edx, [rel %1 + 20]
    mov  ecx, [rel %1 + 24]
    mov  r8d, [rel %1 + 28]
    lea  rdi, [rel %3]
    xor  eax, eax
    call printf
%endmacro

; ---------------------------------------------------------------------------
; check_result —— 自检：这个宿主机/虚拟机的 SIMD 到底算对了几个 lane？
;
; 入参：
;   rdi = 结果数组（8 个 int32）
;   esi = 前 4 个 lane 的期望和（汇编期算好的立即数）
;   edx = 全部 8 个 lane 的期望和（立即数）
; 返回：无（直接打印自检结论）
;
; 为什么用立即数当期望值，而不是在运行时再算一遍：
;   * 初学阶段最好的校验标准就是「先手算好，再让程序对答案」；
;   * 也避免校验代码本身依赖太多寄存器状态。
;
; 现实意义：本项目在 Debian 13 的一台虚拟机上实测到，
;   paddd / pmulld 这类「多 lane 32 位」指令只算了低 64 位
;   （第 5~8 个元素恒为 0），而 paddq / paddb / movdqu 都正常。
;   这是宿主机/虚拟化层的问题，不是程序写错了 —— 程序自己把它指出来，
;   学员就不会怀疑自己的代码。正常 CPU 上一定打印「8 个 lane 全部正确」。
; ---------------------------------------------------------------------------
check_result:
    push rbx
    push rbp

    ; ---- 1) 前 4 个 lane 的和（这些 lane 在所有机器上都应该是对的）----
    xor  ebx, ebx
    add  ebx, [rdi + 0]
    add  ebx, [rdi + 4]
    add  ebx, [rdi + 8]
    add  ebx, [rdi + 12]

    ; ---- 2) 全部 8 个 lane 的和 ----
    mov  ebp, ebx
    add  ebp, [rdi + 16]
    add  ebp, [rdi + 20]
    add  ebp, [rdi + 24]
    add  ebp, [rdi + 28]

    ; ---- 3) 顺便数一下非零 lane 的个数 ----
    xor  ecx, ecx                       ; 非零 lane 计数
    xor  r8d, r8d                       ; 下标
.count:
    mov  eax, [rdi + r8*4]
    test eax, eax
    jz   .skip
    inc  ecx
.skip:
    inc  r8d
    cmp  r8d, 8
    jb   .count

    lea  r9, [rel fmt_chk_sum]
    mov  r10d, ecx                      ; 保存计数（printf 会破坏寄存器）
    mov  rdi, r9
    mov  r8d, ebx                       ; 低 4 lane 实测和
    mov  ecx, ebp                       ; 全部 8 lane 实测和

    mov  r9d, edx                       ; 期望的 8 lane 和（第 4 个参数位置）
    xor  eax, eax
    push r10                            ; 借一个栈槽暂存计数（保持 8 字节对齐）
    mov  esi, r8d                       ; 参数 1：8 lane 实测和
    mov  edx, r9d                       ; 参数 2：期望的 8 lane 和
    mov  ecx, ebx                       ; 参数 3：低 4 lane 实测和
    mov  r8d, r10d                      ; 参数 4：非零 lane 个数
    call printf
    pop  r10

    ; ---- 4) 结论 ----
    cmp  r10d, 8
    jne  .bad
    lea  rdi, [rel fmt_chk_ok]
    jmp  .print
.bad:
    lea  rdi, [rel fmt_chk_bad]
.print:
    xor  eax, eax
    call printf

    pop  rbp
    pop  rbx
    ret

; ===========================================================================
; 第 11 课示例 2：simd_strlen.asm
;   用 pcmpeqb + pmovmskb + bsf 一次查 16 个字节找字符串结尾，
;   和「逐字节比较」的做法对比。
;
; ★ 入口说明：本文件是**纯系统调用程序**，不是 libc 程序。
;   * 入口是 _start（不是 main），文件名也没有 lc_ 前缀 —— 这正是项目约定：
;     文件名不以 lc_ 开头的示例用 nasm + ld 链接，入口必须是 _start。
;   * 输出用 ../include/io.inc 里的 io_print / io_println / io_print_uint，
;     完全不碰 libc（没有 crt1.o 做初始化时直接 call printf 会崩）。
;   * 第 11 课的两个示例故意一个用 libc（lc_simd_intro.asm，入口 main）、
;     一个用纯系统调用（本文件，入口 _start），正好复习第 1 课讲的
;     「两种风格绝不能混」。
;
; 编译运行：
;   nasm -f elf64 -g -F dwarf -I ../include -o simd_strlen.o simd_strlen.asm
;   ld -o simd_strlen simd_strlen.o
;   ./simd_strlen
;
; 用到的指令（全部 SSE2）：
;   pxor      xmm, xmm          把 xmm 清零
;   movdqu    xmm, [mem]        一次读 16 字节
;   movdqa    xmm, [mem]        一次读 16 字节（要求地址 16 字节对齐）
;   pcmpeqb   xmm, xmm          16 个字节逐字节比较，相等则那个字节变 0xFF
;   pmovmskb  r32, xmm          把 16 个字节的最高位收成一个 16 位掩码
;   bsf       r32, r32          找最低位的 1 在哪一位（bit scan forward）
;     掩码的第 i 位 = 1 表示第 i 个字节是 '\0'，bsf 一下就得到它的下标
; ===========================================================================

default rel                             ; 数据寻址默认 RIP 相对
bits 64

%include "io.inc"                       ; ⚠ 一个程序里只能 %include 一次

section .data
    s1      db "Hello, SIMD!", 0
    s2      db "x", 0
    s3      db "SIMD 让 16 个字节一次比完", 0
    s4      db "AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA", 0

    m_head  db "=== strlen: 逐字节 vs SIMD ==="
    m_head_len equ $ - m_head
    m_cmp   db "  两种实现结果一致: "
    m_cmp_len equ $ - m_cmp
    m_yes   db "是", 10
    m_yes_len equ $ - m_yes
    m_no    db "否（有 bug！）", 10
    m_no_len equ $ - m_no
    m_s     db "  串: ["
    m_s_len equ $ - m_s
    m_e     db "]", 10
    m_e_len equ $ - m_e
    m_l1    db "  逐字节比较 strlen: "
    m_l1_len equ $ - m_l1
    m_l2    db "        SIMD strlen: "
    m_l2_len equ $ - m_l2

section .text
    global _start

; ===========================================================================
; strlen_byte —— 教科书式逐字节实现
; 入参：rdi = 字符串首地址
; 返回：rax = 长度（不含结尾的 '\0'）
; 特点：每读 1 个字节判断一次，长度为 N 就要循环 N 次。
; ===========================================================================
strlen_byte:
    xor  rax, rax                       ; rax = 已经数过的字节数
.loop:
    cmp  byte [rdi + rax], 0            ; 当前字节是 '\0' 吗？
    je   .done
    inc  rax
    jmp  .loop
.done:
    ret

; ===========================================================================
; strlen_simd —— 一次处理 16 字节
; 入参：rdi = 字符串首地址
; 返回：rax = 长度
;
; 主循环：把字符串按「相对 rdi 的偏移」切成 16 字节块。
;   每块先用 movdqu 读进来，pcmpeqb 和全 0 比较（相等的字节变 0xFF），
;   再用 pmovmskb 把 16 个字节的最高位收成 16 位掩码：
;       bit i = 1  表示第 i 个字节是 '\0'
;   掩码为 0 说明这 16 字节里没有 '\0'，偏移加 16 继续。
;   掩码不为 0，bsf 找出最低的那个 1，长度 = 偏移 + 位号。
;
; 收尾：如果这个 16 字节块会越过字符串末尾（连带 '\0' 一起读会跨页），
;   就退回到「16 字节对齐块」的安全做法：
;     1. 把 rdi 向下对齐到 16 字节边界，整块用 movdqa 读 16 字节；
;        这样读的地址永远在字符串所在的那一页里，不会踩到未映射页。
;     2. 但块里可能包含字符串起点之前的字节，它们也可能碰巧是 0，
;        会给掩码贡献一些「假」的 1。把它们清掉：
;        对于对齐块内的下标 p，它对应的绝对偏移是 base+p；
;        只有 base+p ≥ rdi 才是字符串自己的字节。
;        用 1<<p 生成每一位，并从低位往高位累加，就能把
;        「绝对偏移 < rdi」的位全部去掉，只保留 ≥ rdi 的位。
;     3. bsf 找最低的 1，长度 = 对齐起点 + p - rdi。
; ===========================================================================
strlen_simd:
    pxor xmm0, xmm0                     ; xmm0 = 全 0
    xor  r10, r10                       ; r10 = 相对 rdi 的偏移
    mov  r11, rdi
    and  r11, -4096                     ; r11 = 起点所在页的首地址
    add  r11, 4096                      ; r11 = 下一页的首地址（本页的上界）

.chunk:
    ; 安全检查：这个 16 字节块会不会跨过页尾？
    ; 会跨页就退回 .tail —— 否则 movdqu 可能读到下一页，触发段错误。
    lea  rax, [rdi + r10 + 15]          ; 本块最后一个字节的地址
    cmp  rax, r11
    jae  .tail                          ; 已经到下一页了，别读
    movdqu xmm1, [rdi + r10]
    pcmpeqb xmm1, xmm0
    pmovmskb ecx, xmm1
    test ecx, ecx
    jnz  .found
    add  r10, 16
    jmp  .chunk

.found:
    bsf  ecx, ecx
    lea  rax, [r10 + rcx]               ; 长度 = 块偏移 + 位号
    ret

.tail:
    ; 把 rdi 向下对齐到 16 字节边界，整块用 movdqa 读 —— 地址永远在
    ; 字符串起点所在的页里（起点本身一定合法），不会踩到未映射页。
    mov  r9, rdi
    and  r9, -16                        ; r9 = 对齐后的地址
    movdqa xmm1, [r9]
    pcmpeqb xmm1, xmm0
    pmovmskb ecx, xmm1                  ; 对齐块里所有 '\0' 的位置

    ; 对齐块里可能包含字符串起点之前的字节，它们也可能为 0，
    ; 会给掩码贡献「假」的 1。把绝对位置 < rdi 的那些位清掉：
    ; 第 p 位对应的绝对位置是 r9 + p，要求 r9 + p ≥ rdi。
    mov  r10, rdi
    sub  r10, r9                        ; r10 = rdi 在块内的下标（0..15）
    mov  eax, 1                         ; eax = 1<<p
    xor  r8d, r8d                       ; r8 = 允许保留的位
    xor  edx, edx                       ; edx = p
.keep:
    cmp  rdx, r10
    jae  .kept
    or   r8d, eax                       ; p < 起点下标：这个位置的对齐字节是
    add  eax, eax                       ; 「假」的，不能算进掩码
    inc  rdx
    jmp  .keep
.kept:
    test r10, r10
    jz   .have_mask                     ; 起点本来就 16 字节对齐：块内没有假位
    and  ecx, r8d                       ; 否则去掉绝对位置 < rdi 的假位
.have_mask:
    test ecx, ecx                       ; 字符串的 '\0' 一定在这块里，理论上不会为 0
    jz   .zero
    bsf  ecx, ecx
    lea  rax, [r9 + rcx]
    sub  rax, rdi                       ; 长度 = 绝对位置 - 起点
    ret
.zero:
    xor  eax, eax
    ret

; ===========================================================================
; print_one —— 打印一个字符串，再用两种实现各算一次长度并比较
; 入参：rdi = 字符串
; 说明：io_* 例程只破坏 rax/rcx/r11，其余寄存器调用前后不变（见 io.inc 头注释），
;       所以这里可以用 r14/r15 保存中间结果，不必反复压栈。
; ===========================================================================
print_one:
    mov  r14, rdi                       ; r14 = 字符串首地址

    ; ---- 第 1 行：串: [xxxxx] ----
    lea  rsi, [rel m_s]
    mov  rdx, m_s_len
    call io_print
    mov  rdi, r14
    call strlen_byte
    mov  r15, rax                       ; r15 = 逐字节算出的长度（后面比较用）
    mov  rsi, r14
    mov  rdx, r15
    call io_print
    lea  rsi, [rel m_e]
    mov  rdx, m_e_len
    call io_print

    ; ---- 第 2 行：逐字节比较 strlen: n ----
    lea  rsi, [rel m_l1]
    mov  rdx, m_l1_len
    call io_print
    mov  rdi, r15
    call io_print_uint
    call io_newline

    ; ---- 第 3 行：SIMD strlen: n ----
    lea  rsi, [rel m_l2]
    mov  rdx, m_l2_len
    call io_print
    mov  rdi, r14
    call strlen_simd
    mov  r12, rax                       ; r12 = SIMD 算出的长度
    mov  rdi, rax
    call io_print_uint
    call io_newline

    ; ---- 比较两者是否一致 ----
    lea  rsi, [rel m_cmp]
    mov  rdx, m_cmp_len
    call io_print
    cmp  r12, r15
    jne  .mismatch
    lea  rsi, [rel m_yes]
    mov  rdx, m_yes_len
    call io_print
    ret
.mismatch:
    lea  rsi, [rel m_no]
    mov  rdx, m_no_len
    call io_print
    ret

_start:
    lea  rsi, [rel m_head]
    mov  rdx, m_head_len
    call io_println
    call io_newline

    lea  rdi, [rel s1]
    call print_one
    call io_newline

    lea  rdi, [rel s2]
    call print_one
    call io_newline

    lea  rdi, [rel s3]
    call print_one
    call io_newline

    lea  rdi, [rel s4]
    call print_one

    mov  rdi, 0
    call io_exit

; ===========================================================================
; 第 12 课示例 2：lc_inline.asm
;   * 一个能**单独运行**的 libc 程序（入口 main），演示「汇编函数怎么调用
;     C 库函数 printf」；
;   * 同时它导出的两个函数 asm_flag / scale_shift 会被同目录的
;     lc_inline.c（用 GCC 内联汇编写的小程序）链接进去，演示 C 与汇编互调。
;
; 单独编译运行：
;   nasm -f elf64 -g -F dwarf -I ../include -o lc_inline.o lc_inline.asm
;   gcc -no-pie -Wall -O0 -o lc_inline lc_inline.o
;   ./lc_inline
;
; 和 lc_inline.c 一起编译运行：
;   nasm -f elf64 -g -F dwarf -I ../include -o lc_inline.o lc_inline.asm
;   gcc -no-pie -Wall -O0 -o lc_inline_c lc_inline.c lc_inline.o
;   ./lc_inline_c
;
; 说明：本文件里不出现任何 __asm__ —— 内联汇编是 **C 语言的特性**，
;       只有 lc_inline.c 里才有；这里是用 NASM 写的「对照版」。
;
; 调用约定复习（SysV AMD64）：
;   整数参数 rdi, rsi, rdx, rcx, r8, r9；浮点参数 xmm0..xmm7
;   调用变参函数 printf 前，eax = 本次调用用到的向量寄存器个数
;   call 时 rsp 必须是 16 的倍数 -> 函数开头 push rbp / mov rbp, rsp
; ===========================================================================

default rel
bits 64

section .data
    ; 每个 printf 只带一个 %f：一个格式串对应一个 xmm 寄存器，
    ; 这样 eax 永远是 1，不容易数错。
    f_shift     db "汇编函数 scale_shift(2.0, 0.5) = %.6f   (期望 4.5)", 10, 0
    f_flag1     db "汇编函数 asm_flag(%ld) 通过 printf 打印:", 10, 0
    f_flag2     db "  [asm] n = %ld", 10, 0
    f_head      db "=== 汇编版对照程序（真正的内联汇编在 lc_inline.c）===", 10, 0

    d_two       dq 2.0
    d_half      dq 0.5
    f_cvt       db "cvtsi2sd: %ld -> %.1f", 10, 0

section .text
    extern printf                       ; 引用 C 库函数
%ifndef NO_MAIN
    global main                         ; 自测入口；和 C 混链时用 -DNO_MAIN 去掉它
%endif
    global asm_flag
    global scale_shift
    global asm_double_int

; ---------------------------------------------------------------------------
; asm_flag(n) —— 汇编函数里调用 C 库的 printf
; 入参：rdi = n（64 位整数，由 C 或 NASM 按调用约定传进来）
; 要点：
;   * 要保留 rdi 跨过第一个 printf，所以先在序言里把 rdi 存到栈上；
;     存 1 个 8 字节会让 rsp 变成 8 mod 16，call 前必须补一个 8 字节填充，
;     否则 printf 内部的 movaps 会因 16 字节未对齐而崩。
;   * 变量的位置：-8(%rbp) 和 -16(%rbp)（push 的份也算在内）。
; ---------------------------------------------------------------------------
asm_flag:
    push rbp
    mov  rbp, rsp
    sub  rsp, 16                        ; 两个 8 字节栈槽：n 和填充
    mov  [rbp-8], rdi                   ; 保存 n

    lea  rdi, [rel f_flag1]             ; 第 1 个参数：格式串
    mov  rsi, [rbp-8]                   ; 第 2 个参数：n（%ld 需要整数参数）
    xor  eax, eax                       ; 没用向量寄存器
    call printf

    lea  rdi, [rel f_flag2]
    mov  rsi, [rbp-8]
    xor  eax, eax
    call printf

    leave                               ; = mov rsp, rbp + pop rbp
    ret

; ---------------------------------------------------------------------------
; scale_shift(x, s) -> x * 2 + s
; 入参：xmm0 = x, xmm1 = s；返回：xmm0
; 这个函数会被 lc_inline.c 里的同名绑定（register double ... asm("xmm0")）
; 调用，正好复用同一套寄存器位置。
; ---------------------------------------------------------------------------
scale_shift:
    addsd xmm0, xmm0                    ; xmm0 = 2x
    addsd xmm0, xmm1                    ; xmm0 = 2x + s
    ret                                 ; 返回值已经在 xmm0 里了

; ---------------------------------------------------------------------------
; asm_double_int(n) -> 2n
; 入参：rdi = n；返回：rax = 2n
; 这是「整数参数、整数返回」的最简单互调例子：
; lc_inline.c 里的 call_asm_double_int 会显式 movq 到 rdi、call、再从 rax 取回。
; 它不调用任何 libc 函数，所以是叶子函数，不需要栈帧。
; ---------------------------------------------------------------------------
asm_double_int:
    lea  rax, [rdi + rdi]               ; rax = rdi * 2（lea 比 imul 更短）
    ret

; ---------------------------------------------------------------------------
; 单独的 main：不依赖 lc_inline.c，自己就能跑
; ---------------------------------------------------------------------------
%ifndef NO_MAIN
main:
    push rbp
    mov  rbp, rsp

    lea  rdi, [rel f_head]
    xor  eax, eax
    call printf

    ; ---- 直接调用 scale_shift：按 ABI 把参数放进 xmm0/xmm1 ----
    movsd xmm0, [rel d_two]             ; 2.0
    movsd xmm1, [rel d_half]            ; 0.5
    call scale_shift                    ; 返回 4.5，在 xmm0 里
    lea  rdi, [rel f_shift]             ; 注意：格式串 -> rdi，和 xmm0 不冲突
    mov  eax, 1                         ; 用了 1 个向量寄存器（xmm0）
    call printf

    ; ---- 演示 cvtsi2sd：整数 -> double，然后交给 printf 打印 ----
    mov  rsi, 42
    mov  ecx, 42
    cvtsi2sd xmm0, ecx                  ; xmm0 = (double)42
    lea  rdi, [rel f_cvt]
    mov  eax, 1
    call printf

    ; ---- 调用自己写的 asm_flag，它会再调 printf ----
    mov  rdi, 99
    call asm_flag

    xor  eax, eax
    leave
    ret

%endif ; NO_MAIN

/* ===========================================================================
 * 第 12 课示例 2 的 C 部分：lc_inline.c
 *   GCC 内联汇编（inline assembly）入门
 *
 *   注意：内联汇编是 **C 语言**的特性，NASM 里没有 __asm__ 这种东西。
 *   本文件的角色是「C 驱动」：里面用 __asm__ volatile 写几个小函数，
 *   再调用同目录 lc_inline.asm（NASM）提供的 asm_flag / scale_shift，
 *   演示 C 与汇编互相调用，并用 printf 验证结果。
 *
 * 编译运行（要把同目录的 lc_inline.o 一起链进来）：
 *   nasm -f elf64 -g -F dwarf -I ../include -o lc_inline.o lc_inline.asm
 *   gcc -no-pie -Wall -O0 -o lc_inline_c lc_inline.c lc_inline.o
 *   ./lc_inline_c
 *
 * ---------------------------------------------------------------------------
 * 五条最基本的 AT&T / Intel 语法差异（gcc 内联汇编默认 AT&T）：
 *
 *   1. 操作数顺序反过来：AT&T 是「源在前、目的在后」
 *        AT&T : movl %eax, %ebx      （ebx = eax）
 *        Intel: mov  ebx, eax       （ebx = eax）
 *   2. 寄存器要加 %，内联汇编里自己的寄存器要写两个 %
 *        GCC 里的 %0/%1 是操作数占位符，所以裸寄存器必须写 %%rax
 *   3. 立即数要加 $：AT&T 的 $2 等于 Intel 的 2
 *   4. 助记符后缀表示数据宽度：b=8 位, w=16 位, l=32 位, q=64 位
 *        addl %eax, %ebx   /  addq %rax, %rbx
 *   5. 内存操作数带括号和段前缀：AT&T 的 (%rdi) 等于 Intel 的 [rdi]
 * ========================================================================= */

#include <stdio.h>

/* ---------------------------------------------------------------------------
 * 1) add_two(x) -> x + 2
 *    "r"  = 让 gcc 任选一个通用寄存器；"=r" 是只写输出
 *    %0 是第 1 个操作数（输出），%1 是第 2 个（输入）
 *    AT&T 的 addl %1, %0 等于 Intel 的 add %0, %1，即 %0 = %0 + %1
 * ------------------------------------------------------------------------- */
static long add_two(long x)
{
    long out;
    __asm__ volatile (
        "movq  %1, %0\n\t"      /* out = x（"=r" 是只写，所以先搬一次） */
        "addq  $2, %0\n\t"      /* out += 2（立即数要写 $2） */
        : "=r" (out)
        : "r"  (x)
        : "cc"                  /* 声明改动了标志位 */
    );
    return out;
}

/* ---------------------------------------------------------------------------
 * 2) double_it(x) -> 2x
 *    这里演示「GCC 帮你搬运」：两个操作数可以落在同一个寄存器里
 *    （x 用完了就没人再要它，gcc 允许复用），于是指令数最少。
 * ------------------------------------------------------------------------- */
static long double_it(long x)
{
    long out;
    __asm__ volatile (
        "movq  %1, %0\n\t"
        "addq  %1, %0\n\t"      /* out = x + x */
        : "=r" (out)
        : "r"  (x)
        : "cc"
    );
    return out;
}

/* ---------------------------------------------------------------------------
 * 3) fma_like(v) -> 3 * v * v
 *    约束 "t" 表示「任选一个 xmm 寄存器」，用于 double/float 运算
 *    addsd/mulsd 是**标量**双精度指令（只算低 64 位那个 double）
 * ------------------------------------------------------------------------- */
static double fma_like(double v)
{
    double out;
    __asm__ (
        "movsd %1, %0\n\t"      /* out = v */
        "addsd %1, %0\n\t"      /* out = 2v */
        "addsd %1, %0\n\t"      /* out = 3v */
        "mulsd %1, %0\n\t"      /* out = 3v * v = 3v^2 */
        /* 输出用 "=&x"（early-clobber）：第一条 movsd 在读完输入之前就写了
           输出寄存器。不加 & 的话 gcc 可能让输出和输入共用同一个 xmm，
           第一条 movsd 会先把 v 覆盖掉，结果全错。
           这里用 "=x"（SSE 寄存器类）而不是 "=t"（栈顶寄存器 st0，属于 x87），
           再用 & 让 gcc 另找一个空闲的 xmm。 */
        : "=&x" (out)
        : "x"  (v)
    );
    return out;                 /* 2.0 -> 12.0 */
}

/* ---------------------------------------------------------------------------
 * 4) 用 GCC 的固定寄存器变量对上 SysV 调用约定
 *    register ... asm("xmm0") 告诉 gcc：这个变量请放在 xmm0。
 *    下面这个函数体里一条指令都没写，纯粹靠 gcc 把参数放进 xmm0/xmm1，
 *    正好就是 NASM 那边 scale_shift 期望的位置 —— 于是可以直接互调。
 *
 *    为什么不在内联汇编里手写 call：
 *      call 会破坏所有调用者保存寄存器，而内联汇编的输入/输出约束
 *      也可能被分配成 xmm0/xmm1，稍微写错就是「输出寄存器被调用破坏」。
 *      教学上更稳的做法是用两个小函数把寄存器搬运交给编译器 ——
 *      本函数就是那个「中转站」。
 * ------------------------------------------------------------------------- */
extern double scale_shift(double x, double s);      /* 由 lc_inline.asm 提供 */

static inline double scale_shift_native(double x, double s)
{
    register double a0 __asm__("xmm0") = x;
    register double a1 __asm__("xmm1") = s;
    __asm__ volatile ("" : "+x" (a0), "+x" (a1));   /* 只是把值钉在 xmm0/xmm1
                                                       （约束 x = SSE 寄存器；t 是 x87 的 st0，不要混用） */
    return a0 * 2.0 + a1;
}

/* ---------------------------------------------------------------------------
 * 4b) 直接用内联汇编调用 NASM 的整数函数
 *     "S" 约束 = rsi，"D" = rdi，"a" = rax。
 *     这里先显式写出寄存器搬运，再 call —— 每个操作数都算得出确切位置，
 *     不存在「输出寄存器被调用破坏」的问题。
 * ------------------------------------------------------------------------- */
extern long asm_double_int(long x);                 /* 由 lc_inline.asm 提供 */

static long call_asm_double_int(long x)
{
    long ret;
    __asm__ volatile (
        "movq  %1, %%rdi\n\t"       /* 第 1 个整数参数放 rdi */
        "call  asm_double_int\n\t"
        "movq  %%rax, %0\n\t"       /* 返回值在 rax */
        : "=r" (ret)
        : "r"  (x)
        : "rdi", "rax", "rcx", "rdx", "rsi", "r8", "r9", "r10", "r11",
          "xmm0", "xmm1", "xmm2", "xmm3", "xmm4", "xmm5", "xmm6",
          "xmm7", "xmm8", "xmm9", "xmm10", "xmm11", "xmm12", "xmm13",
          "xmm14", "xmm15", "cc", "memory"
    );
    return ret;
}

/* ---------------------------------------------------------------------------
 * 5) 跨文件调用 NASM 里的 asm_flag（它内部会调用 libc 的 printf）
 * ------------------------------------------------------------------------- */
extern void asm_flag(long n);

int main(void)
{
    printf("1. add_two(40)          = %ld\n", add_two(40));           /* 42 */
    printf("2. double_it(21)        = %ld\n", double_it(21));         /* 42 */
    printf("3. fma_like(2.0)        = %.6f\n", fma_like(2.0));        /* 12.0 */
    printf("4. 汇编 scale_shift(5.0, 1.5) = %.6f\n",
           scale_shift_native(5.0, 1.5));                              /* 11.5 */
    printf("5. 汇编 asm_double_int(21)    = %ld\n",
           call_asm_double_int(21));                                   /* 42 */

    asm_flag(7);                    /* 汇编函数反过来调用 printf */

    printf("7. 内联汇编演示结束\n");
    return 0;
}

; ===========================================================================
; 第 12 课示例 1：macro_demo.asm
;   NASM 预处理：%define、%macro/%endmacro、带参数的宏、%rep、%ifdef/%if
;
; ★ 入口说明：本文件是**纯系统调用程序**（文件名没有 lc_ 前缀）：
;   入口是 _start，用 nasm + ld 链接，输出走 ../include/io.inc。
;
; 编译运行：
;   nasm -f elf64 -g -F dwarf -I ../include -o macro_demo.o macro_demo.asm
;   ld -o macro_demo macro_demo.o
;   ./macro_demo
;
; 本课最重要的一个概念：
;   所有 % 开头的东西（%define/%macro/%rep/%if/%include/%assign…）都在
;   **汇编期**由 NASM 的预处理器处理，它们自己不生成任何机器码。
;   生成机器码的是预处理完之后剩下的那些指令。
;   所以：
;       %rep 3        不是循环，是「把这段源码抄 3 遍」；
;       %ifdef DEBUG  不是运行时判断，是「条件不满足就根本不汇编这段」。
; ===========================================================================

default rel                             ; 数据寻址默认 RIP 相对，所有 [标签] 都自动变成 [rel 标签]
bits 64

%include "syscalls.inc"                 ; 提供 SYS_write / STDOUT / SYS_exit
%include "io.inc"                       ; 提供 io_print / io_print_uint / io_newline / io_exit
                                        ; （io.inc 里已经 %include 了 syscalls.inc，重复 include 无害）

; ---------------------------------------------------------------------------
; 一、%define：定义「单行宏」，也就是汇编期的文本替换
;     它没有类型、不占存储空间，展开发生在源码文本层面。
; ---------------------------------------------------------------------------
%define VERSION_MAJOR   12
%define VERSION_MINOR   0
%define VERSION_STR     "12.0"
%define BUF_SIZE        64              ; 试试改成 8 或 128，看输出怎么变
%define NEWLINE         10

%define DEBUG                           ; 保留这行 -> 走 Debug 分支
                                        ; 注释掉它   -> 那段代码根本不出现在可执行文件里

; 一个小检查：条件不满足时直接让汇编失败，比运行时出错更省事
%if BUF_SIZE < 8
    %error "BUF_SIZE 至少要 8 个字节"
%endif

; ---------------------------------------------------------------------------
; 二、%macro/%endmacro：带参数的宏
;     宏体里用 %1、%2… 引用参数，%0 是参数个数。
;     %1 是纯文本替换：
;       * 想当字符串用（塞进 db "..."）   -> 用 %defstr 再引用
;       * 想当数字算                        -> 用 %assign / %if 里直接写
;     宏没有 call 开销（因为压根没有 call），代价是每展开一次就复制一份代码。
; ---------------------------------------------------------------------------

; ---- 写文件描述符 %1，缓冲区 %2，长度 %3 ----
; Linux 64 位：系统调用号在 rax，参数依次 rdi/rsi/rdx/r10/r8/r9
%macro SYS_WRITE3 3
    mov  rax, SYS_write
    mov  rdi, %1
    mov  rsi, %2
    mov  rdx, %3
    syscall                             ; syscall 会破坏 rcx 和 r11
%endmacro

; ---- 往 stdout 写一段字符串并换行 ----
; %1 是 "带引号的字符串字面量"。
; 标签用 %% 前缀（%%name）：每次宏展开都会生成唯一的名字，不会重复定义。
; 注意：这里不能用 %$name —— %$ 属于「宏上下文（%push 建立的 context）」机制，
; 没有 %push 直接用会报 `%$tag': context stack is empty，所以教学示例用 %% 更合适。
%macro PRINT_LINE 1
    jmp  short %%past                   ; 先跳过内嵌的数据，别让 CPU 执行到它
%%tag:  db %1
%%len_tag equ $ - %%tag
%%nl_tag: db NEWLINE
%%past:
    mov  rax, SYS_write
    mov  rdi, STDOUT
    lea  rsi, [rel %%tag]
    mov  rdx, %%len_tag
    syscall
    mov  rax, SYS_write
    mov  rdi, STDOUT
    lea  rsi, [rel %%nl_tag]
    mov  rdx, 1
    syscall
%endmacro

; ---- 用 %strlen 在汇编期拿到字符串长度 ----
; 注意：不要写成 %defstr %%s %1 —— 参数里只要出现非 ASCII 字符（中文），
; NASM 在 %defstr 里会把多字节当成字符常量处理而报
; "character constant too long"；直接对带引号的参数用 %strlen 就没这个问题。
%macro SHOW_STR_INFO 1
    %strlen %%n %1                      ; 汇编期算出「字符串的字节数」
    jmp  short %%body                   ; ⚠ 必须先跳过下面的字符串数据
%%s: db %1                              ; 数据就摆在代码段里，不跳过会被当指令执行
%%body:
    mov  rax, SYS_write
    mov  rdi, STDOUT
    lea  rsi, [rel %%s]
    mov  rdx, %%n
    syscall

/* ===========================================================================
 * c_driver.c —— C 主程序，调用 lc_asm_lib.asm 里用汇编实现的函数
 *
 * 编译（和汇编目标文件一起，必须 -no-pie）：
 *   nasm -f elf64 -g -F dwarf -o lc_asm_lib.o lc_asm_lib.asm
 *   gcc -no-pie -o mix c_driver.c lc_asm_lib.o
 *   ./mix
 *
 * 为什么必须 -no-pie：
 *   -no-pie 生成“非位置无关”的可执行文件，汇编里直接写 lea rax,[rdi+rsi]
 *   这类绝对/相对寻址都按传统的固定基址链接；而默认的 PIE 会把程序放到
 *   随机地址，汇编里手写的绝对地址（本课虽然没有）就会失效。
 *   教程统一用 -no-pie，和 nasm + ld 的写法保持一致。
 * ===========================================================================
 */

#include <stdio.h>

/* 汇编里用 global 导出的符号，在 C 里用 extern 声明。
 * 类型必须和汇编里的实际实现一致：
 *   asm_add       整数参数 rdi,rsi      → rax
 *   asm_sum_array 指针 rdi + 长度 rsi   → rax
 *   asm_hypot     浮点参数 xmm0,xmm1    → xmm0
 *   asm_strlen    指针 rdi              → rax
 *   asm_dot       两个指针 + 长度       → xmm0
 */
extern long   asm_add(long a, long b);
extern long   asm_sum_array(const long *arr, long n);
extern double asm_hypot(double x, double y);
extern long   asm_strlen(const char *s);
extern double asm_dot(const double *a, const double *b, long n);

int main(void)
{
    long   nums[8] = {1, 2, 3, 4, 5, 6, 7, 8};
    double a[3] = {1.0, 2.0, 3.0};
    double b[3] = {4.0, 5.0, 6.0};
    const char *s = "hello";

    printf("asm_add(2,3)        = %ld\n", asm_add(2, 3));
    printf("asm_sum_array(nums) = %ld\n", asm_sum_array(nums, 8));
    printf("asm_hypot(3,4)      = %f\n", asm_hypot(3.0, 4.0));
    printf("asm_strlen(%s)  = %ld\n", s, asm_strlen(s));
    printf("asm_dot(a,b)        = %f\n", asm_dot(a, b, 3));

    return 0;
}

; ===========================================================================
; lc_asm_lib.asm —— 用汇编写一个能被 C 调用的“小库”（System V AMD64 调用约定）
;
; 文件名以 lc_ 开头 → 这个文件是“libc 程序”风格，入口符号叫 main，
; 用 gcc -no-pie 链接（不是 ld），因为链接脚本要 crt1.o 帮忙初始化 libc。
;
; 调用约定速查（详见本课 README）：
;   整数参数 rdi, rsi, rdx, rcx, r8, r9   返回值 rax
;   浮点参数 xmm0 ~ xmm7                  返回值 xmm0
;   调用者保存（可以随便改）：rax rcx rdx rsi rdi r8 r9 r10 r11 + 全部 xmm
;   被调用者保存（改了必须还原）：rbx rbp r12 r13 r14 r15
;   本文件所有函数只用调用者保存寄存器，所以一条 push/pop 都不需要。
;
; 三个使用场景（README 里都有完整命令）：
;   1) 单独链接成可执行文件（make all 就是这么做的）：
;        nasm -f elf64 -g -F dwarf -I ../include -o lc_asm_lib.o lc_asm_lib.asm
;        gcc -no-pie -o lc_asm_lib lc_asm_lib.o
;      能成功是因为下面有一个“弱符号 main”（global main:weak）做自测。
;   2) 和 C 混编（本课主角）：
;        nasm -f elf64 -g -F dwarf -o lc_asm_lib.o lc_asm_lib.asm
;        gcc -no-pie -o mix c_driver.c lc_asm_lib.o
;      c_driver.c 里的强符号 main 会覆盖这里的弱符号 main，
;      所以既不会 duplicate symbol，也不会执行到自测那段代码。
;   3) 如果你的工具链不喜欢弱符号，可以用 -DNO_MAIN 编译出纯函数库：
;        nasm -f elf64 -g -F dwarf -DNO_MAIN -o lc_asm_lib.o lc_asm_lib.asm
;      这时 lc_asm_lib.o 里只有 asm_* 子程序，没有 main（不能单独用 gcc 链接）。
; ===========================================================================

default rel

; ---- 外部符号：printf 由 libc 提供，用 nasm 时必须显式 extern ---------------
extern printf

; ---- 导出的符号（:function 是 ELF 符号类型标注，写不写都能跑，写上更规范）--
global asm_add:function
global asm_sum_array:function
global asm_hypot:function
global asm_strlen:function
global asm_dot:function

%ifndef NO_MAIN
global main:weak            ; 弱符号：单独链接时当入口；遇到 C 的强 main 时自动让位
%endif

section .text

; ---------------------------------------------------------------------------
; long asm_add(long a, long b)
;   a → rdi, b → rsi, 返回 rax
; ---------------------------------------------------------------------------
asm_add:
    lea rax, [rdi + rsi]    ; 一条指令算完：rax = rdi + rsi，不碰内存、不碰标志位
    ret

; ---------------------------------------------------------------------------
; long asm_sum_array(const long *arr, long n)
;   arr → rdi, n → rsi, 返回 rax
;   约定：n <= 0 时返回 0
; ---------------------------------------------------------------------------
asm_sum_array:
    xor eax, eax            ; 累加器 = 0（写 eax 自动清零 rax 的高 32 位）
    test rsi, rsi
    jle .done               ; n <= 0：直接返回 0
    xor ecx, ecx            ; 下标 = 0
.loop:
    add rax, [rdi + rcx*8]  ; 取 arr[i]（8 字节）加到 rax
    inc rcx
    cmp rcx, rsi
    jb .loop                ; i < n 就继续（无符号比较，n 已确认 > 0）
.done:
    ret

; ---------------------------------------------------------------------------
; double asm_hypot(double x, double y)
;   x → xmm0, y → xmm1，返回 xmm0 = sqrt(x*x + y*y)
;   浮点参数走 xmm 寄存器，这和整数参数走 rdi/rsi 一样是约定的一部分。
; ---------------------------------------------------------------------------
asm_hypot:
    mulsd xmm0, xmm0        ; xmm0 = x*x
    mulsd xmm1, xmm1        ; xmm1 = y*y
    addsd xmm0, xmm1        ; xmm0 = x*x + y*y
    sqrtsd xmm0, xmm0       ; xmm0 = sqrt(xmm0)；sqrtsd 是“目标 = 源开方”
    ret

; ---------------------------------------------------------------------------
; long asm_strlen(const char *s)
;   s → rdi，返回 rax（不含结尾的 '\0'）
; ---------------------------------------------------------------------------
asm_strlen:
    xor eax, eax
.loop:
    cmp byte [rdi + rax], 0 ; 一次比较一个字节
    je .done
    inc rax
    jmp .loop
.done:
    ret

; ---------------------------------------------------------------------------
; double asm_dot(const double *a, const double *b, long n)
;   a → rdi, b → rsi, n → rdx，返回 xmm0 = Σ a[i]*b[i]（标量循环版）
;   C 原型：double asm_dot(const double *, const double *, long);
; ---------------------------------------------------------------------------
asm_dot:
    pxor xmm0, xmm0         ; 累加器 = 0.0（pxor 比 movsd xmm0, 0 更省事）
    test rdx, rdx
    jle .done
    xor ecx, ecx
.loop:
    movsd xmm1, [rdi + rcx*8]   ; 取 a[i]（8 字节 double）
    mulsd xmm1, [rsi + rcx*8]   ; xmm1 *= b[i]（内存操作数可以直接参与运算）
    addsd xmm0, xmm1
    inc rcx
    cmp rcx, rdx
    jb .loop
.done:
    ret

; ===========================================================================
; 下面是“单独链接”用的自测 main（弱符号）
;   与 c_driver.c 一起链接时不会用到它，只是让 make all 能通过。
;   用 -DNO_MAIN 编译时这一整段不参与汇编。
; ===========================================================================
%ifndef NO_MAIN

section .data
nums:       dq 1, 2, 3, 4, 5, 6, 7, 8
vec_a:      dq 1.0, 2.0, 3.0
vec_b:      dq 4.0, 5.0, 6.0
hello:      db "hello", 0
three:      dq 3.0
four:       dq 4.0

fmt_add:    db "asm_add(2,3)        = %ld", 10, 0
fmt_sum:    db "asm_sum_array(nums) = %ld", 10, 0
fmt_hyp:    db "asm_hypot(3,4)      = %f", 10, 0
fmt_len:    db "asm_strlen(%s)  = %ld", 10, 0
fmt_dot:    db "asm_dot(a,b)        = %f", 10, 0
fmt_head:   db "=== lc_asm_lib 自测（弱符号 main）===", 10, 0

section .text

main:
    push rbp
    mov rbp, rsp
    ; 入口处 rsp % 16 == 8（call 压了返回地址），push rbp 之后 == 0，
    ; 所以下面每次 call printf 时栈都是 16 字节对齐的。

    lea rdi, [rel fmt_head]
    xor eax, eax
    call printf

    mov rdi, 2
    mov rsi, 3
    call asm_add
    lea rdi, [rel fmt_add]
    mov rsi, rax
    xor eax, eax                ; 变参函数：al = 用到的向量寄存器个数 = 0
    call printf

    lea rdi, [rel nums]
    mov rsi, 8
    call asm_sum_array
    lea rdi, [rel fmt_sum]
    mov rsi, rax
    xor eax, eax
    call printf

    movsd xmm0, [rel three]
    movsd xmm1, [rel four]
    call asm_hypot
    lea rdi, [rel fmt_hyp]
    mov eax, 1                  ; 用到了 xmm0 → al = 1
    call printf

    lea rdi, [rel hello]
    call asm_strlen
    lea rdi, [rel fmt_len]
    lea rsi, [rel hello]
    mov rdx, rax
    xor eax, eax
    call printf

    lea rdi, [rel vec_a]
    lea rsi, [rel vec_b]
    mov rdx, 3
    call asm_dot
    lea rdi, [rel fmt_dot]
    mov eax, 1
    call printf

    xor eax, eax                ; return 0
    leave                       ; mov rsp, rbp; pop rbp
    ret

%endif ; NO_MAIN

; ===========================================================================
; lc_opt.asm —— 同一个任务两种写法：朴素 vs 优化（SSE2），并说明为什么快
;
; 文件名以 lc_ 开头 → 入口 main，用 gcc -no-pie 链接。
;
; 任务一：求 1024 个 int64 元素的和
;   sum_naive : 每轮加 1 个元素 —— 1 次内存读 + inc + cmp + 条件跳转
;   sum_simd  : 每轮加 2 个元素 —— movdqu + paddq，指令数减半、分支减半
; 任务二：统计 4096 字节的缓冲区里有多少个 'a'
;   count_a_naive : 逐字节比较 + 每字节一次条件跳转（分支预测器压力大）
;   count_a_simd  : 16 字节一次比较（pcmpeqb），把 0xFF 掩成 0x01 后 psadbw 求和
;
; 默认构建只打印“确定的结果”，方便 test.sh 比对。
; 想看真实耗时（每次都不一样，只供观察）：
;   nasm -f elf64 -g -F dwarf -DMEASURE -o lc_opt_m.o lc_opt.asm
;   gcc -no-pie -o lc_opt_m lc_opt_m.o && ./lc_opt_m
; ===========================================================================

default rel

%define SYS_clock_gettime 228
%define CLOCK_MONOTONIC   1

%define ARR_N 1024              ; 数组元素个数
%define STR_N 4096              ; 字符串字节数

extern printf               ; libc 提供的函数：用 nasm 时必须显式 extern

global main

section .data
; ---- SSE2 用的常量（movdqa 要求 16 字节对齐）------------------------------
align 16
xmm_a:      times 16 db 'a'      ; 16 个 'a'，pcmpeqb 的掩码
xmm_one:    times 16 db 1        ; 16 个 0x01，把 0xFF 掩成 0x01

; ---- arr[i] = i % 100 + 1，共 1024 个 qword ---------------------------------
; 这样 1..100 的完整循环有 10 轮（10 * 5050 = 50500），再加 1..24（= 300），
; 所以正确的和是 50800，手算就能验证。
align 8
arr:
%assign i 0
%rep ARR_N
    dq (i % 100) + 1
    %assign i i + 1
%endrep

; ---- strbuf：每 26 个字节一个 'a' ------------------------------------------
; 26 * 157 = 4082，再补 1 个 'a' 和 13 个 'b' 正好 4096 字节，
; 所以正确的 'a' 个数是 157 + 1 = 158。
strbuf:
%rep (STR_N / 26)
    db 'a'
    times 25 db 'b'
%endrep
    db 'a'
    times (STR_N - (STR_N / 26) * 26 - 1) db 'b'

; ---- 输出格式 --------------------------------------------------------------
fmt_ns:     db "naive  sum        = %ld", 10, 0
fmt_ss:     db "simd   sum        = %ld", 10, 0
fmt_nc:     db "naive  count 'a'  = %ld", 10, 0
fmt_sc:     db "simd   count 'a'  = %ld", 10, 0
fmt_ck:     db "check  : results match (%ld)", 10, 0
%ifdef MEASURE
fmt_t1:     db "measure: naive sum   = %ld ns", 10, 0
fmt_t2:     db "measure: simd  sum   = %ld ns", 10, 0
fmt_t3:     db "measure: naive count = %ld ns", 10, 0
fmt_t4:     db "measure: simd  count = %ld ns", 10, 0
%define REPS_SUM 20000          ; 求和跑 2 万遍，让耗时能被 clock_gettime 量到
%define REPS_CNT 5000           ; 统计跑 5 千遍
%endif

section .bss
%ifdef MEASURE
ts:         resq 2              ; clock_gettime 的 struct timespec {tv_sec, tv_nsec}
%endif

section .text
global sum_naive
global sum_simd
global count_a_naive
global count_a_simd

; ---------------------------------------------------------------------------
; long sum_naive(const long *arr, long n)      arr→rdi, n→rsi, 返回 rax
; ---------------------------------------------------------------------------
sum_naive:
    xor eax, eax
    test rsi, rsi
    jle .done
    xor ecx, ecx
.loop:
    add rax, [rdi + rcx*8]      ; 每轮只处理 1 个元素
    inc rcx
    cmp rcx, rsi
    jb .loop
.done:
    ret

; ---------------------------------------------------------------------------
; long sum_simd(const long *arr, long n)
;   一次处理 2 个 qword（一个 xmm 寄存器 128 位），再水平合并。
; ---------------------------------------------------------------------------
sum_simd:
    pxor xmm0, xmm0             ; 向量累加器 = 0
    xor eax, eax                ; 标量结果默认 0（没有整块时用）
    mov rcx, rsi
    shr rcx, 1                  ; 成对的个数 = n / 2
    test rcx, rcx
    jz .tail
    mov rdx, rdi
.blk:
    movdqu xmm1, [rdx]          ; 一次读 16 字节 = 2 个 qword
    paddq xmm0, xmm1            ; 64 位整数逐元素相加（不会进位串到隔壁）
    add rdx, 16
    dec rcx
    jnz .blk
    movhlps xmm1, xmm0          ; 把高 64 位搬到 xmm1 的低 64 位
    paddq xmm0, xmm1            ; 两个 64 位计数器合并
    movq rax, xmm0              ; 取低 64 位作为结果
.tail:
    test rsi, 1                 ; n 是奇数？
    jz .done
    mov rdx, rsi
    dec rdx
    add rax, [rdi + rdx*8]      ; 处理最后一个元素
.done:
    ret

; ---------------------------------------------------------------------------
; long count_a_naive(const char *s, long n)
; ---------------------------------------------------------------------------
count_a_naive:
    xor eax, eax
    test rsi, rsi
    jle .done
    xor ecx, ecx
.loop:
    cmp byte [rdi + rcx], 'a'   ; 每字节一次比较
    jne .next                   ; 每字节一次条件跳转（不可预测，很贵）
    inc rax
.next:
    inc rcx
    cmp rcx, rsi
    jb .loop
.done:
    ret

; ---------------------------------------------------------------------------
; long count_a_simd(const char *s, long n)
;   每轮 16 字节：pcmpeqb 得到 0xFF/0x00，pand 成 0x01/0x00，
;   psadbw 把 16 个字节求和，paddq 累加到 64 位计数器。
; ---------------------------------------------------------------------------
count_a_simd:
    xor eax, eax
    pxor xmm0, xmm0             ; 64 位计数器（只用到低 64 位）
    pxor xmm3, xmm3             ; 全 0，psadbw 的第二个操作数
    movdqa xmm2, [rel xmm_a]    ; 16 个 'a'
    movdqa xmm4, [rel xmm_one]  ; 16 个 0x01
    mov rcx, rsi
    shr rcx, 4                  ; 完整 16 字节块的个数 = n / 16
    test rcx, rcx
    jz .tail
    mov rdx, rdi
.blk:
    movdqu xmm1, [rdx]          ; 16 字节，不要求对齐（字符串未必对齐）
    pcmpeqb xmm1, xmm2          ; 等于 'a' 的字节 → 0xFF，其余 → 0x00
    pand xmm1, xmm4             ; 0xFF → 0x01，0x00 → 0x00
    psadbw xmm1, xmm3           ; 16 个字节的差之和 = 这一块里 'a' 的个数
    paddq xmm0, xmm1
    add rdx, 16
    dec rcx
    jnz .blk
    ; psadbw 把两个 8 字节半区的和分别放在低 64 位和高 64 位，要合并起来
    movq rax, xmm0
    psrldq xmm0, 8
    movq rcx, xmm0
    add rax, rcx
.tail:
    mov rcx, rsi
    and rcx, 15                 ; 尾部不足 16 字节的部分
    jz .done
    mov rdx, rsi
    sub rdx, rcx                ; 尾部起始下标
.tloop:
    cmp byte [rdi + rdx], 'a'
    jne .tnext
    inc rax
.tnext:
    inc rdx
    dec rcx
    jnz .tloop
.done:
    ret

%ifdef MEASURE
; ---------------------------------------------------------------------------
; now_ns() → rax：读 CLOCK_MONOTONIC，换算成纳秒
; （直接发 clock_gettime 系统调用，参数 rdi=时钟号, rsi=缓冲区）
; ---------------------------------------------------------------------------
now_ns:
    mov rax, SYS_clock_gettime
    mov edi, CLOCK_MONOTONIC
    lea rsi, [rel ts]
    syscall
    mov rax, [rel ts + 8]       ; tv_nsec
    mov rdx, [rel ts]           ; tv_sec
    imul rdx, rdx, 1000000000
    add rax, rdx
    ret
%endif

; ===========================================================================
; int main(void)
; ===========================================================================
main:
    push rbp
    mov rbp, rsp
    push r12                    ; 入口 rsp%16==8，push rbp 后 ==0，四个 push 后仍 ==0
    push r13
    push r14
    push r15

    ; ---- 两种求和实现 ------------------------------------------------------
    lea rdi, [rel arr]
    mov rsi, ARR_N
    call sum_naive
    mov r12, rax

    lea rdi, [rel arr]
    mov rsi, ARR_N
    call sum_simd
    mov r13, rax

    ; ---- 两种字符统计实现 --------------------------------------------------
    lea rdi, [rel strbuf]
    mov rsi, STR_N
    call count_a_naive
    mov r14, rax

    lea rdi, [rel strbuf]
    mov rsi, STR_N
    call count_a_simd
    mov r15, rax

    lea rdi, [rel fmt_ns]
    mov rsi, r12
    xor eax, eax
    call printf

    lea rdi, [rel fmt_ss]
    mov rsi, r13
    xor eax, eax
    call printf

    lea rdi, [rel fmt_nc]
    mov rsi, r14
    xor eax, eax
    call printf

    lea rdi, [rel fmt_sc]
    mov rsi, r15
    xor eax, eax
    call printf

    ; ---- 一致性检查：两种实现结果必须相同 ---------------------------------
    xor eax, eax
    cmp r12, r13
    jne .no
    cmp r14, r15
    jne .no
    mov eax, 1
.no:
    lea rdi, [rel fmt_ck]
    mov rsi, rax
    xor eax, eax
    call printf

%ifdef MEASURE
    ; ---- 计时：朴素求和 ----------------------------------------------------
    call now_ns
    mov r12, rax
    mov r13, REPS_SUM
.m_naive_sum:
    lea rdi, [rel arr]
    mov rsi, ARR_N
    call sum_naive
    dec r13
    jnz .m_naive_sum
    call now_ns
    sub rax, r12

    call printf

    mov r13, REPS_SUM

    mov rsi, ARR_N
    call sum_simd
    dec r13
    jnz .m_simd_sum
    call now_ns
    sub rax, r12
    lea rdi, [rel fmt_t2]
    mov rsi, rax
    xor eax, eax
    call printf

    ; ---- 计时：朴素统计 ----------------------------------------------------
    call now_ns
    mov r12, rax
    mov r13, REPS_CNT
.m_naive_cnt:

    mov rsi, STR_N
    call count_a_naive
    dec r13
    jnz .m_naive_cnt
    call now_ns
    sub rax, r12
    lea rdi, [rel fmt_t3]
    mov rsi, rax
    xor eax, eax
    call printf

    ; ---- 计时：SIMD 统计 ---------------------------------------------------
    call now_ns
    mov r12, rax

.m_simd_cnt:

    call now_ns
    sub rax, r12

    mov rsi, rax
    xor eax, eax
    call printf
%endif

    pop r15
    pop r14
    pop r13
    pop r12
    xor eax, eax
    leave

; ===========================================================================
; debug_target.asm —— 第 14 课的“调试靶子”：一个平均值计算程序
;
; 纯系统调用风格：文件名不以 lc_ 开头、入口 _start、用 ld 直接链接（不链 libc）。
; 输入输出用 ../include/io.inc 里的小工具（io_print / io_print_int / io_newline）。
;
; ★ 默认构建（不加 -DBUGGY_VERSION）是「已经修好」的版本，能正常跑完：
;     make && ./debug_target
;
; ★ 故意写错的版本用条件汇编包着，需要用 -DBUGGY_VERSION 显式打开：
;     nasm -f elf64 -g -F dwarf -DBUGGY_VERSION -I ../include -o debug_target.o debug_target.asm
;     ld -o debug_target_buggy debug_target.o
;     ./debug_target_buggy
;   错版本不会崩溃、也不会死循环，但会打印出明显不对的 count/sum/avg，
;   README 里用 gdb + objdump 一步步演示怎么定位这两个 bug。
;
; 两个 bug（都在下面用 BUG 1 / BUG 2 标出）：
;   BUG 1  循环条件用 jbe（<=）而不是 jb（<）→ 多循环一次，读了数组后面一个元素
;   BUG 2  累加器用 32 位 eax、读内存也只读 dword → 高位丢掉（本例正好丢成 0）
; ===========================================================================

default rel

section .text
%include "io.inc"               ; 只能 include 一次；它自己会打开 section .text

section .data
COUNT   equ 8                   ; 数组元素个数

; 8 个元素，每个 0x20000000 = 536870912：
;   正确的 64 位和 = 8 * 0x20000000 = 0x100000000 = 4294967296
;   用 32 位累加就会截断成 0
arr:
    dq 0x20000000, 0x20000000, 0x20000000, 0x20000000
    dq 0x20000000, 0x20000000, 0x20000000, 0x20000000

; 紧跟在数组后面的一块数据（哨兵）。它在 .data 里，所以就算被越界读也不会段错误，
; 但会让结果对不上 —— 这正是我们希望 gdb 暴露出来的现象。
sentinel:
    dq 0x40000000

msg_count:  db "count = "
msg_count_len equ $ - msg_count
msg_sum:    db "sum   = "
msg_sum_len equ $ - msg_sum
msg_avg:    db "avg   = "
msg_avg_len equ $ - msg_avg

section .text
global _start

_start:
    lea rbx, [rel arr]          ; rbx = 数组基址（rbx 是被调用者保存，跨 io_* 调用安全）

%ifdef BUGGY_VERSION
    ; ------------------------- 有 bug 的版本 -----------------------------
    xor eax, eax                ; BUG 2：累加器只用 32 位
    xor ecx, ecx
.sum_loop:
    add eax, dword [rbx + rcx*8]    ; BUG 2：只读低 4 字节，高 4 字节丢掉
    inc rcx
    cmp rcx, COUNT
    jbe .sum_loop               ; BUG 1：应该是 jb（rcx < COUNT）
%else
    ; ------------------------- 正确版本 -----------------------------------
    xor rax, rax                ; 64 位累加器
    xor ecx, ecx
.sum_loop:
    add rax, [rbx + rcx*8]      ; 一次读满 8 字节
    inc rcx
    cmp rcx, COUNT
    jb .sum_loop                ; rcx < COUNT 才继续 → 正好 8 次
%endif

    mov r13, rcx                ; 循环次数 = 元素个数（正确 8，错误 9）
    mov r12, rax                ; 求和结果

    ; ---- 打印 count --------------------------------------------------------
    lea rsi, [rel msg_count]
    mov rdx, msg_count_len
    call io_print
    mov rdi, r13
    call io_print_int
    call io_newline

    ; ---- 打印 sum ----------------------------------------------------------
    lea rsi, [rel msg_sum]
    mov rdx, msg_sum_len
    call io_print
    mov rdi, r12
    call io_print_int
    call io_newline

    ; ---- avg = sum / count -------------------------------------------------
    mov rax, r12
    xor rdx, rdx                ; div 之前必须清零 rdx（被除数的高 64 位）
    div r13                     ; rax = 商
    mov r14, rax

    lea rsi, [rel msg_avg]
    mov rdx, msg_avg_len
    call io_print
    mov rdi, r14
    call io_print_int
    call io_newline

    xor edi, edi
    call io_exit                ; 结束进程（纯 syscall 程序不能 ret 到不存在的调用者）

; ===========================================================================
; hexdump.asm —— 模仿 `hexdump -C`（BSD / C 风格）把标准输入打印成十六进制
;
; 纯系统调用风格：入口 _start，用 ld 链接，不依赖 libc，也不用 printf。
; 输入输出用 ../include/io.inc 里的 io_write / io_exit。
;
; 输出格式（和 Linux 的 hexdump -C 一模一样）：
;   00000000  68 65 6c 6c 6f 0a                                 |hello.|
;   00000006
; 每行固定 16 字节，分两栏各 8 字节（中间多一个空格），
; 右边是 ASCII 视图：0x20~0x7e 原样显示，其它（含 0x80 以上）显示成 '.'
; 数据读完后还会单独打印一行“总偏移”。
; 输入为空（例如 make run 时的 /dev/null）时不输出任何东西，正常退出。
;
; 运行：
;   ./hexdump < 文件
;   printf 'hello\n' | ./hexdump
; ===========================================================================

default rel

section .text
%include "io.inc"               ; 只 include 一次

section .data
hexdig: db "0123456789abcdef"   ; 十六进制字符表：下标 0~15 → '0'~'f'

section .bss
inbuf:  resb 16                 ; 当前这一行的 16 字节数据
have:   resq 1                  ; 当前这一行实际读到了几个字节
total:  resq 1                  ; 已经处理过的总字节数（= 下一行的偏移）
line:   resb 128                ; 输出行缓冲区（最长 79 字节，留足余量）

section .text
global _start

; ---------------------------------------------------------------------------
; _start：主循环 —— 每次攒够 16 字节（或遇到 EOF）就交给 emit_line 打一行
; ---------------------------------------------------------------------------
_start:
    mov qword [rel total], 0

.next_line:
    mov qword [rel have], 0

    ; ---- 把这一行的数据读满（read 可能短读，所以要循环凑）----------------
.fill:
    mov rax, SYS_read
    xor edi, edi                ; fd = 0（标准输入）
    lea rsi, [rel inbuf]
    add rsi, [rel have]         ; 接着上次的位置写
    mov rdx, 16
    sub rdx, [rel have]         ; 还差几个字节
    syscall
    test rax, rax
    jle .filled                 ; 0 = EOF，负数 = 出错，都按结束处理
    add [rel have], rax
    cmp qword [rel have], 16
    jb .fill                    ; 没满 16 字节就继续读

.filled:
    cmp qword [rel have], 0
    je .finish                  ; 一个字节都没读到 → 输入结束
    call emit_line
    mov rax, [rel have]
    add [rel total], rax        ; 偏移前进
    cmp qword [rel have], 16
    jb .finish                  ; 这一行不满 16 字节说明已经读到文件末尾
    jmp .next_line

.finish:
    cmp qword [rel total], 0
    je .exit                    ; 输入为空：什么都不打印
    call emit_offset_line       ; 最后补一行“总偏移”，和 hexdump -C 一致
.exit:
    xor edi, edi
    call io_exit

; ---------------------------------------------------------------------------
; emit_line —— 用 [have]/[total]/inbuf 生成一行 hexdump -C 格式并写出
;   行布局（共 79 字节）：
;     偏移(8) 空格空格 十六进制区(8*3 + 1 + 8*3 = 49) 空格 '|' ASCII(≤16) '|' '\n'
; ---------------------------------------------------------------------------
emit_line:
    push rbx
    push r12
    push r13
    push r14
    push r15
    lea r14, [rel hexdig]       ; 字符表
    lea r15, [rel line]         ; 输出游标
    lea rbx, [rel inbuf]        ; 数据源
    mov r13, [rel have]         ; 本行有效字节数（后面循环用，省得反复读内存）

    ; ---- 1) 8 位十六进制偏移 ---------------------------------------------
    mov ecx, 28                 ; 从第 28 位（最高半字节）开始，每次右移 4 位
.off:
    mov rdx, [rel total]
    shr rdx, cl                 ; cl = 本次的移位量
    and edx, 0x0f
    mov al, [r14 + rdx]
    mov [r15], al
    inc r15
    sub ecx, 4
    jns .off                    ; 28,24,...,0 共 8 位

    mov byte [r15], ' '
    mov byte [r15 + 1], ' '
    add r15, 2

    ; ---- 2) 16 个字节的十六进制（不足的补空格）---------------------------
    xor r12d, r12d
.bytes:
    cmp r12, r13
    jae .pad
    movzx eax, byte [rbx + r12]
    mov edx, eax
    shr edx, 4                  ; 高半字节
    mov dl, [r14 + rdx]
    mov [r15], dl
    inc r15
    mov edx, eax
    and edx, 0x0f               ; 低半字节
    mov dl, [r14 + rdx]
    mov [r15], dl
    inc r15
    mov byte [r15], ' '
    inc r15
    jmp .after_byte
.pad:
    mov byte [r15], ' '
    mov byte [r15 + 1], ' '
    mov byte [r15 + 2], ' '
    add r15, 3
.after_byte:
    cmp r12, 7
    jne .next_byte
    mov byte [r15], ' '         ; 第 8 个字节后额外空一格（hexdump -C 的分栏）
    inc r15
.next_byte:
    inc r12
    cmp r12, 16
    jb .bytes

    ; ---- 3) ASCII 视图 ----------------------------------------------------
    mov byte [r15], ' '
    inc r15
    mov byte [r15], '|'
    inc r15
    xor r12d, r12d
.ascii:
    cmp r12, r13
    jae .ascii_done
    movzx eax, byte [rbx + r12]
    cmp al, 0x20                ; 小于空格 → '.'
    jb .dot
    cmp al, 0x7e                ; 大于 '~'（含 0x7f~0xff）→ '.'
    ja .dot
    mov [r15], al
    jmp .ascii_next
.dot:
    mov byte [r15], '.'
.ascii_next:
    inc r15
    inc r12
    jmp .ascii
.ascii_done:
    mov byte [r15], '|'
    inc r15
    mov byte [r15], LF
    inc r15

    ; ---- 4) 一次 write 输出整行 ------------------------------------------
    lea rsi, [rel line]
    mov rdx, r15
    sub rdx, rsi                ; 长度 = 游标 - 行首
    mov rdi, STDOUT
    call io_write

    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

; ---------------------------------------------------------------------------
; emit_offset_line —— 打印最后一行“总偏移”（8 位十六进制 + 换行）
; ---------------------------------------------------------------------------
emit_offset_line:
    push rbx
    push r12
    push r14
    lea r14, [rel hexdig]
    lea rbx, [rel line]
    mov ecx, 28
.off:
    mov rdx, [rel total]
    shr rdx, cl
    and edx, 0x0f
    mov al, [r14 + rdx]
    mov [rbx], al
    inc rbx
    sub ecx, 4
    jns .off
    mov byte [rbx], LF
    inc rbx

    lea rsi, [rel line]
    mov rdx, rbx
    sub rdx, rsi
    mov rdi, STDOUT
    call io_write

    pop r14
    pop r12
    pop rbx
    ret

; ===========================================================================
; hexdump2.asm —— hexdump 增强版：支持命令行参数指定文件名
;
; 纯系统调用风格：入口 _start，用 ld 链接，不依赖 libc。
; 输出格式和 `hexdump -C`（BSD / C 风格）完全一致（第 14 课的 hexdump 相同引擎）。
;
; 用法：
;   ./hexdump2                # 读标准输入
;   ./hexdump2 FILE...        # 依次读这些文件
;   ./hexdump2 - FILE         # "-" 表示标准输入
;   cat a b | ./hexdump2      # 等价于 ./hexdump2 a b（偏移连续）
;
; 与第 14 课 hexdump 相比多出来的东西：
;   1. 从 [rsp] / [rsp+8] 取 argc / argv，用 openat(2) 打开文件
;   2. 多个文件当成一条连续字节流（偏移不重置），最后只打印一行总偏移
;   3. 打开失败往 stderr 报告并继续处理后面的文件，最终退出码为 1
;
; 命令行入口栈布局（内核给的）：
;   [rsp]      = argc
;   [rsp + 8]  = argv[0]（程序名）
;   [rsp + 16] = argv[1]（第一个参数）
;   ...        = NULL 结尾
; 注意：取参数之前不要 push 任何东西，否则 rsp 就指偏了。
; ===========================================================================

default rel

%define AT_FDCWD -100           ; openat(2) 的“相对当前工作目录”特殊值

section .text
%include "io.inc"               ; 一次；提供 io_write / io_exit / 系统调用号常量

section .data
hexdig:     db "0123456789abcdef"
msg_prefix: db "hexdump2: "
msg_prefix_len equ $ - msg_prefix
msg_err:    db ": cannot open", 10
msg_err_len equ $ - msg_err

section .bss
inbuf:  resb 16                 ; 当前行的 16 字节
have:   resq 1                  ; 当前行已有字节数
total:  resq 1                  ; 累计偏移（所有文件连续计数）
err_flag: resq 1                ; 有文件打不开就置 1，作为退出码
line:   resb 128                ; 输出行缓冲（最长 79 字节）

section .text
global _start

; ---------------------------------------------------------------------------
; _start：解析参数 → 逐个文件 dump → 最后补一行总偏移
; ---------------------------------------------------------------------------
_start:
    mov rbx, [rsp]              ; argc（rbx 会被 dump/emit_* 保存）
    lea r12, [rsp + 8]          ; &argv[0]
    mov qword [rel total], 0
    mov qword [rel have], 0     ; 半行缓冲跨文件保留（多文件 = 一条连续字节流）
    mov qword [rel err_flag], 0

    cmp rbx, 2
    jae .files                  ; 至少有一个参数 → 按文件处理

    ; ---- 没有参数：读标准输入 --------------------------------------------
    mov rdi, STDIN
    call dump
    jmp .finish

.files:
    mov r13, 1                  ; 从 argv[1] 开始
.next_arg:
    cmp r13, rbx
    jae .finish
    mov rsi, [r12 + r13*8]      ; argv[i]
    cmp byte [rsi], '-'         ; 参数正好是 "-"？
    jne .open
    cmp byte [rsi + 1], 0
    jne .open
    mov rdi, STDIN
    call dump
    jmp .next_file

.open:
    mov rax, SYS_openat
    mov rdi, AT_FDCWD
    mov rdx, O_RDONLY
    xor r10d, r10d              ; 只有 O_CREAT 才需要 mode
    syscall
    test rax, rax
    js .open_error
    mov r14, rax                ; fd（dump 会保存并恢复 r14）
    mov rdi, r14
    call dump
    mov rax, SYS_close
    mov rdi, r14
    syscall
    jmp .next_file

.open_error:
    mov qword [rel err_flag], 1
    mov rsi, [r12 + r13*8]
    call report_error

.next_file:
    inc r13
    jmp .next_arg

.finish:
    ; ---- 所有输入都读完了：先把没攒满的半个行打出来 -----------------------
    cmp qword [rel have], 0
    je .offset_line
    call emit_line
    mov rax, [rel have]
    add [rel total], rax
    mov qword [rel have], 0
.offset_line:
    cmp qword [rel total], 0
    je .exit
    call emit_offset_line       ; 全部数据读完，补一行总偏移
.exit:
    mov rdi, [rel err_flag]
    call io_exit

; ---------------------------------------------------------------------------
; dump —— 把 fd 的数据接到当前行缓冲后面（多文件 = 连续字节流）
;   入参：rdi = fd      结果：更新 [have]/[total]
;   攒满 16 字节就打一行；这个 fd 读完了就返回，剩下的半行留给下一个文件接着填。
; ---------------------------------------------------------------------------
dump:
    push r14
    mov r14, rdi                ; fd
.fill:
    mov rax, SYS_read
    mov rdi, r14
    lea rsi, [rel inbuf]
    add rsi, [rel have]         ; 接着写（可能是上一个文件剩下的半行）
    mov rdx, 16
    sub rdx, [rel have]         ; 还差多少
    syscall
    test rax, rax
    jle .ret                    ; EOF 或出错：这个 fd 结束
    add [rel have], rax
    cmp qword [rel have], 16
    jb .fill
    call emit_line              ; 攒满一行
    add qword [rel total], 16
    mov qword [rel have], 0
    jmp .fill
.ret:
    pop r14
    ret

; ---------------------------------------------------------------------------
; report_error —— rsi = 文件名；把 "hexdump2: <name>: cannot open" 写到 stderr
; ---------------------------------------------------------------------------
report_error:
    push rbx
    mov rbx, rsi
    lea rsi, [rel msg_prefix]
    mov rdx, msg_prefix_len
    mov rdi, STDERR
    call io_write
    mov rdi, rbx
    call str_len                ; argv 里的字符串只保证以 NUL 结尾，长度要自己数
    mov rdx, rax
    mov rsi, rbx
    mov rdi, STDERR
    call io_write
    lea rsi, [rel msg_err]
    mov rdx, msg_err_len
    mov rdi, STDERR
    call io_write
    pop rbx
    ret

; ---------------------------------------------------------------------------
; str_len —— rdi = 字符串 → rax = 长度（不含结尾的 0）
; ---------------------------------------------------------------------------
str_len:
    xor eax, eax
.loop:
    cmp byte [rdi + rax], 0
    je .done
    inc rax
    jmp .loop
.done:
    ret

; ---------------------------------------------------------------------------
; emit_line —— 用 [have]/[total]/inbuf 生成并输出一行
;   布局：偏移(8) 2 空格 十六进制区(49) 空格 '|' ASCII '|' '\n'
; ---------------------------------------------------------------------------
emit_line:
    push rbx
    push r12
    push r13
    push r14
    push r15
    lea r14, [rel hexdig]
    lea r15, [rel line]
    lea rbx, [rel inbuf]
    mov r13, [rel have]

    ; ---- 1) 8 位十六进制偏移 ---------------------------------------------
    mov ecx, 28
.off:
    mov rdx, [rel total]
    shr rdx, cl                 ; cl = 移位量
    and edx, 0x0f
    mov al, [r14 + rdx]
    mov [r15], al
    inc r15
    sub ecx, 4
    jns .off

    mov byte [r15], ' '
    mov byte [r15 + 1], ' '
    add r15, 2

    ; ---- 2) 16 个字节的十六进制，不足补空格 ------------------------------
    xor r12d, r12d
.bytes:
    cmp r12, r13
    jae .pad
    movzx eax, byte [rbx + r12]
    mov edx, eax
    shr edx, 4
    mov dl, [r14 + rdx]
    mov [r15], dl
    inc r15
    mov edx, eax
    and edx, 0x0f
    mov dl, [r14 + rdx]
    mov [r15], dl
    inc r15
    mov byte [r15], ' '
    inc r15
    jmp .after_byte
.pad:
    mov byte [r15], ' '
    mov byte [r15 + 1], ' '
    mov byte [r15 + 2], ' '
    add r15, 3
.after_byte:
    cmp r12, 7
    jne .next_byte
    mov byte [r15], ' '         ; 分栏：第 8 个字节后多一个空格
    inc r15
.next_byte:
    inc r12
    cmp r12, 16
    jb .bytes

    ; ---- 3) ASCII 视图 ----------------------------------------------------
    mov byte [r15], ' '
    inc r15
    mov byte [r15], '|'
    inc r15
    xor r12d, r12d
.ascii:
    cmp r12, r13
    jae .ascii_done
    movzx eax, byte [rbx + r12]
    cmp al, 0x20
    jb .dot
    cmp al, 0x7e
    ja .dot
    mov [r15], al
    jmp .ascii_next
.dot:
    mov byte [r15], '.'
.ascii_next:
    inc r15
    inc r12
    jmp .ascii
.ascii_done:
    mov byte [r15], '|'
    inc r15
    mov byte [r15], LF
    inc r15

    ; ---- 4) 一次 write 输出整行 ------------------------------------------
    lea rsi, [rel line]
    mov rdx, r15
    sub rdx, rsi
    mov rdi, STDOUT
    call io_write

    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

; ---------------------------------------------------------------------------
; emit_offset_line —— 打印 “总偏移” 行
; ---------------------------------------------------------------------------
emit_offset_line:
    push rbx
    push r14
    lea r14, [rel hexdig]
    lea rbx, [rel line]
    mov ecx, 28
.off:
    mov rdx, [rel total]
    shr rdx, cl
    and edx, 0x0f
    mov al, [r14 + rdx]
    mov [rbx], al
    inc rbx
    sub ecx, 4
    jns .off
    mov byte [rbx], LF
    inc rbx

    lea rsi, [rel line]
    mov rdx, rbx
    sub rdx, rsi
    mov rdi, STDOUT
    call io_write

    pop r14
    pop rbx
    ret

-------------------------------- 文件：lesson15/wordcount.asm --------------------------------

; ===========================================================================
; wordcount.asm —— 统计文件的行数 / 单词数 / 字节数（像个小型 wc）
;
; 纯系统调用风格：入口 _start，用 ld 链接，不依赖 libc。
; 输入输出只用 ../include/io.inc 里的 io_write / io_exit 和裸系统调用。
;
; 用法：
;   ./wordcount FILE...      # 一个或多个文件，多于一个时最后打印 total 行
;   ./wordcount              # 没有参数就读标准输入（不打文件名）
;   ./wordcount - FILE       # "-" 表示标准输入
;
; 输出格式与 GNU coreutils 的 `wc -l -w -c` 一致：
;   * 三列依次是 行数、单词数、字节数，用单个空格分隔
;   * 所有数字共用同一个宽度（右对齐）：普通文件按总字节数的位数算；
;     只要输入里有“非普通文件”（管道、/dev/null、"−"），最小列宽就是 7
;     —— 这是 wc 的怪癖，我们用 fstat(2) 判断文件类型，行为完全对齐
;   * 数字后面再空一格接文件名；读标准输入时没有文件名
;   * 只有一个输入时不打印 total 行
; 例：
;   $ printf 'hello world\nfoo bar baz\n' > t1.txt
;   $ ./wordcount t1.txt
;   2 5 24 t1.txt
;   $ ./wordcount t1.txt t2.txt
;      2    5   24 t1.txt
;      1    1    4 t2.txt
;      3    6   28 total
;
; 单词的定义和 wc 一样：一段连续的“非空白字符”（空白 = 空格 \t \n \v \f \r）。
; 关键点是“是否正在单词里”这个状态要跨 read 边界保留：buffer 正好断在单词中间时
; 不能把它算成两个词（下面用 r15 保存这个状态）。
; ===========================================================================

default rel

%define AT_FDCWD -100
%define MAXF     64             ; 最多同时统计 64 个文件
%define BUFSZ    65536          ; 每次读 64 KiB

section .text
%include "io.inc"               ; 一次；提供 io_write / io_exit / 常量

section .data
total_label:  db "total", 0
msg_prefix:   db "wordcount: "
msg_prefix_len equ $ - msg_prefix
msg_err:      db ": cannot open", 10
msg_err_len   equ $ - msg_err
msg_many:     db "wordcount: too many files (max 64)", 10
msg_many_len  equ $ - msg_many

section .bss
buf:        resb BUFSZ          ; 读取缓冲区
lines_arr:  resq MAXF           ; 每个文件的行数
words_arr:  resq MAXF           ; 每个文件的单词数
bytes_arr:  resq MAXF           ; 每个文件的字节数
names_arr:  resq MAXF           ; 每个文件的名字指针（0 = 标准输入，不打名字）
nfiles:     resq 1
width:      resq 1              ; 数字列宽（和 GNU wc 一样动态计算）
tlines:     resq 1
twords:     resq 1
tbytes:     resq 1
statbuf:    resb 144            ; fstat(2) 的 struct stat
nonreg:     resq 1              ; 有非普通文件输入就置 1（列宽最小 7）
cur_lines:  resq 1              ; 当前文件的统计结果
cur_words:  resq 1
cur_bytes:  resq 1
err_flag:   resq 1
linebuf:    resb 512            ; 输出行缓冲

section .text
global _start

; ---------------------------------------------------------------------------
; _start
; ---------------------------------------------------------------------------
_start:
    mov rbx, [rsp]              ; argc
    lea r12, [rsp + 8]          ; &argv[0]
    mov qword [rel nfiles], 0
    mov qword [rel err_flag], 0
    mov qword [rel nonreg], 0

    cmp rbx, 2
    jae .file_args

    ; ---- 没有参数：读标准输入 --------------------------------------------
    mov rdi, STDIN
    call count_fd
    xor r15d, r15d              ; 名字 = 0（表示不打印文件名）
    call store_result
    jmp .print_all

.file_args:
    mov r13, 1                  ; argv 下标
.next_arg:
    cmp r13, rbx
    jae .print_all
    cmp qword [rel nfiles], MAXF
    jb .arg_ok
    lea rsi, [rel msg_many]     ; 文件太多：报错并停止解析
    mov rdx, msg_many_len
    mov rdi, STDERR
    call io_write
    mov qword [rel err_flag], 1
    jmp .print_all

.arg_ok:
    mov rsi, [r12 + r13*8]      ; argv[i]
    cmp byte [rsi], '-'         ; "-" → 标准输入
    jne .open
    cmp byte [rsi + 1], 0
    jne .open
    mov rdi, STDIN
    call count_fd
    mov r15, [r12 + r13*8]      ; 名字就显示 "-"
    call store_result
    jmp .next_file

.open:
    mov rax, SYS_openat
    mov rdi, AT_FDCWD
    mov rdx, O_RDONLY
    xor r10d, r10d              ; mode：O_RDONLY 用不到
    syscall
    test rax, rax
    js .open_error
    mov r14, rax                ; fd
    mov rdi, r14
    call count_fd
    mov rax, SYS_close
    mov rdi, r14
    syscall
    mov r15, [r12 + r13*8]      ; 记住文件名
    call store_result
    jmp .next_file

.open_error:
    mov qword [rel err_flag], 1
    mov rsi, [r12 + r13*8]
    call report_error

.next_file:
    inc r13
    jmp .next_arg

; ---------------------------------------------------------------------------
; .print_all：先算总和与列宽，再逐行打印（和 GNU wc 一样的两步走）
; ---------------------------------------------------------------------------
.print_all:
    mov qword [rel tlines], 0
    mov qword [rel twords], 0
    mov qword [rel tbytes], 0
    xor ecx, ecx
.tot_loop:
    cmp rcx, [rel nfiles]
    jae .width_calc
    lea rax, [rel lines_arr]
    mov rdx, [rax + rcx*8]
    add [rel tlines], rdx
    lea rax, [rel words_arr]
    mov rdx, [rax + rcx*8]
    add [rel twords], rdx
    lea rax, [rel bytes_arr]
    mov rdx, [rax + rcx*8]
    add [rel tbytes], rdx
    inc rcx
    jmp .tot_loop

.width_calc:
    mov qword [rel width], 1
    xor r12d, r12d              ; 这里起 r12（原 argv）不再需要
.wloop:
    cmp r12, [rel nfiles]
    jae .w_totals
    lea rax, [rel lines_arr]
    mov rdi, [rax + r12*8]
    call num_width
    call widen
    lea rax, [rel words_arr]
    mov rdi, [rax + r12*8]
    call num_width
    call widen
    lea rax, [rel bytes_arr]
    mov rdi, [rax + r12*8]
    call num_width
    call widen
    inc r12
    jmp .wloop
.w_totals:
    mov rdi, [rel tlines]
    call num_width
    call widen
    mov rdi, [rel twords]
    call num_width
    call widen
    mov rdi, [rel tbytes]
    call num_width
    call widen
    ; ---- GNU wc 的怪癖：有非普通文件输入时列宽至少 7 ----------------------
    cmp qword [rel nonreg], 0
    je .print_lines
    cmp qword [rel width], 7
    jae .print_lines
    mov qword [rel width], 7

.print_lines:
    xor ebx, ebx                ; 下标（print_line 会保存/恢复 rbx）
.ploop:
    cmp rbx, [rel nfiles]
    jae .print_total
    lea rax, [rel lines_arr]
    mov r12, [rax + rbx*8]
    lea rax, [rel words_arr]
    mov r13, [rax + rbx*8]
    lea rax, [rel bytes_arr]
    mov r14, [rax + rbx*8]
    lea rax, [rel names_arr]
    mov r15, [rax + rbx*8]
    call print_line
    inc rbx
    jmp .ploop

.print_total:
    cmp qword [rel nfiles], 1
    jbe .exit                   ; 只有 0/1 个输入：没有 total 行
    mov r12, [rel tlines]
    mov r13, [rel twords]
    mov r14, [rel tbytes]
    lea r15, [rel total_label]
    call print_line

.exit:
    mov rdi, [rel err_flag]
    call io_exit

; ---------------------------------------------------------------------------
; count_fd —— 统计一个文件描述符
;   入参：rdi = fd
;   结果：[cur_lines] / [cur_words] / [cur_bytes]
; ---------------------------------------------------------------------------
count_fd:
    push rbx
    push r12
    push r13
    push r14
    push r15
    mov r12, rdi                ; fd
    mov qword [rel cur_lines], 0
    mov qword [rel cur_words], 0
    mov qword [rel cur_bytes], 0
    xor r15d, r15d              ; 0 = 当前不在单词里（跨 read 保留）

    ; ---- 用 fstat(2) 看是不是普通文件（决定列宽是否要用最小 7）-----------
    mov rax, SYS_fstat
    mov rdi, r12
    lea rsi, [rel statbuf]
    syscall
    test rax, rax
    js .stat_done               ; 失败就当成普通文件，不设标记
    mov eax, dword [rel statbuf + 24]   ; st_mode 在 struct stat 的偏移 24
    and eax, 0xF000             ; 取文件类型位
    cmp eax, 0x8000             ; S_IFREG = 普通文件
    je .stat_done
    mov qword [rel nonreg], 1
.stat_done:

.read:
    mov rax, SYS_read
    mov rdi, r12
    lea rsi, [rel buf]
    mov rdx, BUFSZ
    syscall
    test rax, rax
    jle .done                   ; 0 = EOF；负数 = 读错误，都结束
    add [rel cur_bytes], rax
    lea rbx, [rel buf]
    mov r13, rax                ; 本次读到的字节数
    xor r14d, r14d              ; 缓冲区下标
.scan:
    movzx eax, byte [rbx + r14]
    cmp al, LF
    jne .not_newline
    inc qword [rel cur_lines]
.not_newline:
    ; ---- 是不是空白字符？------------------------------------------------
    cmp al, SPACE
    je .is_space
    cmp al, TAB
    je .is_space
    cmp al, LF
    je .is_space
    cmp al, 0x0b                ; \v
    je .is_space
    cmp al, 0x0c                ; \f
    je .is_space
    cmp al, 0x0d                ; \r
    je .is_space
    ; ---- 非空白：如果刚才不在单词里，就是一个新单词的开头 ----------------
    test r15, r15
    jnz .next
    inc qword [rel cur_words]
    mov r15d, 1
    jmp .next
.is_space:
    xor r15d, r15d
.next:
    inc r14
    cmp r14, r13
    jb .scan
    jmp .read
.done:
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

; ---------------------------------------------------------------------------
; store_result —— 把 cur_* 和 r15（文件名指针）追加到数组，nfiles++
; ---------------------------------------------------------------------------
store_result:
    mov rcx, [rel nfiles]
    lea rax, [rel lines_arr]
    mov rdx, [rel cur_lines]
    mov [rax + rcx*8], rdx
    lea rax, [rel words_arr]
    mov rdx, [rel cur_words]
    mov [rax + rcx*8], rdx
    lea rax, [rel bytes_arr]
    mov rdx, [rel cur_bytes]
    mov [rax + rcx*8], rdx
    lea rax, [rel names_arr]
    mov [rax + rcx*8], r15
    inc qword [rel nfiles]
    ret

; ---------------------------------------------------------------------------
; num_width —— rdi = 非负整数 → rax = 十进制位数（至少 1）
; ---------------------------------------------------------------------------
num_width:
    push rbx
    mov rax, rdi
    mov rbx, 10
    mov ecx, 1
.loop:
    cmp rax, rbx
    jb .done
    xor edx, edx
    div rbx                     ; rax /= 10
    inc ecx
    jmp .loop
.done:
    mov eax, ecx
    pop rbx
    ret

; ---------------------------------------------------------------------------
; widen —— rax = 候选宽度；比 [width] 大就更新 [width]
; ---------------------------------------------------------------------------
widen:
    cmp rax, [rel width]
    jbe .done
    mov [rel width], rax
.done:
    ret

; ---------------------------------------------------------------------------
; append_num —— 把 rdi（数值）按 rsi（列宽）右对齐追加到 [rbx]，rbx 前进
; ---------------------------------------------------------------------------
append_num:
    sub rsp, 32
    lea r8, [rsp + 32]          ; 缓冲区末尾（数字从后往前填）
    mov r9, r8
    mov rax, rdi
    mov r10, 10
.digits:
    xor edx, edx
    div r10                     ; rax /= 10，rdx = 余数
    add dl, '0'
    dec r9
    mov [r9], dl
    test rax, rax
    jnz .digits
    ; 位数 = r8 - r9；需要补的空格 = 宽度 - 位数
    mov rcx, r8
    sub rcx, r9
    mov rax, rsi
    sub rax, rcx
    jle .copy
.pad:
    mov byte [rbx], ' '
    inc rbx
    dec rax
    jnz .pad
.copy:
    mov rcx, r8
    sub rcx, r9
.copy_loop:
    mov dl, [r9]
    mov [rbx], dl
    inc r9
    inc rbx
    dec rcx
    jnz .copy_loop
    add rsp, 32
    ret

; ---------------------------------------------------------------------------
; print_line —— 打印一行：行数 / 单词数 / 字节数 / 文件名
;   入参：r12=行数 r13=单词数 r14=字节数 r15=文件名指针（0 = 不打印）
; ---------------------------------------------------------------------------
print_line:
    push rbx
    push r12
    push r13
    push r14
    push r15
    lea rbx, [rel linebuf]
    mov rdi, r12
    mov rsi, [rel width]
    call append_num
    mov byte [rbx], ' '
    inc rbx
    mov rdi, r13
    mov rsi, [rel width]
    call append_num
    mov byte [rbx], ' '
    inc rbx
    mov rdi, r14
    mov rsi, [rel width]
    call append_num
    test r15, r15
    jz .no_name
    mov byte [rbx], ' '
    inc rbx
.name_loop:
    mov al, [r15]
    test al, al
    jz .no_name
    mov [rbx], al
    inc rbx
    inc r15
    jmp .name_loop
.no_name:
    mov byte [rbx], LF
    inc rbx
    lea rsi, [rel linebuf]
    mov rdx, rbx
    sub rdx, rsi
    mov rdi, STDOUT
    call io_write
    pop r15
    pop r14
    pop r13
    pop r12
    pop rbx
    ret

; ---------------------------------------------------------------------------
; report_error —— rsi = 文件名；stderr 输出 "wordcount: <name>: cannot open"
; ---------------------------------------------------------------------------
report_error:
    push rbx
    mov rbx, rsi
    lea rsi, [rel msg_prefix]
    mov rdx, msg_prefix_len
    mov rdi, STDERR
    call io_write
    mov rdi, rbx
    call str_len
    mov rdx, rax
    mov rsi, rbx
    mov rdi, STDERR
    call io_write
    lea rsi, [rel msg_err]
    mov rdx, msg_err_len
    mov rdi, STDERR
    call io_write
    pop rbx
    ret

; ---------------------------------------------------------------------------
; str_len —— rdi = 字符串 → rax = 长度
; ---------------------------------------------------------------------------
str_len:
    xor eax, eax
.loop:
    cmp byte [rdi + rax], 0
    je .done
    inc rax
    jmp .loop
.done:
    ret
